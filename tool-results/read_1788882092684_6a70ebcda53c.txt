    76→function DifficultyPanel({ raw }: { raw: string | DifficultyProfile | undefined }) {
    77→  if (raw == null) {
    78→    return (
    79→      <div className="rp-wide mt-4 rounded-[10px] border border-[#e1e5df] bg-[#f4f6f0] px-3 py-2.5">

    82→          Difficulty profile
    83→        </div>
    84→        <p className="mt-1 text-[10px] leading-relaxed text-[#8a968f]">
    85→          Unrated — not computed for this revision. Rebuild with the upgraded compiler to analyze difficulty.
    86→        </p>
    87→      </div>
    88→    );
    89→  }
    90→  if (typeof raw === "string") {
    91→    return (
    92→      <div className="rp-wide mt-4 rounded-[10px] border border-[#e1e5df] bg-[#f4f6f0] px-3 py-2.5">
    93→        <div className="flex items-center gap-1.5 text-[10px] font-semibold text-[#778481]">
    94→          <Gauge className="size-3.5" aria-hidden />
    95→          Difficulty profile
    96→        </div>
    97→        <p className="mt-1 text-[10px] leading-relaxed text-[#8a968f]">Unrated</p>
    98→      </div>
    99→    );
   100→  }
   101→  const profile = raw;
   102→  const score = Math.max(0, Math.min(100, Math.round(num(profile.score) ?? 0)));
   103→  const tierIndex = Math.max(0, TIERS.findIndex((t) => score < t.max) === -1 ? TIERS.length - 1 : TIERS.findIndex((t) => score < t.max));
   104→  const metrics = profile.metrics ?? {};
   105→  const ambiguity = metrics.paletteAmbiguity;
   106→  const ambiguityTone =
   107→    ambiguity === "high"
   108→      ? "border-[#e8cfc7] bg-[#fdf3f1] text-[#ba463f]"
   109→      : ambiguity === "medium"
   110→        ? "border-[#e8d9b8] bg-[#fdf6e3] text-[#957242]"
   111→        : "border-[#cce7dc] bg-[#e5f3ed] text-[#087f74]";
   112→  const rows: Array<[string, string]> = [
   113→    ["Regions", num(metrics.regionCount)?.toLocaleString("en-US") ?? "—"],
   114→    ["Median region area", num(metrics.medianRegionArea) != null ? `${num(metrics.medianRegionArea)!.toLocaleString("en-US", { maximumFractionDigits: 1 })} px²` : "—"],
   115→    ["Tiny targets", num(metrics.tinyRegionPct) != null ? `${num(metrics.tinyRegionPct)!.toFixed(1)} %` : "—"],
   116→    ["Required zoom", num(metrics.requiredZoom) != null ? `${num(metrics.requiredZoom)!.toFixed(1)} ×` : "—"],
   117→    ["Palette groups", num(metrics.paletteGroups)?.toLocaleString("en-US") ?? "—"],
   118→    ["Avg neighbours", num(metrics.avgNeighbors) != null ? num(metrics.avgNeighbors)!.toFixed(2) : "—"],
   119→    ["Subdivision edges", num(metrics.subdivisionEdges)?.toLocaleString("en-US") ?? "—"],
   120→    ["Object density", num(metrics.objectDensity) != null ? num(metrics.objectDensity)!.toFixed(2) : "—"],
   121→    ["Label clearance", String(metrics.labelClearance ?? "—")],
   122→  ];
   123→  return (
   124→    <div className="rp-wide mt-4 rounded-[10px] border border-[#dfe6d8] bg-[#eff3ec] px-3 py-2.5">
   125→      <div className="flex items-center justify-between gap-2">
   126→        <div className="flex items-center gap-1.5 text-[10px] font-semibold text-[#6c7c6b]">
   127→          <Gauge className="size-3.5" aria-hidden />
   128→          Difficulty profile
   129→        </div>
   130→        <span
   131→          className="rounded-full border px-2 py-0.5 text-[9px] font-bold uppercase tracking-[0.08em]"
   132→          style={{ color: TIERS[tierIndex].color, borderColor: TIERS[tierIndex].color, background: "white" }}
   133→        >
   134→          {TIERS[tierIndex].label}
   135→        </span>
   136→      </div>
   137→      {/* 4-tier segmented bar: each block = 25 points; filled up to score */}
   138→      <div
   139→        className="mt-2 flex h-2.5 gap-0.5 overflow-hidden rounded-full"
   140→        role="img"
   141→        aria-label={`Difficulty score ${score} of 100 — ${TIERS[tierIndex].label}`}
   142→      >
   143→        {TIERS.map((t, i) => {
   144→          const fill = Math.max(0, Math.min(1, (score - i * 25) / 25));
   145→          return (
   146→            <div key={t.key} className="relative h-full flex-1 rounded-[2px] bg-[#e1e5df]" title={`${t.label}: ${i * 25}–${t.max}`}>
   147→              <div className="h-full rounded-[2px]" style={{ width: `${fill * 100}%`, background: t.color }} />
   148→            </div>
   149→          );
   150→        })}
   151→      </div>
   152→      <div className="mt-1 flex justify-between text-[9px] text-[#778481]">
   153→        <span className={tierIndex === 0 ? "font-bold text-[#087f74]" : ""}>Easy</span>
   154→        <span className={tierIndex === 1 ? "font-bold text-[#087f74]" : ""}>Medium</span>
   155→        <span className={tierIndex === 2 ? "font-bold text-[#087f74]" : ""}>Hard</span>
   156→        <span className={tierIndex === 3 ? "font-bold text-[#087f74]" : ""}>Master</span>
   157→      </div>
   158→      <p className="mt-1.5 text-[11px] font-bold text-[#183837]">
   159→        Score {score} / 100
   160→      </p>
   161→      {ambiguity != null && (
   162→        <span className={`mt-1.5 inline-block rounded-full border px-2 py-0.5 text-[9px] font-semibold capitalize ${ambiguityTone}`}>
   163→          Palette ambiguity · {String(ambiguity)}
   164→        </span>
   165→      )}
   166→      <dl className="mt-2 grid grid-cols-2 gap-x-2.5 gap-y-1 text-[10px] leading-snug">
   167→        {rows.map(([label, value]) => (
   168→          <div key={label} className="flex items-baseline justify-between gap-1.5">
   169→            <dt className="text-[#778481]">{label}</dt>
   170→            <dd className="font-semibold text-[#183837]">{value}</dd>
   171→          </div>
   172→        ))}
   173→      </dl>
   174→    </div>
   175→  );
   176→}

   178→function NumberField({
   179→  id,
   180→  label,
   181→  value,
   182→  min,
   183→  max,
   184→  step = 1,
   185→  onCommit,
   186→}: {
   187→  id: string;
   188→  label: string;
   189→  value: number;
   190→  min: number;
   191→  max: number;
   192→  step?: number;
   193→  onCommit: (v: number) => void;
   194→}) {
   195→  return (
   196→    <div>
   197→      <Label htmlFor={id} className="text-[10px] leading-snug text-[#657671]">
   198→        {label}
   199→      </Label>
   200→      <Input
   201→        id={id}
   202→        type="number"
   203→        min={min}
   204→        max={max}
   205→        step={step}
   206→        value={value}
   207→        className="mt-1 h-8 rounded-md bg-white text-xs"
   208→        onChange={(e) => {
   209→          const v = Number(e.target.value);
   210→          if (e.target.value !== "" && Number.isFinite(v)) onCommit(v);
   211→        }}
   212→        onBlur={(e) => {
   213→          const v = Number(e.target.value);
   214→          if (!Number.isFinite(v) || v < min || v > max) onCommit(Math.min(max, Math.max(min, Number.isFinite(v) ? v : min)));
   215→        }}
   216→      />
   217→    </div>
   218→  );
   219→}
