   910→        if not point_in_rings_rule(rings, r['label']['x'], r['label']['y'], rule):
   911→            outside_labels.append(r['id'])
   912→        # hit-test alignment: label point + bbox-inset probes must resolve to
   913→        # THIS region. With visible-region geometry no other region may own
   914→        # the point (fill-order independence: every region can be colored
   915→        # first without depending on another region's completion).
   916→        probes = [(r['label']['x'], r['label']['y'], True)]
   917→        x0, y0, x1, y1 = r['bbox']
   918→        for dx, dy in ((.25, .25), (.75, .25), (.25, .75), (.75, .75)):
   919→            probe = (x0 + dx * (x1 - x0), y0 + dy * (y1 - y0))
   920→            if point_in_rings_rule(rings, probe[0], probe[1], rule):
   921→                probes.append((probe[0], probe[1], False))
   922→        for px, py, is_label in probes:
   923→            hit_samples += 1
   924→            owners = [other for other, other_rings in prep
   925→                      if other is not r and other in g['regions']
   926→                      and other['bbox'][0] <= px <= other['bbox'][2]
   927→                      and other['bbox'][1] <= py <= other['bbox'][3]
   928→                      and point_in_rings_rule(other_rings, px, py,
   929→                                              other.get('fillRule', 'evenodd'))]
   930→            if source == 'svg-master' and owners:
   931→                if is_label:
   932→                    # Acceptance rule: every number must hit its own region.
   933→                    label_mismatches.append(f'{r["id"]}@{px:.1f},{py:.1f}→{owners[0]["id"]}')
   934→                z_overlaps += 1
   935→            elif owners:
   936→                hit_mismatches.append(f'{r["id"]}@{px:.1f},{py:.1f}→{owners[0]["id"]}')
   937→    if outside_labels:
   938→        errors.append('Labels outside region interiors: ' + ','.join(outside_labels[:5]))
   939→    if label_mismatches:
   940→        errors.append('Label ownership: number labels that resolve to a different region (visible-region geometry broken): '
   941→                      + '; '.join(label_mismatches[:5]))
   942→    # partition topology on the tolerance-flattened rings. Decorations are
   943→    # not interactive (no masks, no hit-testing), so the gameplay overlap
   944→    # gate measures playable regions only; raster keeps its stricter
   945→    # full-partition check (decorations were part of the SLIC partition).
   946→    if source == 'svg-master':
   947→        region_polys = [region_polygon(r) for r in g['regions']]
   948→        combined = unary_union([make_valid(poly) for poly in region_polys if not poly.is_empty])
   949→        total_area = sum(poly.area for poly in region_polys if not poly.is_empty)
   950→    else:
   951→        combined = unary_union([make_valid(poly) for poly in polys if not poly.is_empty])
   952→        total_area = sum(poly.area for poly in polys if not poly.is_empty)
   953→    overlap = max(0.0, total_area - combined.area)
   954→    missing = max(0.0, w * h - combined.area)
   955→    # Shared chains make the curved partition exact up to the fit tolerance:
   956→    # sub-tolerance slivers (<1px² total, e.g. one crossing on a 1px feature)
   957→    # are a documented band; real defects are orders of magnitude larger.
   958→    band = max(allowance + 0.01, 1.0)
   959→    if source == 'raster':
   960→        if missing > band or overlap > band:
   961→            errors.append(f'Region partition has missing ({missing:.2f}px) or overlapping ({overlap:.2f}px) area beyond tolerance {allowance:.2f}.')
   962→        elif overlap > 0.01 or missing > 0.01:
   963→            warnings.append(f'Partition deviation within the curve-fit tolerance band: overlap {overlap:.2f}px², missing {missing:.2f}px² (shared chains keep gaps bounded; check 1px features at zoom).')
   964→    else:
   965→        # Visible-region geometry: masks must not overlap beyond the fit
   966→        # tolerance band, otherwise coloring one region depends on another
   967→        # region's completion (fill-order independence breaks).
   968→        if overlap > band:
   969→            errors.append(f'SVG-master regions overlap by {overlap:.1f}px² (beyond tolerance {band:.2f}); '
   970→                          'visible-region subtraction failed - re-import the master.')
   971→        elif overlap > 0.01:
   972→            warnings.append(f'SVG-master regions overlap by {overlap:.1f}px² (within the fit tolerance band).')
   973→        if missing > 0.01:
   974→            warnings.append(f'SVG-master regions leave {missing:.1f}px² of the canvas uncovered; '
   975→                            'shading and decorative shapes are expected to be non-playable.')
   976→    empty_pixels = None
   977→    if roundtrip and source == 'raster':
   978→        back = _rasterize_regions(regs, w, h)
   979→        empty_pixels = int((back == 0).sum())
   980→        if empty_pixels > 8:
   981→            errors.append(f'Raster roundtrip left {empty_pixels} uncovered pixels.')
   982→        elif empty_pixels:
   983→            warnings.append(f'Raster roundtrip left {empty_pixels} sub-pixel boundary pixels uncovered (within flatten tolerance).')
   984→    small = [r['id'] for r in g['regions'] if 2 * r['label']['clearance'] * 360 / w * 8 < 24]
   985→    if small:
   986→        warnings.append(f'{len(small)} regions have less than a 24px inscribed target at 8x zoom on a 360px-wide canvas; inspect or merge them.')
   987→    if hit_mismatches:
   988→        warnings.append(f'Hit-test alignment: {len(hit_mismatches)}/{hit_samples} interior probes resolve to a lower region than expected: '
   989→                        + '; '.join(hit_mismatches[:4]))
   990→    if z_overlaps:
   991→        warnings.append(f'Hit-test alignment: {z_overlaps}/{hit_samples} interior probes lie under another region; '
   992→                        'visible-region geometry should own every tap surface - inspect the affected shapes.')
   993→    if paint.get('sourceColorShapeCount', 0) > 15000:
   994→        warnings.append('Detailed vector painting is heavy. Cache/rasterize its static layer in the game and test real devices.')
   995→    warnings.append('Automatic regions are drafts, not guaranteed to follow semantic object boundaries. Human visual review is required.')
   996→    fixed = sum(r['area'] for r in g.get('decorations', []))
   997→    report = {
   998→        'passed': not errors, 'errors': errors, 'warnings': warnings,
   999→        'playableRegions': len(g['regions']),
  1000→        'fixedRegions': len(g.get('decorations', [])), 'paletteGroups': len(p),
  1001→        'paintPaths': len(paint['paths']), 'paintSubshapes': paint.get('sourceColorShapeCount'),
  1002→        'area': total_area, 'canvasArea': w * h, 'overlapArea': overlap, 'missingArea': missing,
  1003→        'roundtripEmptyPixels': empty_pixels, 'smallTargetCount': len(small),
  1004→        'precoloredAreaPercent': round(100 * fixed / (w * h), 3), 'humanReviewed': False,
  1005→        'geometry': {
  1006→            'schema': g.get('geometrySchema', 1),
  1007→            'masterAuthority': 'regions[*].master.d (curved SVG path commands)',
  1008→            'flattenedDerivation': f'rings/flat.rings are derived at ±{flat_tol}px tolerance; not master geometry',
  1009→            'backend': g.get('backend', 'legacy'),
  1010→            'curveFitTolerance': g.get('curveFitTolerance'),
  1011→            'curvedCommands': curved_cmds, 'totalCommands': total_cmds,
  1012→            'curvedCommandRatio': round(curved_cmds / total_cmds, 4) if total_cmds else 0.0,
  1013→            'hitTestProbes': hit_samples, 'hitTestConflicts': len(hit_mismatches),
  1014→            'hitTestZOverlaps': z_overlaps, 'labelOwnershipConflicts': len(label_mismatches),
  1015→            'partitionToleranceAllowance': allowance,
  1016→            'fillRules': fill_rules,
  1017→            'visibleRegionGeometry': source == 'svg-master',
  1018→            'acceptanceRule': ('every number label hits its own region, and every region can be colored first '
  1019→                               'without depending on another region completion (non-overlapping masks)'),
  1020→        },
  1021→        'visualReview': {
  1022→            'required': True,
  1023→            'separateFromGeometryTests': True,
  1024→            'limitations': [
  1025→                'Geometry tests prove watertightness, label containment, bounds and hit-test alignment at the documented flatten tolerance; they cannot judge aesthetics.',
  1026→                'Whether traced boundaries follow the intended object edges (semantic quality) needs human review at 400%+ zoom.',
  1027→                'Curve-fit tolerance hides sub-tolerance detail; small deliberate features may be smoothed.',
  1028→                'Colour harmony, shading intent and gameplay difficulty are not machine-validated.',
  1029→            ],
  1030→        },
  1031→        'checkScope': 'IDs, palette refs, versions, closed curved paths, master/flat consistency, polygon validity, bounds, label interiors, '
  1032→                      'partition union/overlap at flatten tolerance, raster coverage, hit-test probe alignment, curve statistics. '
  1033→                      'Not semantic/artistic quality.',
  1034→    }
  1035→    return report

  1093→                         output_width=640, output_height=round(640 * h / w))
  1094→    m['checksums'] = {f: checksum(folder / f) for f in ['regions.json', 'palette.json', 'paint.json', 'colored.svg', 'numbered.svg', 'linework.svg', 'ink.svg']}
  1095→    write_json(folder / 'artwork.json', m)

  1216→def _split_disconnected_master(cmds: List[Command], fit_tolerance: float):
  1217→    """Split a master whose subpaths form disconnected tap targets.
  1218→
  1219→    Returns [(part_commands, part_flat_rings)] — one per connected geometry.
  1220→    """
  1221→    subs = subpaths_of(cmds)
  1222→    parts = []
  1223→    for sub in subs:
  1224→        flat = flatten_path(sub, FLATTEN_TOLERANCE)
  1225→        if flat:
  1226→            parts.append({'cmds': sub, 'rings': flat})
  1227→    # merge parts whose polygons touch/overlap (connected tap target)
  1228→    merged = []
  1229→    for part in parts:
  1230→        poly = Polygon(part['rings'][0], part['rings'][1:]) if len(part['rings']) > 1 else Polygon(part['rings'][0])
  1231→        part['poly'] = poly if poly.is_valid else make_valid(poly)
  1232→        placed = False
  1233→        for group in merged:
  1234→            if any(g['poly'].intersects(part['poly']) or g['poly'].touches(part['poly']) or
  1235→                   g['poly'].buffer(0.01).intersects(part['poly']) for g in group):
  1236→                group.append(part)
  1237→                placed = True
  1238→                break
  1239→        if not placed:
  1240→            merged.append([part])
  1241→    out = []
  1242→    for group in merged:
  1243→        cmds_out: List[Command] = []
  1244→        rings_out = []
  1245→        for part in group:
  1246→            cmds_out.extend(part['cmds'])
  1247→            rings_out.extend(part['rings'])
  1248→        out.append((cmds_out, rings_out))
  1249→    return out

  1252→def compile_svg_master(source: Path, output: Path, *, artwork_id: str, version: str, title: str,
  1253→                        settings: BuildSettings, provenance: dict | None = None,
  1254→                        progress: Callable = lambda *_: None) -> dict:
  1255→    """Compile a sanitized SVG master into the detailed-vector bundle.
  1256→
  1257→    Trust rules enforced here (review stage 1):
  1258→    - VISIBLE-REGION GEOMETRY: an opaque shape drawn above another covers it.
  1259→      Gameplay regions are the *visible surfaces* (own geometry minus the
  1260→      coverage of every later opaque shape), so region masks never overlap:
  1261→      every number hits its own region and every region can be colored
  1262→      first, whatever order the player fills in.
