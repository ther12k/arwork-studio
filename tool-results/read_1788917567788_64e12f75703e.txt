     1→/** Dependency-free region renderer. Copyright/provenance: see docs/PROVENANCE.md.
     2→ * Consume our trusted, validated JSON; do NOT inject arbitrary uploaded SVG markup.
     3→ * The demo is an offline solo asset tester, not an authoritative multiplayer server.
     4→ *
     5→ * SHARED FORMAT CONTRACT (detailed-vector schema 2) — this validator and the
     6→ * studio's src/lib/detailed-board.ts accept the same field set, and the studio's
     7→ * export pipeline (pipeline.validate_runtime_contract) mirrors these rules:
     8→ *  - geometry.geometrySchema 1 (legacy polygons) or 2 (curved masters)
     9→ *  - regions[*].d uses M/L/C/Q/Z path commands (curves preserved)
    10→ *  - regions[*].fillRule is 'evenodd' (default) or 'nonzero' (source rule kept)
    11→ *  - paint.paths[*] fill is #RRGGBB or url(#g-...) referencing paint.gradients
    12→ *  - paint paths carry optional fillRule/fillOpacity/opacity/stroke/strokeWidth/z
    13→ *  - paint.inkPaths may be OPEN paths (stroke line art) when strokeWidth/filled=false
    14→ *  - Regions are VISIBLE SURFACES: masks do not overlap, so any fill order
    15→ *    colors correctly (fill-order independence).
    16→ *  - geometry.edges (stage 2, optional): boundary entries {id, d (M/L/C/Q/Z,
    17→ *    open or closed), kind: 'artwork' | 'subdivision', leftRegion/rightRegion
    18→ *    (null or an existing region id)} plus optional geometry.boundaryStyle
    19→ *    overrides. When edges is a non-empty array, region paths render FILL-ONLY
    20→ *    and an edges overlay group (above masks/ink, below labels) draws the
    21→ *    boundaries: artwork = solid (geometry.stroke, width 1.6, round caps),
    22→ *    subdivision = light dashed (#7A8C94, width 0.85, dash '3 2.2' userSpace).
    23→ *    Absent/empty edges = legacy stroke rendering, unchanged.
    24→ *  - Free color: session.freeColors maps regionId -> #RRGGBB hex (flat fill,
    25→ *    no gradient); board.setFreeColor(hex) is the custom brush; palette swatch
    26→ *    selection in free mode sets the free color to that palette's hex; no
    27→ *    mistake counting in free mode. BoardState.customColor exposes the brush.
    28→ *  - Gestures are rAF-batched: pointermove/wheel store the latest event and run
    29→ *    ONE scheduled frame; label visibility updates pause during a gesture.
    30→ */

    40→const SAFE_PATH = /^M[\s\d.,eE+\-MLQCZ]+Z$/;
    41→const SAFE_PATH_OPEN = /^M[\s\d.,eE+\-MLQCZ]+$/;
    42→const SAFE_HEX = /^#[0-9A-Fa-f]{6}$/;
    43→const SAFE_GRADIENT_REF = /^url\(#g-[a-zA-Z0-9_-]+\)$/;
    44→const SAFE_GRADIENT_ID = /^g-[a-zA-Z0-9_-]+$/;
    45→const finite = (v) => Number.isFinite(v);

    47→export function validateBundle(bundle) {
    48→  const {manifest: m, geometry: g, palette: p} = bundle || {};
    49→  if (!m || !g || !Array.isArray(p) || !['color-duel-vector-1','color-duel-detailed-vector-1'].includes(m.format)) throw new Error('Unsupported artwork bundle');
    50→  if (m.id !== g.artworkId || m.version !== g.artworkVersion || m.regionCount !== g.regions?.length) throw new Error('Artwork identity/count mismatch');
    51→  if (!Array.isArray(g.viewBox) || g.viewBox.length !== 4 || g.viewBox.some(x => !Number.isFinite(x)) || g.viewBox[2] <= 0 || g.viewBox[3] <= 0) throw new Error('Invalid viewBox');
    52→  const paletteIds = new Set(p.map(x => x.id));
    53→  if (paletteIds.size !== p.length) throw new Error('Duplicate palette ID');
    54→  const ids = new Set();
    55→  for (const r of [...g.regions, ...g.decorations]) {
    56→    if (ids.has(r.id) || !/^[a-zA-Z0-9_-]+$/.test(r.id)) throw new Error('Duplicate or unsafe region ID');
    57→    ids.add(r.id);
    58→    const rule = r.fillRule === undefined ? 'evenodd' : r.fillRule;
    59→    if (!paletteIds.has(r.paletteId) || !['evenodd','nonzero'].includes(rule) || !SAFE_PATH.test(r.d)) throw new Error('Invalid region geometry or palette');
    60→  }
    61→  // Stage-2 boundary contract: optional edges + boundaryStyle (absent = legacy).
    62→  if (g.edges !== undefined) {
    63→    if (!Array.isArray(g.edges)) throw new Error('Invalid edges');
    64→    const edgeIds = new Set();
    65→    for (const e of g.edges) {
    66→      if (!e || typeof e !== 'object') throw new Error('Invalid edge entry');
    67→      if (!/^[a-zA-Z0-9_-]+$/.test(e.id) || edgeIds.has(e.id)) throw new Error('Duplicate or unsafe edge ID');
    68→      edgeIds.add(e.id);
    69→      if (typeof e.d !== 'string' || !SAFE_PATH_OPEN.test(e.d)) throw new Error('Invalid edge path');
    70→      if (!['artwork','subdivision'].includes(e.kind)) throw new Error('Invalid edge kind');
    71→      for (const side of ['leftRegion','rightRegion']) {
    72→        const v = e[side];
    73→        if (v !== undefined && v !== null && !ids.has(v)) throw new Error('Edge references unknown region');
    74→      }
    75→    }
    76→    if (g.boundaryStyle !== undefined) {
    77→      const bs = g.boundaryStyle;
    78→      if (!bs || typeof bs !== 'object' || Array.isArray(bs)) throw new Error('Invalid boundaryStyle');
    79→      for (const kind of ['artwork','subdivision']) {
    80→        const s = bs[kind];
    81→        if (s === undefined || s === null) continue;
    82→        if (typeof s !== 'object' || !SAFE_HEX.test(s.stroke) || !finite(s.strokeWidth) || s.strokeWidth <= 0) throw new Error(`Invalid boundaryStyle ${kind}`);
    83→        if (s.dash !== undefined && s.dash !== null && (typeof s.dash !== 'string' || s.dash.length > 40)) throw new Error(`Invalid boundaryStyle ${kind} dash`);
    84→      }
    85→    }
    86→  }
    87→  if (m.format === 'color-duel-detailed-vector-1') {
    88→    const paint = bundle.paint;
    89→    if (!paint || paint.artworkId !== m.id || !Array.isArray(paint.paths) || !Array.isArray(paint.inkPaths)) throw new Error('Missing detailed vector paint');
    90→    const gradientIds = new Set((paint.gradients ?? []).map(x => x.id));
    91→    for (const gr of paint.gradients ?? []) {
    92→      if (!SAFE_GRADIENT_ID.test(gr.id) || !['linear','radial'].includes(gr.type) || !Array.isArray(gr.stops) || !gr.stops.length) throw new Error('Invalid gradient definition');
    93→      for (const s of gr.stops) if (!SAFE_HEX.test(s.color) || !finite(s.offset)) throw new Error('Invalid gradient stop');
    94→    }
    95→    for (const path of paint.paths) {
    96→      const fill = path.fill;
    97→      const okFill = SAFE_HEX.test(fill) || (SAFE_GRADIENT_REF.test(fill) && gradientIds.has(fill.slice(5, -1)));
    98→      if (!okFill || !SAFE_PATH.test(path.d)) throw new Error('Invalid paint path');
    99→      if (path.fillRule !== undefined && !['evenodd','nonzero'].includes(path.fillRule)) throw new Error('Invalid paint fill rule');
   100→      for (const key of ['opacity','fillOpacity','strokeWidth','z']) {
   101→        const v = path[key];
   102→        if (v !== undefined && !finite(v)) throw new Error(`Invalid paint ${key}`);
   103→      }
   104→      if (path.opacity !== undefined && (path.opacity < 0 || path.opacity > 1)) throw new Error('Invalid paint opacity');
   105→      if (path.fillOpacity !== undefined && (path.fillOpacity < 0 || path.fillOpacity > 1)) throw new Error('Invalid paint fill-opacity');
   106→    }
   107→    // Ink layer holds closed filled shapes OR open stroke line art.
   108→    for (const path of paint.inkPaths) {
   109→      const fill = path.fill;
   110→      const okFill = SAFE_HEX.test(fill) || (SAFE_GRADIENT_REF.test(fill) && gradientIds.has(fill.slice(5, -1)));
   111→      const okPath = path.strokeWidth != null || path.filled === false ? SAFE_PATH_OPEN.test(path.d) : SAFE_PATH.test(path.d);
   112→      if (!okFill || !okPath) throw new Error('Invalid ink path');
   113→    }
   114→  }
   115→  return bundle;
   116→}

   118→/** URL must point at artwork.json. Relative paths are resolved beside that manifest. */
   119→export async function loadArtwork(manifestUrl, {signal} = {}) {
   120→  async function get(url) {
   121→    const res = await fetch(url, {signal});
   122→    if (!res.ok) throw new Error(`Artwork request failed: ${res.status} (${url})`);
   123→    return res.json();
   124→  }
   125→  const absolute = new URL(manifestUrl, document.baseURI).href;
   126→  const manifest = await get(absolute);
   127→  const [geometry, palette, paint] = await Promise.all([
   128→    get(new URL(manifest.assets.regions, absolute)),
   129→    get(new URL(manifest.assets.palette, absolute)),
   130→    manifest.assets.paint ? get(new URL(manifest.assets.paint, absolute)) : Promise.resolve(null)
   131→  ]);
   132→  return validateBundle({manifest, geometry, palette, paint});
   133→}

   135→/** Drop stale/unknown IDs; never restore progress across different geometry hashes. */
   136→export function normalizeSession(bundle, raw, mode = 'number') {
   137→  validateBundle(bundle);
   138→  const m = bundle.manifest, valid = new Set(bundle.geometry.regions.map(r => r.id));
   139→  const ids = new Set(mode === 'free' ? bundle.palette.map(p => p.id) : bundle.geometry.regions.map(r => r.paletteId));
   140→  const clean = {schemaVersion:1, artworkId:m.id, artworkVersion:m.version,
   141→    contentHash:m.contentHash, mode, completedRegionIds:[], selectedPaletteId:[...ids].sort((a,b)=>a-b)[0],
   142→    freeColors:{}, mistakes:0, updatedAt:new Date().toISOString()};
   143→  if (!raw || raw.schemaVersion !== 1 || raw.artworkId !== m.id || raw.artworkVersion !== m.version || raw.contentHash !== m.contentHash || raw.mode !== mode) return clean;
   144→  clean.completedRegionIds = [...new Set(Array.isArray(raw.completedRegionIds) ? raw.completedRegionIds.filter(id => valid.has(id)) : [])];
   145→  if (ids.has(raw.selectedPaletteId)) clean.selectedPaletteId = raw.selectedPaletteId;
   146→  if (Number.isSafeInteger(raw.mistakes) && raw.mistakes >= 0) clean.mistakes = raw.mistakes;
   147→  if (mode === 'free' && raw.freeColors && typeof raw.freeColors === 'object') {
   148→    // Stage-2 free colors: values are #RRGGBB hexes (legacy numeric values drop).
   149→    for (const id of clean.completedRegionIds) {
   150→      const c = raw.freeColors[id];
   151→      if (typeof c === 'string' && SAFE_HEX.test(c)) clean.freeColors[id] = c;
   152→    }
   153→    clean.completedRegionIds = clean.completedRegionIds.filter(id => id in clean.freeColors);
   154→  }
   155→  return clean;
   156→}

   158→export class VectorBoard {
   159→  constructor(svg, bundle, options = {}) {
   160→    if (!(svg instanceof SVGSVGElement)) throw new Error('Pass an <svg> element to VectorBoard');
   161→    validateBundle(bundle);
   162→    this.svg = svg; this.bundle = bundle; this.options = options;
   163→    this.mode = options.mode || 'number';
   164→    if (!['number','memory','free'].includes(this.mode)) throw new Error('Invalid coloring mode');
   165→    this.prefix = `cdv-${++sequence}-`;
   166→    this.handlers = []; this.pointers = new Map(); this.history = []; this.preview = false;
   167→    this.ctx = document.createElement('canvas').getContext('2d');
   168→    if (!this.ctx) throw new Error('Canvas hit-testing is unavailable');
   169→    this.regions = new Map(bundle.geometry.regions.map(r => [r.id, r]));
   170→    this.paths = new Map([...this.regions].map(([id,r]) => [id,new Path2D(r.d)]));
   171→    this.edges = Array.isArray(bundle.geometry.edges) && bundle.geometry.edges.length ? bundle.geometry.edges : null;
   172→    this.edgeMode = !!this.edges;
   173→    this.storageKey = `color-duel:vector:${bundle.manifest.id}:${bundle.manifest.version}:${bundle.manifest.contentHash}:${this.mode}`;
   174→    let stored = options.session || null;
   175→    if (!stored && options.persist !== false) {
   176→      try { stored = JSON.parse(localStorage.getItem(this.storageKey) || 'null'); } catch { this.storageWarning = true; }
   177→    }
   178→    this.session = normalizeSession(bundle,stored,this.mode);
   179→    this.completed = new Set(this.session.completedRegionIds);
   180→    this.base = [...bundle.geometry.viewBox]; this.view = [...this.base];
   181→    this.mount(); this.bindGestures(); this.refresh();
   182→  }
   183→  mount() {
   184→    this.svg.replaceChildren();
   185→    this.svg.setAttribute('viewBox', this.view.join(' '));
   186→    this.svg.setAttribute('aria-label', `${this.bundle.manifest.title}, interactive coloring artwork`);
   187→    this.svg.setAttribute('role','group');
   188→    this.svg.style.touchAction = 'none';
   189→    const defs = svgNode('defs');
   190→    for (const p of this.bundle.palette) {
   191→      const grad = svgNode('linearGradient',{id:`${this.prefix}paint-${p.id}`, x1:'0%',y1:'0%',x2:'100%',y2:'100%'});
   192→      for (const stop of p.paint.stops) grad.append(svgNode('stop',{offset:stop.offset,'stop-color':stop.color}));
   193→      defs.append(grad);
   194→    }
   195→    if (this.bundle.paint?.gradients) {
   196→      for (const g of this.bundle.paint.gradients) {
   197→        const attrs = g.type === 'linear'
   198→          ? {id: g.id, gradientUnits: 'userSpaceOnUse', x1: g.x1, y1: g.y1, x2: g.x2, y2: g.y2}
   199→          : {id: g.id, gradientUnits: 'userSpaceOnUse', cx: g.cx, cy: g.cy, r: g.r, fx: g.fx, fy: g.fy};
   200→        const node = svgNode(g.type === 'linear' ? 'linearGradient' : 'radialGradient', attrs);
   201→        for (const stop of g.stops) {
   202→          const stopAttrs = {offset: stop.offset, 'stop-color': stop.color};
   203→          if (stop.opacity != null && stop.opacity !== 1) stopAttrs['stop-opacity'] = stop.opacity;
   204→          node.append(svgNode('stop', stopAttrs));
   205→        }
   206→        defs.append(node);
   207→      }
   208→    }
   209→    const hatch = svgNode('pattern',{id:this.prefix+'selected',width:12,height:12,patternUnits:'userSpaceOnUse'});
   210→    hatch.append(svgNode('rect',{width:12,height:12,fill:'#EDF1F4'}),svgNode('path',{d:'M0 0H6V6H0Z M6 6H12V12H6Z',fill:'#C3CED4'}));
   211→    defs.append(hatch); this.svg.append(defs);
   212→    const g = this.bundle.geometry;
   213→    this.detailed = this.bundle.manifest.format === 'color-duel-detailed-vector-1';
   214→    if (this.detailed) {
   215→      // Appearance layer below the masks: filled paths in ORIGINAL drawing
   216→      // order (z), honouring per-path fill rule, opacity and strokes.
   217→      // (Stroke ink is rendered above the masks in the ink layer so the
   218→      // linework stays visible during play; see the numbered.svg preview.)
   219→      const art = svgNode('g',{'data-layer':'vector-paint','pointer-events':'none'});
   220→      for (const p of this.orderedPaint()) {
   221→        if (p.filled === false || (p.strokeWidth != null && !SAFE_HEX.test(p.fill))) {
   222→          continue; // stroke-only ink: rendered in the ink layer above masks
   223→        }
   224→        const gradientFill = p.fill.startsWith('url(#');
   225→        const attrs = {d: p.d, fill: p.fill, 'fill-rule': p.fillRule || 'evenodd'};
   226→        if (!gradientFill) { attrs.stroke = p.stroke || p.fill; attrs['stroke-width'] = p.strokeWidth ?? 0.55; attrs['stroke-linejoin'] = 'round'; }
   227→        if (p.fillOpacity != null && p.fillOpacity < 0.999) attrs['fill-opacity'] = p.fillOpacity;
   228→        if (p.opacity != null && p.opacity < 0.999) attrs.opacity = p.opacity;
   229→        if (p.stroke && p.strokeWidth > 0) { attrs.stroke = p.stroke; attrs['stroke-width'] = p.strokeWidth; attrs['stroke-linejoin'] = 'round'; }
   230→        art.append(svgNode('path', attrs));
   231→      }
   232→      this.svg.append(art);
   233→    }
   234→    const regions = svgNode('g',{'stroke': this.edgeMode ? 'none' : g.stroke,'stroke-width':g.strokeWidth,'stroke-linejoin':'round'});
   235→    this.elements = new Map(); this.labels = new Map();
   236→    for (const r of g.regions) {
   237→      const node = svgNode('path',{id:this.prefix+r.id,'data-region-id':r.id,'data-palette-id':r.paletteId,d:r.d,tabindex:0,role:'button',
   238→        'fill-rule': r.fillRule || 'evenodd',
   239→        'aria-label':`Region ${r.id}, palette ${this.mode==='memory'?'hidden':r.paletteId}`});
   240→      this.elements.set(r.id,node); regions.append(node);
   241→    }
   242→    this.svg.append(regions);
   243→    const fixed = svgNode('g',{'pointer-events':'none','stroke':g.stroke,'stroke-width':1.3,'stroke-linejoin':'round','fill-rule':'evenodd'});
   244→    for (const r of (this.detailed ? [] : g.decorations)) fixed.append(svgNode('path',{d:r.d,fill:`url(#${this.prefix}paint-${r.paletteId})`}));
   245→    this.svg.append(fixed);
   246→    const details = svgNode('g',{'pointer-events':'none',fill:'none','stroke-linejoin':'round','stroke-linecap':'round'});
   247→    for (const d of g.detailPaths) details.append(svgNode('path',{d:d.d,stroke:d.stroke,'stroke-width':d.strokeWidth,opacity:d.opacity}));
   248→    this.svg.append(details);
   249→    if (this.detailed) {
   250→      const ink = svgNode('g',{'data-layer':'ink','pointer-events':'none'});
   251→      for (const p of this.bundle.paint.inkPaths) {
   252→        if (p.strokeWidth != null || p.filled === false) {
   253→          ink.append(svgNode('path',{d:p.d, fill:'none', stroke:p.fill,
   254→            'stroke-width':p.strokeWidth ?? 1.5, 'stroke-linecap':'round', 'stroke-linejoin':'round'}));
   255→        } else {
   256→          ink.append(svgNode('path',{d:p.d, fill:p.fill, 'fill-rule':p.fillRule || 'evenodd',
   257→            ...(p.opacity != null && p.opacity < 0.999 ? {opacity: p.opacity} : {})}));
   258→        }
   259→      }
   260→      this.svg.append(ink);
   261→    }
   262→    if (this.edges) {
   263→      // Stage-2 edges overlay: ABOVE masks/ink, BELOW labels. Region paths
   264→      // render fill-only; boundaries are drawn here per kind (boundaryStyle
   265→      // overrides the documented defaults).
   266→      const bs = g.boundaryStyle || {};
   267→      const styleOf = (kind) => {
   268→        const base = kind === 'artwork'
   269→          ? {stroke: g.stroke || '#22333B', 'stroke-width': 1.6}
   270→          : {stroke: '#7A8C94', 'stroke-width': 0.85, 'stroke-dasharray': '3 2.2'};
   271→        const custom = bs[kind];
   272→        if (custom) {
   273→          if (custom.stroke !== undefined) base.stroke = custom.stroke;
   274→          if (custom.strokeWidth !== undefined) base['stroke-width'] = custom.strokeWidth;
   275→          if (custom.dash !== undefined) base['stroke-dasharray'] = custom.dash;
   276→        }
   277→        return base;
   278→      };
   279→      const styles = {artwork: styleOf('artwork'), subdivision: styleOf('subdivision')};
   280→      const overlay = svgNode('g',{'data-layer':'edges','pointer-events':'none','fill':'none','stroke-linecap':'round','stroke-linejoin':'round'});
   281→      for (const e of this.edges) {
   282→        const s = styles[e.kind] || styles.artwork;
   283→        overlay.append(svgNode('path',{id:this.prefix+e.id,'data-edge-kind':e.kind,d:e.d,...s}));
   284→      }
   285→      this.svg.append(overlay);
   286→    }
   287→    const labels = svgNode('g',{'pointer-events':'none','font-family':'Arial,sans-serif',fill:'#33444C','text-anchor':'middle','dominant-baseline':'central'});
   288→    for (const r of g.regions) {
   289→      const label = svgNode('text',{'data-label-for':r.id,x:r.label.x,y:r.label.y,'font-size':r.label.fontSize});
   290→      label.textContent = String(r.paletteId); labels.append(label); this.labels.set(r.id,label);
   291→    }
   292→    this.svg.append(labels);
   293→    this.listen(this.svg,'keydown',e => {
   294→      const id = e.target.getAttribute?.('data-region-id');
   295→      if (id && (e.key==='Enter' || e.key===' ')) {e.preventDefault();this.paint(id);}
   296→    });
   297→  }
   298→  /** Paint + ink entries merged into original drawing order (by z). */
   299→  orderedPaint() {
   300→    const paint = this.bundle.paint;
   301→    if (!paint) return [];
   302→    const entries = [...(paint.paths ?? []), ...(paint.inkPaths ?? [])];
   303→    return entries.sort((a, b) => (a.z ?? Infinity) - (b.z ?? Infinity) || 0);
   304→  }
   305→  listen(el,type,fn,options) {el.addEventListener(type,fn,options);this.handlers.push(()=>el.removeEventListener(type,fn,options));}
   306→  state() {
   307→    const completedRegionIds = [...this.completed];
   308→    const objects = Object.fromEntries(this.bundle.manifest.objectGroups.map(group => [group.id,
   309→      {completed:group.regionIds.filter(id=>this.completed.has(id)).length,total:group.regionIds.length}]));
   310→    return {...this.session, completedRegionIds, freeColors:{...this.session.freeColors},
   311→      customColor: this.session.freeColor ?? null,
   312→      total:this.regions.size, completed:completedRegionIds.length, progress:completedRegionIds.length/this.regions.size,
   313→      objects,preview:this.preview,zoom:this.base[2]/this.view[2],storageWarning:!!this.storageWarning};
   314→  }
   315→  save() {
   316→    this.session.completedRegionIds = [...this.completed]; this.session.updatedAt = new Date().toISOString();
   317→    if (this.options.persist !== false) try { localStorage.setItem(this.storageKey,JSON.stringify(this.session)); }
   318→      catch { this.storageWarning = true; }
   319→  }
   320→  notify(reason) {this.options.onChange?.(this.state(),reason);}
   321→  refresh() {
   322→    for (const [id,r] of this.regions) {
   323→      const done = this.completed.has(id), node = this.elements.get(id), label = this.labels.get(id);
   324→      let fill = '#FFFFFF';
   325→      if (this.preview || done) {
   326→        if (this.mode==='free' && !this.preview) {
   327→          fill = this.session.freeColors[id] || '#FFFFFF';   // flat hex, not a gradient
   328→        } else {
   329→          fill = `url(#${this.prefix}paint-${r.paletteId})`;
   330→        }
   331→      } else if (this.mode==='number' && r.paletteId===this.session.selectedPaletteId) fill=`url(#${this.prefix}selected)`;
   332→      if (this.detailed && (this.preview || (done && this.mode!=='free'))) fill='none';
   333→      node.setAttribute('fill',fill);
   334→      node.setAttribute('stroke',this.edgeMode || (this.detailed && (done || this.preview)) ? 'none' : this.bundle.geometry.stroke);
   335→      node.dataset.completed=String(done);
   336→      node.setAttribute('tabindex',done && this.mode!=='free' ? '-1':'0');
   337→      node.setAttribute('aria-pressed',String(done));
   338→      label.style.display = done || this.preview || this.mode!=='number' ? 'none':'';
   339→    }
   340→    this.updateLabelVisibility();
   341→    this.notify('render');
   342→  }
   343→  updateLabelVisibility() {
   344→    if (!this.detailed || !this.labels) return;
   345→    if (this.pointers && this.pointers.size) return;   // skip while a gesture is active
   346→    const screen = this.svg.getBoundingClientRect();
   347→    const scale = Math.min(screen.width/this.view[2],screen.height/this.view[3]);
   348→    for (const [id,r] of this.regions) {
   349→      const visible = !this.completed.has(id) && !this.preview && this.mode==='number' && r.label.fontSize*scale >= 9;
   350→      this.labels.get(id).style.display = visible ? '' : 'none';
   351→    }
   352→  }
   353→  setPalette(id) {
   354→    const entry = this.bundle.palette.find(p=>p.id===id);
   355→    if (!entry) throw new Error('Unknown palette ID');
   356→    if (this.mode==='free') {
   357→      // Palette swatches in free mode are quick access to that palette's hex.
   358→      this.session.freeColor = entry.hex; this.save(); this.refresh(); return;
   359→    }
   360→    this.session.selectedPaletteId=id; this.save(); this.refresh();
   361→  }
   362→  /** Free-mode custom color brush: any #RRGGBB hex (flat fill, no gradient). */
   363→  setFreeColor(hex) {
   364→    if (typeof hex !== 'string' || !SAFE_HEX.test(hex)) throw new Error('Use a #RRGGBB hex color');
   365→    this.session.freeColor = hex; this.save(); this.refresh();
   366→  }
   367→  paletteHex(id) { const p = this.bundle.palette.find(q=>q.id===id); return p ? p.hex : null; }
   368→  paint(id) {
   369→    const r=this.regions.get(id);
   370→    if (!r || this.preview) return 'ignored';
   371→    if (this.completed.has(id) && this.mode!=='free') return 'already-complete';
   372→    if (this.mode!=='free' && r.paletteId!==this.session.selectedPaletteId) {
   373→      this.session.mistakes++;this.save();this.notify('wrong-color');return 'wrong-color';
   374→    }
   375→    if (this.mode==='free') {
   376→      // Free mode: store the active hex (custom brush or the selected palette's
   377→      // hex); repainting with a different color re-colors; NO mistake counting.
   378→      const hex = this.session.freeColor || this.paletteHex(this.session.selectedPaletteId) || '#9AA7AD';
   379→      if (this.completed.has(id) && this.session.freeColors[id]===hex) return 'already-complete';
   380→      this.history.push({id,wasCompleted:this.completed.has(id),oldColor:this.session.freeColors[id]});
   381→      this.completed.add(id);
   382→      this.session.freeColors[id]=hex;
   383→      this.save();this.refresh();this.notify('paint');return 'painted';
   384→    }
   385→    this.history.push({id,wasCompleted:this.completed.has(id),oldColor:this.session.freeColors[id]});
   386→    this.completed.add(id);
   387→    this.save();this.refresh();this.notify('paint');return 'painted';
   388→  }
   389→  undo() {
   390→    if (this.preview) return false;
   391→    const previous=this.history.pop(); if (!previous) return false;
   392→    if (!previous.wasCompleted) this.completed.delete(previous.id);
   393→    if (previous.oldColor===undefined) delete this.session.freeColors[previous.id]; else this.session.freeColors[previous.id]=previous.oldColor;
   394→    this.save();this.refresh();return true;
   395→  }
   396→  reset() {
   397→    this.completed.clear();this.session.freeColors={};this.session.mistakes=0;this.history=[];this.preview=false;
   398→    this.save();this.refresh();this.fit();
   399→  }
   400→  setPreview(value) {this.preview=!!value;this.refresh();}
   401→  hitTest(x,y) {
   402→    // Iterate in reverse document order (topmost wins). Regions are visible
   403→    // surfaces and must not overlap; per-region fill rules are honoured.
   404→    for (const [id,r] of [...this.regions].reverse()) {
   405→      const b=r.bbox;
   406→      const rule = r.fillRule || 'evenodd';
   407→      if (x>=b[0] && x<=b[2] && y>=b[1] && y<=b[3] && this.ctx.isPointInPath(this.paths.get(id),x,y,rule)) return id;
   408→    }
   409→    return null;
   410→  }
   411→  clientToArt(x,y,matrix=null) {
   412→    const ctm=matrix || this.svg.getScreenCTM()?.inverse();
   413→    if (!ctm) return null;
   414→    return new DOMPoint(x,y).matrixTransform(ctm);
   415→  }
   416→  applyView(view) {
   417→    const [bx,by,bw,bh]=this.base;
   418→    const w=clamp(view[2],bw/10,bw),h=w*bh/bw;
   419→    this.view=[clamp(view[0],bx,bx+bw-w),clamp(view[1],by,by+bh-h),w,h];
   420→    this.svg.setAttribute('viewBox',this.view.join(' '));this.updateLabelVisibility();this.notify('viewport');
   421→  }
   422→  fit() {this.applyView([...this.base]);}
   423→  zoom(factor,anchor=null) {
   424→    const v=this.view, a=anchor || {x:v[0]+v[2]/2,y:v[1]+v[3]/2};
   425→    const w=clamp(v[2]/factor,this.base[2]/10,this.base[2]),h=w*this.base[3]/this.base[2];
   426→    this.applyView([a.x-(a.x-v[0])*w/v[2],a.y-(a.y-v[1])*h/v[3],w,h]);
   427→  }
   428→  nextRegion() {
   429→    if (this.mode!=='number' || this.preview) return null;
   430→    const r=[...this.regions.values()].find(r=>!this.completed.has(r.id) && r.paletteId===this.session.selectedPaletteId);
   431→    if (!r) return null;
   432→    const w=Math.max(this.base[2]/10,Math.min(this.base[2],Math.max(r.bbox[2]-r.bbox[0],(r.bbox[3]-r.bbox[1])*this.base[2]/this.base[3])*1.6));
   433→    this.applyView([r.label.x-w/2,r.label.y-w*this.base[3]/this.base[2]/2,w,w*this.base[3]/this.base[2]]);
   434→    this.elements.get(r.id).focus({preventScroll:true});return r.id;
   435→  }
   436→  bindGestures() {
   437→    // rAF-batched pointermove/wheel: store the latest event, run ONE scheduled
   438→    // frame per batch (high-frequency gestures no longer thrash the DOM).
   439→    this._rafPending=false; this._lastMove=null; this._lastWheel=null;
   440→    const raf = typeof requestAnimationFrame === 'function' ? requestAnimationFrame : (cb)=>setTimeout(cb,16);
   441→    const schedule = (fn) => {
   442→      if (this._rafPending) return;
   443→      this._rafPending = true;
   444→      raf(() => { this._rafPending = false; fn(); });
   445→    };
   446→    const startPinch=()=>{
   447→      const pts=[...this.pointers.values()];
   448→      const mid={x:(pts[0].x+pts[1].x)/2,y:(pts[0].y+pts[1].y)/2};
   449→      const inverse=this.svg.getScreenCTM().inverse();
   450→      this.pinch={distance:Math.max(1,Math.hypot(pts[0].x-pts[1].x,pts[0].y-pts[1].y)),view:[...this.view],inverse,anchor:this.clientToArt(mid.x,mid.y,inverse)};
   451→      this.suppressTap=true;
   452→    };
   453→    this.listen(this.svg,'pointerdown',e=>{
   454→      if (e.pointerType==='mouse' && e.button!==0)return;
   455→      this.pointers.set(e.pointerId,{x:e.clientX,y:e.clientY});
   456→      try { this.svg.setPointerCapture(e.pointerId); } catch { /* synthetic/stale pointer */ }
   457→      if (this.pointers.size===1) {
   458→        this.suppressTap=false;
   459→        const inverse=this.svg.getScreenCTM().inverse();
   460→        this.drag={x:e.clientX,y:e.clientY,view:[...this.view],inverse,anchor:this.clientToArt(e.clientX,e.clientY,inverse),moved:false};
   461→      } else if (this.pointers.size===2) startPinch();
   462→    });
   463→    this.listen(this.svg,'pointermove',e=>{
   464→      if (!this.pointers.has(e.pointerId))return;
   465→      this.pointers.set(e.pointerId,{x:e.clientX,y:e.clientY});
   466→      this._lastMove = e;
   467→      schedule(() => { const ev = this._lastMove; if (ev) this.handleMove(ev); });
   468→    });
   469→    const finish=(e,cancel=false)=>{
   470→      if (!this.pointers.has(e.pointerId))return;
   471→      const tap=!cancel && this.pointers.size===1 && !this.suppressTap && !this.drag?.moved;
   472→      this.pointers.delete(e.pointerId);
   473→      if (this.svg.hasPointerCapture(e.pointerId))this.svg.releasePointerCapture(e.pointerId);
   474→      if (tap) {const p=this.clientToArt(e.clientX,e.clientY);if(p)this.paint(this.hitTest(p.x,p.y));}
   475→      if (!this.pointers.size) {this.drag=null;this.pinch=null;this.suppressTap=false;} else this.suppressTap=true;
   476→    };
   477→    this.listen(this.svg,'pointerup',e=>finish(e));
   478→    this.listen(this.svg,'pointercancel',e=>finish(e,true));
   479→    this.listen(this.svg,'wheel',e=>{
   480→      e.preventDefault();                       // must stay synchronous
   481→      this._lastWheel = e;
   482→      schedule(() => {
   483→        const ev = this._lastWheel; if (!ev) return;
   484→        this.zoom(Math.exp(-ev.deltaY*.0015), this.clientToArt(ev.clientX,ev.clientY));
   485→      });
   486→    },{passive:false});
   487→  }
   488→  /** Deferred per-frame gesture math (pan + pinch); runs inside one rAF. */
   489→  handleMove(e) {
   490→    if (this.pointers.size>=2 && this.pinch) {
   491→      const pts=[...this.pointers.values()], mid={x:(pts[0].x+pts[1].x)/2,y:(pts[0].y+pts[1].y)/2};
   492→      const distance=Math.max(1,Math.hypot(pts[0].x-pts[1].x,pts[0].y-pts[1].y));
   493→      const p=this.pinch, now=this.clientToArt(mid.x,mid.y,p.inverse),w=clamp(p.view[2]*p.distance/distance,this.base[2]/10,this.base[2]),h=w*this.base[3]/this.base[2];
   494→      this.applyView([p.anchor.x-(now.x-p.view[0])*w/p.view[2],p.anchor.y-(now.y-p.view[1])*h/p.view[3],w,h]);
   495→    } else if (!this.suppressTap && this.drag) {
   496→      const d=this.drag;
   497→      if (Math.hypot(e.clientX-d.x,e.clientY-d.y)>6) d.moved=true;
   498→      if (d.moved) {const now=this.clientToArt(e.clientX,e.clientY,d.inverse);this.applyView([d.view[0]-(now.x-d.anchor.x),d.view[1]-(now.y-d.anchor.y),d.view[2],d.view[3]]);}
   499→    }
   500→  }
   501→  destroy() {for(const remove of this.handlers)remove();this.handlers=[];this.svg.replaceChildren();this.pointers.clear();}
   502→}
