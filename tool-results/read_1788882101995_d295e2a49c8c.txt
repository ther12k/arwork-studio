  2043→def _tint_gradient(gradient: dict, target_hex: str) -> None:
  2044→    """Shift a gradient's stop colors toward ``target_hex`` while keeping the
  2045→    stop-to-stop shading relationship (light/dark structure is preserved)."""
  2046→    stops = gradient.get('stops') or []
  2047→    if not stops:
  2048→        return
  2049→    avg = [0.0, 0.0, 0.0]
  2050→    for s in stops:
  2051→        c = s.get('color', '#808080').lstrip('#')
  2052→        avg[0] += int(c[0:2], 16); avg[1] += int(c[2:4], 16); avg[2] += int(c[4:6], 16)
  2053→    avg = [v / len(stops) for v in avg]
  2054→    tgt = [int(target_hex.lstrip('#')[k:k + 2], 16) for k in (0, 2, 4)]
  2055→    ratios = [(tgt[k] / avg[k]) if avg[k] > 1e-6 else (1.0 if tgt[k] > 0 else 0.4) for k in range(3)]
  2056→    for s in stops:
  2057→        c = s.get('color', '#808080').lstrip('#')
  2058→        rgb = [min(255, max(0, round(int(c[k:k + 2], 16) * ratios[k // 2]))) for k in (0, 2, 4)]
  2059→        s['color'] = '#' + ''.join(f'{v:02X}' for v in rgb)

  2062→def _recolor_bundle(bundle: dict, chosen, color: str, preserve_shading: bool) -> None:
  2063→    """Recolor the *visible appearance* of the chosen regions' source shapes.
  2064→
  2065→    Separate from the 'palette' action (which assigns the number group):
  2066→    this changes what the player sees. ``preserve_shading`` keeps gradient
  2067→    shading (tinted toward the target); otherwise the fill is replaced.
  2068→    """
  2069→    paint = bundle['paint']
  2070→    shape_ids = set()
  2071→    for r in chosen:
  2072→        sid = r.get('masterShapeId')
  2073→        if not sid:
  2074→            raise ValueError(f'Region {r["id"]} has no paintable source shape; '
  2075→                             'raster-built bundles can only be re-colored by rebuilding.')
  2076→        shape_ids.add(sid)
  2077→    paths = [p for p in (paint.get('paths') or []) if p.get('shapeId') in shape_ids]
  2078→    if not paths:
  2079→        raise ValueError('No paint paths found for the selected regions.')
  2080→    grad_ids = {p['fill'][5:-1] for p in paths if str(p.get('fill', '')).startswith('url(#')}
  2081→    for path in paths:
  2082→        if preserve_shading and str(path.get('fill', '')).startswith('url(#'):
  2083→            continue                      # tinted through the gradient below
  2084→        path['fill'] = color
  2085→    if preserve_shading and grad_ids:
  2086→        for gradient in (paint.get('gradients') or []):
  2087→            if gradient.get('id') in grad_ids:
  2088→                _tint_gradient(gradient, color)
  2089→    # keep the palette swatch in sync with the new appearance
  2090→    pids = {r['paletteId'] for r in chosen}
  2091→    for entry in bundle['palette']:
  2092→        if entry['id'] in pids:
  2093→            entry['hex'] = color
  2094→            entry['paint'] = {'type': 'linearGradient',
  2095→                              'stops': [{'offset': 0, 'color': color}, {'offset': 1, 'color': color}]}

  2289→def validate_runtime_contract(bundle: dict) -> list:
  2290→    """Mirror of the shipped game adapter's validateBundle (shared contract).
  2291→
  2292→    Both sides must accept exactly the same field set: region ids, palette
  2293→    references, per-region fill rules, curved M/L/C/Q/Z path data, paint
  2294→    fills (#hex or url(#g-...)), gradients, strokes and open ink paths.
  2295→    Runs on every export so "passed validation" implies "the game can load it".
  2296→    """
  2297→    errors = []
  2298→    m, g, p, paint = bundle['manifest'], bundle['geometry'], bundle['palette'], bundle['paint']
  2299→    if m.get('format') not in ('color-duel-vector-1', SCHEMA):
  2300→        errors.append('Unsupported artwork bundle format.')
  2301→    if m['id'] != g.get('artworkId') or m['version'] != g.get('artworkVersion') \
  2302→            or m.get('regionCount') != len(g.get('regions', [])):
  2303→        errors.append('Artwork identity/count mismatch.')
  2304→    vb = g.get('viewBox')
  2305→    if not isinstance(vb, list) or len(vb) != 4 or not all(isinstance(v, (int, float)) for v in vb) \
  2306→            or vb[2] <= 0 or vb[3] <= 0:
  2307→        errors.append('Invalid viewBox.')
  2308→    palette_ids = set()
  2309→    for entry in p:
  2310→        if entry['id'] in palette_ids:
  2311→            errors.append('Duplicate palette ID.')
  2312→        palette_ids.add(entry['id'])
  2313→    grad_ids = {gr.get('id') for gr in (paint.get('gradients') or [])}
  2314→    seen = set()
  2315→    for r in [*g.get('regions', []), *g.get('decorations', [])]:
  2316→        if r['id'] in seen or not SAFE_ID.match(r['id']):
  2317→            errors.append('Duplicate or unsafe region ID.')
  2318→        seen.add(r['id'])
  2319→        if r.get('paletteId') not in palette_ids:
  2320→            errors.append(f'Unknown palette group: {r["id"]}')
  2321→        rule = r.get('fillRule', 'evenodd')
  2322→        if rule not in ('evenodd', 'nonzero'):
  2323→            errors.append(f'Unsupported region fill rule: {r["id"]}')
  2324→        if not SAFE_D.match(r['d']):
  2325→            errors.append(f'Invalid region path data: {r["id"]}')
  2326→    for path in (paint.get('paths') or []):
  2327→        fill = path.get('fill', '')
  2328→        ok_fill = SAFE_HEX.match(fill) or (SAFE_GRAD_REF.match(fill) and fill[5:-1] in grad_ids)
  2329→        if not ok_fill:
  2330→            errors.append('Invalid paint fill.')
  2331→        if not SAFE_D.match(path['d']):
  2332→            errors.append('Invalid paint path.')
  2333→        if path.get('fillRule') not in (None, 'evenodd', 'nonzero'):
  2334→            errors.append('Invalid paint fill rule.')
  2335→        for key in ('opacity', 'fillOpacity', 'strokeWidth', 'z'):
  2336→            v = path.get(key)
  2337→            if v is not None and (not isinstance(v, (int, float)) or not (0 <= v if key != 'z' else True)):
  2338→                errors.append(f'Invalid paint {key}.')
  2339→    for path in (paint.get('inkPaths') or []):
  2340→        fill = path.get('fill', '')
  2341→        ok_fill = SAFE_HEX.match(fill) or (SAFE_GRAD_REF.match(fill) and fill[5:-1] in grad_ids)
  2342→        open_ok = (path.get('strokeWidth') is not None or path.get('filled') is False) and SAFE_D_OPEN.match(path['d'])
  2343→        if not ok_fill:
  2344→            errors.append('Invalid ink fill.')
  2345→        if not (SAFE_D.match(path['d']) or open_ok):
  2346→            errors.append('Invalid ink path.')
  2347→    for gr in (paint.get('gradients') or []):
  2348→        if not _re.match(r'^g-[a-zA-Z0-9_-]+$', gr.get('id', '')):
  2349→            errors.append('Unsafe gradient id.')
  2350→        if gr.get('type') not in ('linear', 'radial') or not gr.get('stops'):
  2351→            errors.append('Invalid gradient definition.')
  2352→    # Optional edges (stage-2 boundary contract): id/path/kind + region refs
  2353→    # must exist; boundaryStyle overrides must stay shapely-safe.
  2354→    edges = g.get('edges')
  2355→    if edges is not None:
  2356→        if not isinstance(edges, list):
  2357→            errors.append('Invalid edges.')
  2358→        else:
  2359→            edge_ids = set()
  2360→            for e in edges:
  2361→                if not isinstance(e, dict):
  2362→                    errors.append('Invalid edge entry.')
  2363→                    continue
  2364→                eid = str(e.get('id', ''))
  2365→                if not SAFE_ID.match(eid) or eid in edge_ids:
  2366→                    errors.append(f'Duplicate or unsafe edge ID: {eid}')
  2367→                edge_ids.add(eid)
  2368→                if not isinstance(e.get('d'), str) or not SAFE_D_OPEN.match(e['d']):
  2369→                    errors.append(f'Invalid edge path data: {eid}')
  2370→                if e.get('kind') not in EDGE_KINDS:
  2371→                    errors.append(f'Invalid edge kind: {eid}')
  2372→                for side in ('leftRegion', 'rightRegion'):
  2373→                    value = e.get(side)
  2374→                    if value is not None and value not in seen:
  2375→                        errors.append(f'Edge {eid} references an unknown region.')
  2376→            style = g.get('boundaryStyle')
  2377→            if style is not None:
  2378→                if not isinstance(style, dict):
  2379→                    errors.append('Invalid boundaryStyle.')
  2380→                else:
  2381→                    for kind in EDGE_KINDS:
  2382→                        part = style.get(kind)
  2383→                        if part is None:
  2384→                            continue
  2385→                        if (not isinstance(part, dict) or not isinstance(part.get('stroke'), str)
  2386→                                or not SAFE_HEX.match(part['stroke'])
  2387→                                or not isinstance(part.get('strokeWidth'), (int, float))
  2388→                                or not (part['strokeWidth'] > 0)):
  2389→                            errors.append(f'Invalid boundaryStyle.{kind}.')
  2390→                            continue
  2391→                        dash = part.get('dash')
  2392→                        if dash is not None and (not isinstance(dash, str) or len(dash) > 40):
  2393→                            errors.append(f'Invalid boundaryStyle.{kind}.dash.')
  2394→    return errors

  2397→def _runtime_geometry(g: dict) -> dict:
  2398→    """Lean runtime geometry: exactly what the game adapter needs."""
  2399→    keep = ('schemaVersion', 'geometrySchema', 'source', 'artworkId', 'artworkVersion',
  2400→            'viewBox', 'fillRule', 'stroke', 'strokeWidth', 'backend',
  2401→            'flattenTolerance', 'curveFitTolerance', 'partitionTolerance',
  2402→            'visibleRegionGeometry', 'detailPaths', 'edges', 'boundaryStyle')

  2404→    out['regions'] = [{k: r[k] for k in RUNTIME_REGION_KEYS if k in r} for r in g.get('regions', [])]
  2405→    out['decorations'] = []

  2416→def make_export(folder: Path, include_authoring=False):
  2417→    """Export the runtime bundle (lean) or the full authoring bundle.
  2418→
  2419→    The runtime export carries only the geometry, paint, palette, labels and
  2420→    metadata the game needs - no duplicate master/ring representations -
  2421→    and is contract-checked against the shipped game adapter rules before
  2422→    it is written, so a "passed validation" export actually loads.
  2423→    """
  2424→    m = read_json(folder / 'artwork.json')
  2425→    g = read_json(folder / 'regions.json')
  2426→    p = read_json(folder / 'palette.json')
  2427→    paint = read_json(folder / 'paint.json')
  2428→    master_name = m['assets'].get('sourceMaster', 'source-master.png')
  2429→    from io import BytesIO
  2430→    output = BytesIO()
  2431→    root = f'artworks/{m["id"]}/'
  2432→    export_m = json.loads(json.dumps(m))
  2433→    contract_errors = validate_runtime_contract({'manifest': m, 'geometry': g, 'palette': p, 'paint': paint})
  2434→    if contract_errors:
  2435→        raise ValueError('Runtime contract check failed: ' + '; '.join(contract_errors[:5]))
  2436→    if include_authoring:
  2437→        names = ['artwork.json', 'regions.json', 'palette.json', 'paint.json', 'colored.svg', 'numbered.svg',
  2438→                 'linework.svg', 'ink.svg', 'selected-preview.svg', 'thumbnail.webp', 'validation.json',
  2439→                 master_name, 'build-settings.json', 'colored-preview.png', 'numbered-preview.png']
  2440→        bundle_kind = 'authoring'
  2441→    else:
  2442→        g_out = _runtime_geometry(g)
  2443→        paint_out = _runtime_paint(paint)
  2444→        write_json(folder / '.runtime-regions.json', g_out)
  2445→        write_json(folder / '.runtime-paint.json', paint_out)
  2446→        slim = {'manifest': export_m, 'geometry': g_out, 'palette': p, 'paint': paint_out}
  2447→        slim_errors = validate_runtime_contract(slim)
  2448→        if slim_errors:
  2449→            (folder / '.runtime-regions.json').unlink(missing_ok=True)
  2450→            (folder / '.runtime-paint.json').unlink(missing_ok=True)
  2451→            raise ValueError('Lean runtime contract check failed: ' + '; '.join(slim_errors[:5]))
  2452→        export_m['assets'] = {k: v for k, v in export_m['assets'].items()
  2453→                              if k in ('regions', 'palette', 'paint')}
  2454→        export_m['exportKind'] = 'runtime'
  2455→        bundle_kind = 'runtime'
  2456→        export_m['runtime'] = {'contract': 'detailed-vector schema 2 (curves, gradients, per-region fill rules)',
  2457→                               'leanGeometry': True,
  2458→                               'contentHash': hashlib.sha256((folder / '.runtime-regions.json').read_bytes()
  2459→                                                             + (folder / '.runtime-paint.json').read_bytes()
  2460→                                                             + (folder / 'palette.json').read_bytes()).hexdigest()}
  2461→        export_m['contentHash'] = export_m['runtime']['contentHash']
  2462→        # ONLY what the game adapter loads: the four JSON files plus the
  2463→        # validation evidence. Preview SVGs and thumbnails stay in the
  2464→        # authoring export - the adapter renders from JSON, not from SVGs.
  2465→        names = ['artwork.json', 'validation.json']
  2466→    with zipfile.ZipFile(output, 'w', zipfile.ZIP_DEFLATED) as z:
  2467→        for name in names:
  2468→            if name == 'artwork.json':
  2469→                z.writestr(root + name, json.dumps(export_m, indent=2))
  2470→            elif (folder / name).is_file():
  2471→                z.write(folder / name, root + name)
  2472→        if not include_authoring:
  2473→            z.writestr(root + 'regions.json', (folder / '.runtime-regions.json').read_text(encoding='utf-8'))
  2474→            z.writestr(root + 'paint.json', (folder / '.runtime-paint.json').read_text(encoding='utf-8'))
  2475→            z.writestr(root + 'palette.json', (folder / 'palette.json').read_text(encoding='utf-8'))
  2476→        z.writestr('catalog-entry.json', json.dumps({'id': m['id'], 'title': m['title'], 'manifest': root + 'artwork.json', 'format': m['format'], 'status': m['qa']['status'], 'exportKind': bundle_kind}, indent=2))
  2477→        z.writestr('IMPORT.md', 'Load artwork.json and its regions/palette/paint files with the detailed-vector adapter (integration/detailed-board.mjs). '
  2478→                                'The runtime bundle was contract-checked against that adapter before export. '
  2479→                                'Geometry schema 2: regions[*].d is the AUTHORITATIVE curved path data (M/L/C/Q/Z); authoring duplicates (master/rings/flat/legacy) are stripped from the runtime export. '
  2480→                                'Regions are visible surfaces: masks do not overlap, so any fill order colors correctly. '
  2481→                                'geometry.edges (when present) carries boundary kinds: artwork = true master boundary (solid), subdivision = artificial gameplay boundary (light dashed); geometry.boundaryStyle overrides the default stroke/dash per kind; render region paths fill-only when edges exist. '
  2482→                                'Preserve viewBox, per-region fill rules (evenodd/nonzero), gradients, contentHash and version. Do not stretch a full painting into each region. Do not use numbered.svg as hit-test metadata. Validation does not establish copyright clearance.\n')
  2483→    (folder / '.runtime-regions.json').unlink(missing_ok=True)
  2484→    (folder / '.runtime-paint.json').unlink(missing_ok=True)
  2485→    return output.getvalue()
