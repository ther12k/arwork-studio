    62→def _attrs(elem: ET.Element) -> Dict[str, str]:
    63→    out: Dict[str, str] = {}

    69→def _num(value: str | None, default: float = 0.0) -> float:
    70→    if value is None:
    71→        return default
    72→    m = re.match(r'^\s*([-+]?(?:\d*\.\d+|\d+\.?)(?:[eE][-+]?\d+)?)', str(value))
    73→    if not m:
    74→        return default
    75→    v = float(m.group(1))
    76→    if not math.isfinite(v) or abs(v) > 1e6:
    77→        raise ValueError('SVG coordinate out of supported range.')
    78→    return v

    81→def _parse_color(value: str | None) -> Optional[str]:
    82→    """Normalize a supported CSS/SVG color to #RRGGBB, else None."""
    83→    if value is None:
    84→        return None
    85→    v = value.strip().lower()
    86→    if v in ('none', 'transparent', ''):
    87→        return None
    88→    if v.startswith('#'):
    89→        hexpart = v[1:]
    90→        if re.fullmatch(r'[0-9a-f]{3}', hexpart):
    91→            return '#' + ''.join(c + c for c in hexpart).upper()
    92→        if re.fullmatch(r'[0-9a-f]{6}', hexpart):
    93→            return '#' + hexpart.upper()
    94→        return None
    95→    m = re.fullmatch(r'rgb\(\s*(\d+)\s*,\s*(\d+)\s*,\s*(\d+)\s*\)', v)
    96→    if m:
    97→        r, g, b = (min(255, int(x)) for x in m.groups())
    98→        return f'#{r:02X}{g:02X}{b:02X}'
    99→    m = re.fullmatch(r'rgb\(\s*([\d.]+)%\s*,\s*([\d.]+)%\s*,\s*([\d.]+)%\s*\)', v)
   100→    if m:
   101→        r, g, b = (min(255, round(float(x) * 2.55)) for x in m.groups())
   102→        return f'#{r:02X}{g:02X}{b:02X}'
   103→    return _NAMED_COLORS.get(v)

   106→def _style_dict(style: str | None) -> Dict[str, str]:
   107→    if not style:
   108→        return {}
   109→    out: Dict[str, str] = {}
   110→    for chunk in style.split(';'):
   111→        if ':' not in chunk:
   112→            continue
   113→        key, value = chunk.split(':', 1)
   114→        key = key.strip().lower()
   115→        if key in _STYLE_KEYS:
   116→            out[key] = value.strip()
   117→    return out

   120→class _Context:
   121→    __slots__ = ('matrix', 'opacity', 'fill', 'fill_opacity', 'fill_rule', 'stroke', 'stroke_width')
   122→
   123→    def __init__(self) -> None:
   124→        self.matrix: Mat = mat_identity()
   125→        self.opacity = 1.0
   126→        self.fill: Optional[str] = '#000000'   # SVG default fill is black
   127→        self.fill_opacity: Optional[float] = 1.0
   128→        self.fill_rule = 'nonzero'
   129→        self.stroke: Optional[str] = None
   130→        self.stroke_width = 1.0
   131→
   132→    def child(self, attrs: Dict[str, str]) -> '_Context':
   133→        style = _style_dict(attrs.get('style'))
   134→        merged = dict(attrs)
   135→        merged.update(style)
   136→        ctx = _Context()
   137→        ctx.matrix = self.matrix
   138→        if 'transform' in merged:
   139→            ctx.matrix = mat_mul(self.matrix, parse_transform(merged['transform']))
   140→        ctx.opacity = self.opacity * _num(merged.get('opacity'), 1.0)
   141→        ctx.fill = _parse_color(merged['fill']) if 'fill' in merged else self.fill
   142→        if 'fill' in merged and merged.get('fill', '').strip().lower().startswith('url('):
   143→            ctx.fill = self.fill  # url() resolved later
   144→        fo = merged.get('fill-opacity')
   145→        ctx.fill_opacity = self.fill_opacity if fo is None else max(0.0, min(1.0, _num(fo, 1.0)))
   146→        ctx.fill_rule = merged.get('fill-rule', self.fill_rule).strip().lower() or 'nonzero'
   147→        ctx.stroke = _parse_color(merged.get('stroke')) if 'stroke' in merged else self.stroke
   148→        sw = merged.get('stroke-width')
   149→        ctx.stroke_width = self.stroke_width if sw is None else max(0.0, _num(sw, 1.0))
   150→        return ctx
   151→
   152→    def fill_ref(self, attrs: Dict[str, str]) -> Optional[str]:
   153→        """Gradient reference from fill="url(#id)" (attribute or inline style)."""
   154→        raw = (attrs.get('fill') or '').strip()
   155→        if not raw:
   156→            raw = (_style_dict(attrs.get('style')).get('fill') or '').strip()
   157→        raw = raw.strip()
   158→        if raw.lower().startswith('url('):
   159→            m = re.search(r'url\(\s*["\']?#([^"\')\s]+)["\']?\s*\)', raw)
   160→            if m:
   161→                return m.group(1)
   162→        return None

   165→def _rect_commands(x: float, y: float, w: float, h: float, rx: float, ry: float) -> List[Command]:
   166→    if w <= 0 or h <= 0:
   167→        raise ValueError('Degenerate rect.')
   168→    rx = max(0.0, min(rx if rx else ry, w / 2))
   169→    ry = max(0.0, min(ry if ry else rx, h / 2))
   170→    if rx <= 0 or ry <= 0:
   171→        return [('M', x, y), ('L', x + w, y), ('L', x + w, y + h), ('L', x, y + h), ('Z')]
   172→    # rounded rect: straight edges + cubic corners (kappa)
   173→    k = 0.5522847498
   174→    cmds: List[Command] = [
   175→        ('M', x + rx, y),
   176→        ('L', x + w - rx, y),
   177→        ('C', x + w - rx + k * rx, y, x + w, y + ry - k * ry, x + w, y + ry),
   178→        ('L', x + w, y + h - ry),
   179→        ('C', x + w, y + h - ry + k * ry, x + w - rx + k * rx, y + h, x + w - rx, y + h),
   180→        ('L', x + rx, y + h),
   181→        ('C', x + rx - k * rx, y + h, x, y + h - ry + k * ry, x, y + h - ry),
   182→        ('L', x, y + ry),
   183→        ('C', x, y + ry - k * ry, x + rx - k * rx, y, x + rx, y),
   184→        ('Z',
   185→        )]
   186→    return cmds

   189→def _ellipse_commands(cx: float, cy: float, rx: float, ry: float) -> List[Command]:
   190→    if rx <= 0 or ry <= 0:
   191→        raise ValueError('Degenerate ellipse.')
   192→    cmds: List[Command] = [('M', cx + rx, cy)]
   193→    for start in (0, math.pi / 2, math.pi, 3 * math.pi / 2):
   194→        arc = arc_to_cubics(cx + rx * math.cos(start), cy + ry * math.sin(start),
   195→                            rx, ry, 0, 0, 1,
   196→                            cx + rx * math.cos(start + math.pi / 2), cy + ry * math.sin(start + math.pi / 2))
   197→        cmds.extend(arc)
   198→    cmds.append(('Z',))
   199→    return cmds

   202→def _points_commands(raw: str, close: bool) -> List[Command]:
   203→    nums = re.findall(r'[-+]?(?:\d*\.\d+|\d+\.?)(?:[eE][-+]?\d+)?', raw or '')
   204→    if len(nums) < 4 or len(nums) % 2:
   205→        raise ValueError('Degenerate polygon/polyline points.')
   206→    pts = [(float(nums[i]), float(nums[i + 1])) for i in range(0, len(nums), 2)]
   207→    for p in pts:
   208→        if not math.isfinite(p[0]) or abs(p[0]) > 1e6 or abs(p[1]) > 1e6:
   209→            raise ValueError('SVG coordinate out of supported range.')
   210→    cmds: List[Command] = [('M', pts[0][0], pts[0][1])]
   211→    cmds.extend(('L', x, y) for x, y in pts[1:])
   212→    if close:
   213→        cmds.append(('Z',))
   214→    return cmds

   217→def _element_commands(tag: str, attrs: Dict[str, str]) -> List[Command]:
   218→    if tag == 'path':
   219→        return parse_path(attrs.get('d', ''))
   220→    if tag == 'rect':
   221→        return _rect_commands(_num(attrs.get('x')), _num(attrs.get('y')),
   222→                              _num(attrs.get('width'), -1), _num(attrs.get('height'), -1),
   223→                              _num(attrs.get('rx'), 0.0), _num(attrs.get('ry'), 0.0))
   224→    if tag == 'circle':
   225→        cx, cy, r = _num(attrs.get('cx')), _num(attrs.get('cy')), _num(attrs.get('r'), -1)
   226→        return _ellipse_commands(cx, cy, r, r)
   227→    if tag == 'ellipse':
   228→        return _ellipse_commands(_num(attrs.get('cx')), _num(attrs.get('cy')),
   229→                                 _num(attrs.get('rx'), -1), _num(attrs.get('ry'), -1))
   230→    if tag == 'polygon':
   231→        return _points_commands(attrs.get('points', ''), close=True)
   232→    if tag == 'polyline':
   233→        return _points_commands(attrs.get('points', ''), close=False)
   234→    if tag == 'line':
   235→        return [('M', _num(attrs.get('x1')), _num(attrs.get('y1'))),
   236→                ('L', _num(attrs.get('x2')), _num(attrs.get('y2')))]
   237→    raise ValueError(f'Unsupported geometry element {tag}.')

   240→class MasterDoc:
   241→    """Parsed, sanitized SVG master — the authoritative curve source."""
   242→
   243→    def __init__(self) -> None:
   244→        self.view_box: Tuple[float, float, float, float] = (0, 0, 576, 768)
   245→        self.shapes: List[dict] = []       # fillable shapes, document order
   246→        self.ink_shapes: List[dict] = []   # stroke-only line art, document order
   247→        self.gradients: Dict[str, dict] = {}
   248→        self.report: dict = {'removed': [], 'warnings': [], 'rasterized': False,
   249→                             'notes': ['Imported SVG masters keep original curve commands; '
   250→                                       'nothing is rasterized or retraced.']}
   251→        self.source_text: str = ''
   252→
   253→    def summary(self) -> dict:
   254→        curved = sum(1 for s in self.shapes + self.ink_shapes
   255→                     if any(c[0] in ('C', 'Q') for c in s['commands']))
   256→        total_cmds = sum(len(s['commands']) for s in self.shapes + self.ink_shapes)
   257→        curved_cmds = sum(1 for s in self.shapes + self.ink_shapes for c in s['commands']
   258→                          if c[0] in ('C', 'Q'))
   259→        return {
   260→            'viewBox': list(self.view_box),
   261→            'shapes': len(self.shapes),
   262→            'inkShapes': len(self.ink_shapes),
   263→            'gradients': len(self.gradients),
   264→            'curvedShapes': curved,
   265→            'commands': total_cmds,
   266→            'curvedCommands': curved_cmds,
   267→            'hiddenShapes': sum(1 for s in self.shapes if s.get('hidden')),
   268→            'removed': self.report['removed'],
   269→            'warnings': self.report['warnings'],
   270→            'rasterized': False,
   271→        }

   274→def _collect_gradients(root: ET.Element, doc: MasterDoc) -> None:
   275→    for elem in root.iter():
   276→        tag = _local(elem.tag)
   277→        if tag not in ('lineargradient', 'radialgradient'):
   278→            continue
   279→        attrs = _attrs(elem)
   280→        gid = attrs.get('id')
   281→        if not gid:
   282→            doc.report['warnings'].append('A gradient without id was dropped.')
   283→            continue
   284→        stops = []
   285→        for child in elem:
   286→            if _local(child.tag) != 'stop':
   287→                continue
   288→            sattrs = _attrs(child)
   289→            sstyle = _style_dict(sattrs.get('style'))
   290→            merged = {**sattrs, **sstyle}
   291→            raw_off = merged.get('offset', '0')
   292→            m = re.match(r'^\s*([\d.]++)\s*(%)?', str(raw_off))
   293→            off = float(m.group(1)) / (100.0 if m.group(2) else 1.0) if m else 0.0
   294→            color = _parse_color(merged.get('stop-color', '#000000'))
   295→            opacity = max(0.0, min(1.0, _num(merged.get('stop-opacity'), 1.0)))
   296→            stops.append({'offset': round(off, 4), 'color': color or '#000000', 'opacity': opacity})
   297→        stops.sort(key=lambda s: s['offset'])
   298→        units = attrs.get('gradientunits', 'objectBoundingBox')
   299→        gtransform = parse_transform(attrs.get('gradientTransform'))
   300→        entry = {'id': gid, 'type': 'linear' if tag == 'lineargradient' else 'radial',
   301→                 'units': units, 'transform': gtransform, 'stops': stops}
   302→        if tag == 'lineargradient':
   303→            entry['x1'] = _num(attrs.get('x1'), 0.0)
   304→            entry['y1'] = _num(attrs.get('y1'), 0.0)
   305→            entry['x2'] = _num(attrs.get('x2'), 1.0)
   306→            entry['y2'] = _num(attrs.get('y2'), 0.0)
   307→        else:
   308→            entry['cx'] = _num(attrs.get('cx'), 0.5)
   309→            entry['cy'] = _num(attrs.get('cy'), 0.5)
   310→            entry['r'] = _num(attrs.get('r'), 0.5)
   311→            entry['fx'] = _num(attrs.get('fx'), entry['cx'])
   312→            entry['fy'] = _num(attrs.get('fy'), entry['cy'])
   313→        if not stops:
   314→            doc.report['warnings'].append(f'Gradient {gid} has no stops and was dropped.')
   315→            continue
   316→        doc.gradients[gid] = entry

   319→def _resolve_gradient(entry: dict, bbox: Tuple[float, float, float, float]) -> dict:
   320→    """Resolve objectBoundingBox units to user space for one shape bbox."""
   321→    x0, y0, x1, y1 = bbox
   322→    w = max(1e-6, x1 - x0)
   323→    h = max(1e-6, y1 - y0)
   324→    tr: Mat = entry['transform']
   325→
   326→    def uv(px: float, py: float) -> Tuple[float, float]:
   327→        if entry['units'] == 'userSpaceOnUse':
   328→            return mat_apply(tr, (px, py))
   329→        return mat_apply(tr, (x0 + px * w, y0 + py * h))
   330→
   331→    resolved = {'type': entry['type'], 'stops': entry['stops'], 'units': 'userSpaceOnUse'}
   332→    if entry['type'] == 'linear':
   333→        a = uv(entry['x1'], entry['y1'])
   334→        b = uv(entry['x2'], entry['y2'])
   335→        resolved['x1'], resolved['y1'] = round(a[0], 3), round(a[1], 3)
   336→        resolved['x2'], resolved['y2'] = round(b[0], 3), round(b[1], 3)
   337→    else:
   338→        c = uv(entry['cx'], entry['cy'])
   339→        f = uv(entry['fx'], entry['fy'])
   340→        # radius: scale relative bbox diagonal-ish; userSpace r is used directly
   341→        if entry['units'] == 'userSpaceOnUse':
   342→            r = entry['r']
   343→        else:
   344→            r = entry['r'] * math.hypot(w, h) / math.sqrt(2)
   345→        resolved['cx'], resolved['cy'] = round(c[0], 3), round(c[1], 3)
   346→        resolved['r'] = round(r, 3)
   347→        resolved['fx'], resolved['fy'] = round(f[0], 3), round(f[1], 3)
   348→    return resolved

   351→def _shapely_poly(rings) -> Polygon:
   352→    if not rings:
   353→        return Polygon()
   354→    prepped = [[(float(x), float(y)) for x, y in ring] for ring in rings if len(ring) >= 3]
   355→    if not prepped:
   356→        return Polygon()
   357→    outer = prepped[0]
   358→    holes = prepped[1:]
   359→    try:
   360→        poly = Polygon(outer, holes)
   361→        if not poly.is_valid:
   362→            from shapely import make_valid
   363→            poly = make_valid(poly)
   364→        return poly
   365→    except Exception:
   366→        return Polygon(outer)

   369→def import_master(text: str) -> MasterDoc:
   370→    """Parse + sanitize SVG text into a MasterDoc (never rasterizes)."""
   371→    if not isinstance(text, str):
   372→        raise ValueError('SVG master must be text.')
   373→    if len(text.encode('utf-8')) > 12 * 1024 * 1024:
   374→        raise ValueError('Use an SVG file under 12 MB.')
   375→    lowered = text.lower()
   376→    if '<!doctype' in lowered or '<!entity' in lowered:
   377→        raise ValueError('DTDs and entity declarations are not accepted in SVG masters.')
   378→    try:
   379→        root = ET.fromstring(text)
   380→    except ET.ParseError as exc:
   381→        raise ValueError(f'The SVG file is not well-formed XML: {exc}') from exc
   382→    if _local(root.tag) != 'svg':
   383→        raise ValueError('The root element must be <svg>.')
   384→
   385→    doc = MasterDoc()
   386→    doc.source_text = text
   387→
   388→    # viewBox / size
   389→    attrs = _attrs(root)
   390→    if 'viewbox' in attrs:
   391→        parts = re.findall(r'[-+]?(?:\d*\.\d+|\d+\.?)', attrs['viewbox'])
   392→        if len(parts) == 4:
   393→            vb = [float(p) for p in parts]
   394→            if vb[2] > 0 and vb[3] > 0:
   395→                doc.view_box = (vb[0], vb[1], vb[2], vb[3])
   396→            else:
   397→                raise ValueError('viewBox must have positive width and height.')
   398→        else:
   399→            raise ValueError('viewBox must contain four numbers.')
   400→    else:
   401→        w = _num(attrs.get('width'), 0.0)
   402→        h = _num(attrs.get('height'), 0.0)
   403→        if w > 0 and h > 0:
   404→            doc.view_box = (0.0, 0.0, w, h)
   405→        else:
   406→            raise ValueError('The SVG needs a viewBox or width/height.')
   407→
   408→    _collect_gradients(root, doc)
   409→    id_map: Dict[str, ET.Element] = {a['id']: elem for elem in root.iter()
   410→                                      if (a := _attrs(elem)).get('id')}
   411→    total_elements = sum(1 for _ in root.iter())
   412→    if total_elements > _MAX_ELEMENTS:
   413→        raise ValueError(f'SVG exceeds {_MAX_ELEMENTS} elements.')
   414→
   415→    order = 0
   416→    seen_use: set = set()
   417→    counter = {'n': 0}
   418→
   419→    def order_counter() -> int:
   420→        counter['n'] += 1
   421→        return counter['n'] - 1
   422→
   423→    def walk(elem: ET.Element, ctx: _Context) -> None:
   424→        for child in elem:
   425→            tag = _local(child.tag)
   426→            cattrs = _attrs(child)
   427→            if tag in ('defs', 'lineargradient', 'radialgradient', 'stop'):
   428→                continue  # definitions, not drawable
   429→            if tag in _DROPPED_ELEMENTS and tag != 'use':
   430→                label = tag if tag not in ('title', 'desc', 'metadata') else None
   431→                if tag in ('script', 'foreignobject', 'image', 'iframe', 'style', 'text',
   432→                           'clippath', 'mask', 'filter', 'pattern', 'symbol', 'animate',
   433→                           'animatetransform', 'animatemotion', 'animatecolor', 'set'):
   434→                    doc.report['removed'].append(f'<{tag}>')
   435→                continue
   436→            if tag == 'g':
   437→                walk(child, ctx.child(cattrs))
   438→                continue
   439→            if tag == 'use':
   440→                href = cattrs.get('href') or cattrs.get('xlink:href')
   441→                if not href or not href.startswith('#'):
   442→                    doc.report['removed'].append('<use> (external or missing reference)')
   443→                    continue
   444→                target = id_map.get(href[1:])
   445→                if target is None or id(target) in seen_use:
   446→                    doc.report['warnings'].append(f'<use {href}> could not be resolved and was skipped.')
   447→                    continue
   448→                seen_use.add(id(target))
   449→                use_ctx = ctx.child(cattrs)
   450→                ux, uy = _num(cattrs.get('x')), _num(cattrs.get('y'))
   451→                use_ctx.matrix = mat_mul(use_ctx.matrix, (1.0, 0.0, 0.0, 1.0, ux, uy))
   452→                ttag = _local(target.tag)
   453→                if ttag == 'g':
   454→                    walk(target, use_ctx)
   455→                elif ttag in _BASIC:
   456→                    _emit_shape(target, use_ctx, ttag, doc, order_counter())
   457→                continue
   458→            if tag in _BASIC:
   459→                _emit_shape(child, ctx, tag, doc, order_counter())
   460→                continue
   461→            doc.report['removed'].append(f'<{tag}>')
   462→
   463→    def _emit_shape(elem: ET.Element, ctx: _Context, tag: str, doc: MasterDoc, ordn: int) -> None:
   464→        sattrs = _attrs(elem)
   465→        sctx = ctx.child(sattrs)
   466→        try:
   467→            cmds = _element_commands(tag, sattrs)
   468→        except ValueError as exc:
   469→            doc.report['warnings'].append(f'A <{tag}> was skipped: {exc}')
   470→            return
   471→        cmds = transform_commands(cmds, sctx.matrix)
   472→        if not cmds or sum(1 for c in cmds if c[0] != 'Z') < 1:
   473→            return
   474→        if sum(len(s['commands']) for s in doc.shapes) + sum(len(s['commands']) for s in doc.ink_shapes) + len(cmds) > _MAX_COMMANDS:
   475→            raise ValueError('SVG master is too complex (command budget exceeded).')
   476→        shape_id = sattrs.get('id') or f's{ordn:03d}'
   477→        role = sattrs.get('data-cd-role') or None
   478→        if role not in (None, 'gameplay', 'shading', 'ink'):
   479→            role = None
   480→        fill_ref = sctx.fill_ref(sattrs) or ctx.fill_ref(sattrs)
   481→        flat = flatten_path(cmds, FLAT_TOL)
   482→        bbox = commands_bbox(cmds)
   483→        has_area = len(flat) >= 1 and any(len(r) >= 3 for r in flat)
   484→        fillable = sctx.fill is not None or fill_ref is not None
   485→        stroke_only = (not fillable) and sctx.stroke is not None
   486→        base = {
   487→            'order': ordn, 'id': shape_id, 'role': role, 'element': tag,
   488→            'commands': cmds, 'd': format_path(cmds),
   489→            'fillRule': 'evenodd' if sctx.fill_rule == 'evenodd' else 'nonzero',
   490→            'bbox': bbox, 'rings': flat, 'opacity': round(sctx.opacity, 4),
   491→        }
   492→        if stroke_only or role == 'ink':
   493→            base.update({'stroke': sctx.stroke, 'strokeWidth': round(sctx.stroke_width, 3)})
   494→            doc.ink_shapes.append(base)
   495→            return
   496→        if not has_area:
   497→            if sctx.stroke is not None:
   498→                base.update({'stroke': sctx.stroke, 'strokeWidth': round(sctx.stroke_width, 3)})
   499→                doc.ink_shapes.append(base)
   500→            else:
   501→                doc.report['warnings'].append(f'Shape {shape_id} has no fill and no stroke; skipped.')
   502→            return
   503→        poly = _shapely_poly(flat)
   504→        area = float(poly.area) if not poly.is_empty else evenodd_area(flat)
   505→        fill = sctx.fill or '#000000'
   506→        if fill_ref:
   507→            if fill_ref not in doc.gradients:
   508→                doc.report['warnings'].append(
   509→                    f'Shape {shape_id} references missing gradient #{fill_ref}; painted with flat fallback.')
   510→            else:
   511→                fill = None  # resolved per-shape below
   512→        base.update({
   513→            'fill': fill, 'gradientRef': fill_ref,
   514→            'fillOpacity': round(sctx.fill_opacity, 4),
   515→            'area': area, 'hidden': False,
   516→            'stroke': sctx.stroke, 'strokeWidth': round(sctx.stroke_width, 3),
   517→        })
   518→        doc.shapes.append(base)
   519→
   520→    walk(root, _Context())
   521→
   522→    # resolve gradient references into per-shape user-space instances
   523→    for shape in doc.shapes:
   524→        ref = shape.get('gradientRef')
   525→        if not ref:
   526→            continue
   527→        entry = doc.gradients.get(ref)
   528→        if not entry:
   529→            shape['fill'] = shape['fill'] or '#808080'
   530→            continue
   531→        resolved = _resolve_gradient(entry, shape['bbox'])
   532→        shape['gradient'] = {
   533→            'id': f"g-{shape['order']}-{ref}",
   534→            'ref': ref,
   535→            **resolved,
   536→        }
   537→        # representative palette color: average of stops
   538→        stops = entry['stops']
   539→        if stops:
   540→            mid = stops[len(stops) // 2]
   541→            shape['fill'] = mid['color']
   542→        else:
   543→            shape['fill'] = '#808080'
   544→
   545→    # hidden-shape exclusion: fully covered by opaque shapes ABOVE (z-order)
   546→    opaque = [s for s in doc.shapes
   547→              if (s.get('fill') or s.get('gradientRef')) and s.get('fillOpacity', 1.0) >= 0.999]
   548→    for idx, shape in enumerate(doc.shapes):
   549→        if shape.get('fillOpacity', 1.0) < 0.999 or not shape.get('fill') and not shape.get('gradientRef'):
   550→            continue
   551→        own = _shapely_poly(shape['rings'])
   552→        if own.is_empty or own.area <= 0:
   553→            continue
   554→        covers = [s for s in opaque if s['order'] > shape['order'] and s is not shape]
   555→        if not covers:
   556→            continue
   557→        try:
   558→            above = unary_union([_shapely_poly(s['rings']) for s in covers])
   559→            visible = own.difference(above)
   560→            ratio = 0.0 if visible.is_empty else visible.area / max(own.area, 1e-9)
   561→            if ratio < 0.02:
   562→                shape['hidden'] = True
   563→                shape['hiddenBy'] = [s['id'] for s in covers[:3]]
   564→        except Exception:
   565→            continue
   566→
   567→    if not doc.shapes and not doc.ink_shapes:
   568→        raise ValueError('The SVG master contains no supported drawable shapes.')
   569→    return doc

   572→def emit_master_svg(doc: MasterDoc) -> str:
   573→    """Serialize the sanitized master as standalone SVG (preview-safe)."""
   574→    x, y, w, h = doc.view_box
   575→    parts = [f'<svg xmlns="http://www.w3.org/2000/svg" viewBox="{fmt_num(x)} {fmt_num(y)} {fmt_num(w)} {fmt_num(h)}" '
   576→             f'width="{fmt_num(w)}" height="{fmt_num(h)}">']
   577→    grads = [s['gradient'] for s in doc.shapes if s.get('gradient')]
   578→    if grads:
   579→        parts.append('<defs>')
   580→        for g in grads:
   581→            stops = ''.join(
   582→                f'<stop offset="{fmt_num(s["offset"], 4)}" stop-color="{s["color"]}"'
   583→                + (f' stop-opacity="{fmt_num(s["opacity"], 3)}"' if s.get('opacity', 1) != 1 else '')
   584→                + '/>' for s in g['stops'])
   585→            if g['type'] == 'linear':
   586→                parts.append(f'<linearGradient id="{g["id"]}" gradientUnits="userSpaceOnUse" '
   587→                             f'x1="{fmt_num(g["x1"])}" y1="{fmt_num(g["y1"])}" '
   588→                             f'x2="{fmt_num(g["x2"])}" y2="{fmt_num(g["y2"])}">{stops}</linearGradient>')
   589→            else:
   590→                parts.append(f'<radialGradient id="{g["id"]}" gradientUnits="userSpaceOnUse" '
   591→                             f'cx="{fmt_num(g["cx"])}" cy="{fmt_num(g["cy"])}" r="{fmt_num(g["r"])}" '
   592→                             f'fx="{fmt_num(g["fx"])}" fy="{fmt_num(g["fy"])}">{stops}</radialGradient>')
   593→        parts.append('</defs>')
   594→    for s in doc.shapes:
   595→        if s.get('hidden'):
   596→            continue
   597→        fill = s.get('fill') or '#000000'
   598→        if s.get('gradient'):
   599→            fill = f'url(#{s["gradient"]["id"]})'
   600→        opacity = s.get('fillOpacity', 1.0)
   601→        attrs = f'fill="{fill}" fill-rule="{s["fillRule"]}"'
   602→        if opacity < 0.999:
   603→            attrs += f' fill-opacity="{fmt_num(opacity, 3)}"'
   604→        if s.get('opacity', 1.0) < 0.999:
   605→            attrs += f' opacity="{fmt_num(s["opacity"], 3)}"'
   606→        parts.append(f'<path id="{s["id"]}" {attrs} d="{s["d"]}"/>')
   607→    for s in doc.ink_shapes:
   608→        stroke = s.get('stroke') or '#29383E'
   609→        parts.append(f'<path id="{s["id"]}" fill="none" stroke="{stroke}" '
   610→                     f'stroke-width="{fmt_num(s.get("strokeWidth", 1.5))}" '
   611→                     f'stroke-linecap="round" d="{s["d"]}"/>')
   612→    parts.append('</svg>')
   613→    return ''.join(parts)

   616→def clean_svg(data: bytes, destination: Path) -> dict:
   617→    """Upload entry point: sanitize an SVG master and persist it."""
   618→    if not data or len(data) > 12 * 1024 * 1024:
   619→        raise ValueError('Use an SVG file under 12 MB.')
   620→    try:
   621→        text = data.decode('utf-8-sig')
   622→    except UnicodeDecodeError as exc:
   623→        raise ValueError('The SVG file must be UTF-8 text.') from exc
   624→    doc = import_master(text)
   625→    destination.parent.mkdir(parents=True, exist_ok=True)
   626→    sanitized = emit_master_svg(doc)
   627→    destination.write_text(sanitized, encoding='utf-8')
   628→    summary = doc.summary()
   629→    if not any(c[0] in ('C', 'Q') for s in doc.shapes for c in s['commands']):
   630→        summary['warnings'] = summary['warnings'] + [
   631→            'This master contains no curve commands; the curve-preserving import had nothing to preserve.']
   632→    return {
   633→        'width': round(doc.view_box[2], 3),
   634→        'height': round(doc.view_box[3], 3),
   635→        'sha256': hashlib.sha256(destination.read_bytes()).hexdigest(),
   636→        'kind': 'svg',
   637→        'summary': summary,
   638→    }
