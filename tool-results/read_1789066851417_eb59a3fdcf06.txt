   310→    const key = JSON.stringify(pbs);

   396→    (board: VectorBoard) => {

   400→        if (!id) return "ignored";

   405→          setPlacing(false);
   406→          void runEditRef.current("label", { x: p.x, y: p.y }).catch((e: Error) => toast(e.message));
   407→          return "inspected";
   408→        }
   409→        const next = new Set(selectedRef.current);
   410→        if (next.has(id)) next.delete(id);
   411→        else next.add(id);
   412→        selectedRef.current = next;
   413→        setSelected(next);

   422→        return "inspected";

   492→      } catch {

   494→      }
   495→    })();
   496→    return () => {
   497→      cancelled = true;
   498→    };
