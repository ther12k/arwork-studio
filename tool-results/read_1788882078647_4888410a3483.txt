    53→type ArtPt = { x: number; y: number };
    54→
    55→/** Drop points closer than `minDist` art units to the previous kept point. */
    56→function dropDensePoints(pts: ArtPt[], minDist: number): ArtPt[] {
    57→  const out: ArtPt[] = [];
    58→  for (const p of pts) {
    59→    const last = out[out.length - 1];
    60→    if (!last || Math.hypot(p.x - last.x, p.y - last.y) >= minDist) out.push(p);
    61→  }
    62→  return out;
    63→}
    64→
    65→function perpendicularDistance(p: ArtPt, a: ArtPt, b: ArtPt): number {
    66→  const dx = b.x - a.x;
    67→  const dy = b.y - a.y;
    68→  const len = Math.hypot(dx, dy);
    69→  if (len < 1e-9) return Math.hypot(p.x - a.x, p.y - a.y);
    70→  return Math.abs(dy * p.x - dx * p.y + b.x * a.y - b.y * a.x) / len;
    71→}
    72→
    73→/** Ramer–Douglas–Peucker simplification (art units). */
    74→function rdp(pts: ArtPt[], epsilon: number): ArtPt[] {
    75→  if (pts.length < 3) return pts;
    76→  const a = pts[0];
    77→  const b = pts[pts.length - 1];
    78→  let maxDist = 0;
    79→  let index = 0;
    80→  for (let i = 1; i < pts.length - 1; i++) {
    81→    const d = perpendicularDistance(pts[i], a, b);
    82→    if (d > maxDist) {
    83→      maxDist = d;
    84→      index = i;
    85→    }
    86→  }
    87→  if (maxDist > epsilon) {
    88→    const left = rdp(pts.slice(0, index + 1), epsilon);
    89→    const right = rdp(pts.slice(index), epsilon);
    90→    return [...left.slice(0, -1), ...right];
    91→  }
    92→  return [a, b];
    93→}
    94→
    95→/** Shoelace area of a closed polygon (art units²). */
    96→function polygonArea(pts: ArtPt[]): number {
    97→  let sum = 0;
    98→  for (let i = 0; i < pts.length; i++) {
    99→    const a = pts[i];
   100→    const b = pts[(i + 1) % pts.length];
   101→    sum += a.x * b.y - b.x * a.y;
   102→  }
   103→  return Math.abs(sum) / 2;
   104→}
   105→
   106→/** Point at half the total arc length of the polyline (stroke midpoint). */
   107→function polylineMidpoint(pts: ArtPt[]): ArtPt | null {
   108→  if (!pts.length) return null;
   109→  const segs: number[] = [];
   110→  let total = 0;
   111→  for (let i = 1; i < pts.length; i++) {
   112→    const d = Math.hypot(pts[i].x - pts[i - 1].x, pts[i].y - pts[i - 1].y);
   113→    segs.push(d);
   114→    total += d;
   115→  }
   116→  if (!segs.length) return pts[0];
   117→  let half = total / 2;
   118→  for (let i = 0; i < segs.length; i++) {
   119→    if (half <= segs[i]) {
   120→      const t = segs[i] > 0 ? half / segs[i] : 0;
   121→      return { x: pts[i].x + (pts[i + 1].x - pts[i].x) * t, y: pts[i].y + (pts[i + 1].y - pts[i].y) * t };
   122→    }
   123→    half -= segs[i];
   124→  }
   125→  return pts[pts.length - 1];
   126→}
   127→
   128→const fmt = (n: number) => (Math.round(n * 100) / 100).toString();
   129→
   130→/** Build the edit-action path string: `M x,y L …` (+ ` Z` when closed). */
   131→function pathFromPoints(pts: ArtPt[], close: boolean): string {
   132→  return `M ${pts.map((p) => `${fmt(p.x)},${fmt(p.y)}`).join(" L ")}${close ? " Z" : ""}`;
   133→}
