   151→   151→
   152→   152→const SAFE_PATH = /^M[\s\d.,eE+\-MLQCZ]+Z$/;
   153→   153→const SAFE_PATH_OPEN = /^M[\s\d.,eE+\-MLQCZ]+$/;
   154→   154→const SAFE_HEX = /^#[0-9A-Fa-f]{6}$/;
   155→   155→const SAFE_GRADIENT_REF = /^url\(#g-[a-zA-Z0-9_-]+\)$/;
   156→   156→
   157→   157→export function validateBundle(bundle: Bundle): Bundle {
   158→   158→  const { manifest: m, geometry: g } = bundle || ({} as Bundle);
   159→   159→  const p = bundle?.palette;
   160→   160→  if (!m || !g || !Array.isArray(p) || !["color-duel-vector-1", "color-duel-detailed-vector-1"].includes(m.format))
   161→   161→    throw new Error("Unsupported artwork bundle");
   162→   162→  if (m.id !== g.artworkId || m.version !== g.artworkVersion || m.regionCount !== g.regions?.length)
   163→   163→    throw new Error("Artwork identity/count mismatch");
   164→   164→  if (
   165→   165→    !Array.isArray(g.viewBox) ||
   166→   166→    g.viewBox.length !== 4 ||
   167→   167→    g.viewBox.some((x) => !Number.isFinite(x)) ||
   168→   168→    g.viewBox[2] <= 0 ||
   169→   169→    g.viewBox[3] <= 0
   170→   170→  )
   171→   171→    throw new Error("Invalid viewBox");
   172→   172→  const paletteIds = new Set(p.map((x) => x.id));
   173→   173→  if (paletteIds.size !== p.length) throw new Error("Duplicate palette ID");
   174→   174→  const ids = new Set<string>();
   175→   175→  for (const r of [...g.regions, ...(g.decorations ?? [])]) {
   176→   176→    if (ids.has(r.id) || !/^[a-zA-Z0-9_-]+$/.test(r.id)) throw new Error("Duplicate or unsafe region ID");
   177→   177→    ids.add(r.id);
   178→   178→    if (!paletteIds.has(r.paletteId) || r.fillRule !== "evenodd" || !SAFE_PATH.test(r.d))
   179→   179→      throw new Error("Invalid region geometry or palette");
   180→   180→  }
   181→   181→  if (m.format === "color-duel-detailed-vector-1") {
   182→   182→    const paint = bundle.paint;
   183→   183→    if (!paint || paint.artworkId !== m.id || !Array.isArray(paint.paths) || !Array.isArray(paint.inkPaths))
   184→   184→      throw new Error("Missing detailed vector paint");
   185→   185→    const gradientIds = new Set((paint.gradients ?? []).map((x) => x.id));
   186→   186→    for (const path of paint.paths) {
   187→   187→      const okFill = SAFE_HEX.test(path.fill) ||
   188→   188→        (SAFE_GRADIENT_REF.test(path.fill) && gradientIds.has(path.fill.slice(5, -1)));
   189→   189→      if (!okFill || !SAFE_PATH.test(path.d)) throw new Error("Invalid paint path");
   190→   190→    }
   191→   191→    // Ink layer holds closed filled shapes OR open stroke line art.
   192→   192→    for (const path of paint.inkPaths) {
   193→   193→      const okFill = SAFE_HEX.test(path.fill) ||
   194→   194→        (SAFE_GRADIENT_REF.test(path.fill) && gradientIds.has(path.fill.slice(5, -1)));
   195→   195→      const okPath = path.strokeWidth != null || path.filled === false ? SAFE_PATH_OPEN.test(path.d) : SAFE_PATH.test(path.d);
   196→   196→      if (!okFill || !okPath) throw new Error("Invalid ink path");
   197→   197→    }
   198→   198→  }
   199→   199→  return bundle;
   200→   200→}

   206→   206→async function fetchJson<T>(url: string): Promise<T> {
   207→   207→  const res = await fetch(url, { headers: { "X-Studio-Request": "1" } });
   208→   208→  if (!res.ok) throw new Error(`Artwork request failed: ${res.status}`);
   209→   209→  return res.json() as Promise<T>;
   210→   210→}

   212→   212→/** Fetch + validate the four bundle files for a revision, through the gateway. */
   213→   213→export async function loadBundle(pid: string, rev: string): Promise<Bundle> {
   214→   214→  const base = `/api/projects/${pid}/revisions/${rev}/files/`;
   215→   215→  const q = "?XTransformPort=8765";
   216→   216→  const [manifest, geometry, palette, paint] = await Promise.all([
   217→   217→    fetchJson<Manifest>(base + "artwork.json" + q),
   218→   218→    fetchJson<Geometry>(base + "regions.json" + q),
   219→   219→    fetchJson<PaletteEntry[]>(base + "palette.json" + q),
   220→   220→    fetchJson<PaintLayer | null>(base + "paint.json" + q),
   221→   221→  ]);
   222→   222→  return validateBundle({ manifest, geometry, palette, paint });
   223→   223→}

   269→   269→export class VectorBoard {
   270→   270→  readonly svg: SVGSVGElement;
   271→   271→  readonly bundle: Bundle;
   272→   272→  readonly options: BoardOptions;
   273→   273→  mode: BoardMode;
   274→   274→  readonly prefix: string;
   275→   275→  private handlers: Array<() => void> = [];
   276→   276→  private pointers = new Map<number, { x: number; y: number }>();
   277→   277→  private history: Array<{ id: string; wasCompleted: boolean; oldColor?: number }> = [];
   278→   278→  preview = false;
   279→   279→  private ctx: CanvasRenderingContext2D | null;
   280→   280→  regions = new Map<string, RegionEntry>();
   281→   281→  private paths = new Map<string, Path2D>();
   282→   282→  readonly elements = new Map<string, SVGPathElement>();
   283→   283→  private labels = new Map<string, SVGTextElement>();
   284→   284→  private session: Session;
   285→   285→  private completed = new Set<string>();
   286→   286→  private base: number[];
   287→   287→  private view: number[];
   288→   288→  private detailed: boolean;
   289→   289→  private drag: { x: number; y: number; view: number[]; inverse: DOMMatrix; anchor: DOMPoint; moved: boolean } | null = null;
   290→   290→  private pinch: { distance: number; view: number[]; inverse: DOMMatrix; anchor: DOMPoint | null } | null = null;
   291→   291→  private suppressTap = false;
   292→   292→  /** Set by the wrapper (clientToArt override) — last art-space point of a tap. */
   293→   293→  lastTapPoint: DOMPoint | null = null;
   294→   294→
   295→   295→  constructor(svg: SVGSVGElement, bundle: Bundle, options: BoardOptions = {}) {
   296→   296→    if (!(svg instanceof SVGSVGElement)) throw new Error("Pass an <svg> element to VectorBoard");
   297→   297→    validateBundle(bundle);
   298→   298→    this.svg = svg;
   299→   299→    this.bundle = bundle;
   300→   300→    this.options = options;
   301→   301→    this.mode = options.mode ?? "number";
   302→   302→    if (!["number", "memory", "free"].includes(this.mode)) throw new Error("Invalid coloring mode");
   303→   303→    this.prefix = `cdv-${++sequence}-`;
   304→   304→    this.ctx = document.createElement("canvas").getContext("2d");
   305→   305→    if (!this.ctx) throw new Error("Canvas hit-testing is unavailable");
   306→   306→    this.regions = new Map(bundle.geometry.regions.map((r) => [r.id, r]));
   307→   307→    this.paths = new Map([...this.regions].map(([id, r]) => [id, new Path2D(r.d)]));
   308→   308→    const paletteIds = [...new Set(bundle.palette.map((p) => p.id))].sort((a, b) => a - b);
   309→   309→    this.session = {
   310→   310→      artworkId: bundle.manifest.id,
   311→   311→      artworkVersion: bundle.manifest.version,
   312→   312→      mode: this.mode,
   313→   313→      completedRegionIds: [],
   314→   314→      selectedPaletteId: paletteIds[0] ?? 1,
   315→   315→      freeColors: {},
   316→   316→      mistakes: 0,
   317→   317→      updatedAt: new Date().toISOString(),
   318→   318→    };
   319→   319→    this.completed = new Set(this.session.completedRegionIds);
   320→   320→    this.base = [...bundle.geometry.viewBox];
   321→   321→    this.view = [...this.base];
   322→   322→    this.detailed = bundle.manifest.format === "color-duel-detailed-vector-1";
   323→   323→    this.mount();
   324→   324→    this.bindGestures();
   325→   325→    this.refresh();
   326→   326→  }
   327→   327→
   328→   328→  /** Tap-to-fill. Kept as an overridable field so authoring modes can intercept. */
   329→   329→  paint: (id: string | null) => PaintResult | string = (id) => {
   330→   330→    const r = id ? this.regions.get(id) : undefined;
   331→   331→    if (!r || this.preview) return "ignored";
   332→   332→    if (this.completed.has(id!) && this.mode !== "free") return "already-complete";
   333→   333→    if (this.mode !== "free" && r.paletteId !== this.session.selectedPaletteId) {
   334→   334→      this.session.mistakes++;
   335→   335→      this.save();
   336→   336→      this.notify("wrong-color");
   337→   337→      return "wrong-color";
   338→   338→    }
   339→   339→    if (this.completed.has(id!) && this.session.freeColors[id!] === this.session.selectedPaletteId)
   340→   340→      return "already-complete";
   341→   341→    this.history.push({ id: id!, wasCompleted: this.completed.has(id!), oldColor: this.session.freeColors[id!] });
   342→   342→    this.completed.add(id!);
   343→   343→    if (this.mode === "free") this.session.freeColors[id!] = this.session.selectedPaletteId;
   344→   344→    this.save();
   345→   345→    this.refresh();
   346→   346→    this.notify("paint");
   347→   347→    return "painted";
   348→   348→  };
   349→   349→
   350→   350→  /** Client-space → artwork-space transform. Overridable (used to capture tap points). */
   351→   351→  clientToArt: (x: number, y: number, matrix?: DOMMatrix | null) => DOMPoint | null = (x, y, matrix) => {
   352→   352→    const ctm = matrix ?? this.svg.getScreenCTM()?.inverse() ?? null;
   353→   353→    if (!ctm) return null;
   354→   354→    return new DOMPoint(x, y).matrixTransform(ctm);
   355→   355→  };

   357→   357→  private mount() {
   358→   358→    this.svg.replaceChildren();
   359→   359→    this.svg.setAttribute("viewBox", this.view.join(" "));
   360→   360→    this.svg.setAttribute("aria-label", `${this.bundle.manifest.title}, interactive coloring artwork`);
   361→   361→    this.svg.setAttribute("role", "group");
   362→   362→    this.svg.style.touchAction = "none";
   363→   363→    const defs = svgNode("defs");
   364→   364→    for (const p of this.bundle.palette) {
   365→   365→      const grad = svgNode("linearGradient", {
   366→   366→        id: `${this.prefix}paint-${p.id}`,
   367→   367→        x1: "0%",
   368→   368→        y1: "0%",
   369→   369→        x2: "100%",
   370→   370→        y2: "100%",
   371→   371→      });
   372→   372→      for (const stop of p.paint.stops) grad.append(svgNode("stop", { offset: stop.offset, "stop-color": stop.color }));
   373→   373→      defs.append(grad);
   374→   374→    }
   375→   375→    if (this.detailed && this.bundle.paint?.gradients) {
   376→   376→      for (const g of this.bundle.paint.gradients) {
   377→   377→        const attrs: Attrs =
   378→   378→          g.type === "linear"
   379→   379→            ? { id: g.id, gradientUnits: "userSpaceOnUse", x1: g.x1, y1: g.y1, x2: g.x2, y2: g.y2 }
   380→   380→            : { id: g.id, gradientUnits: "userSpaceOnUse", cx: g.cx, cy: g.cy, r: g.r, fx: g.fx, fy: g.fy };
   381→   381→        const node = svgNode(g.type === "linear" ? "linearGradient" : "radialGradient", attrs);
   382→   382→        for (const stop of g.stops) {
   383→   383→          const stopAttrs: Attrs = { offset: stop.offset, "stop-color": stop.color };
   384→   384→          if (stop.opacity != null && stop.opacity !== 1) stopAttrs["stop-opacity"] = stop.opacity;
   385→   385→          node.append(svgNode("stop", stopAttrs));
   386→   386→        }
   387→   387→        defs.append(node);
   388→   388→      }
   389→   389→    }
   390→   390→    const hatch = svgNode("pattern", {
   391→   391→      id: this.prefix + "selected",
   392→   392→      width: 12,
   393→   393→      height: 12,
   394→   394→      patternUnits: "userSpaceOnUse",
   395→   395→    });
   396→   396→    hatch.append(
   397→   397→      svgNode("rect", { width: 12, height: 12, fill: "#EDF1F4" }),
   398→   398→      svgNode("path", { d: "M0 0H6V6H0Z M6 6H12V12H6Z", fill: "#C3CED4" })
   399→   399→    );
   400→   400→    defs.append(hatch);
   401→   401→    this.svg.append(defs);
   402→   402→    const g = this.bundle.geometry;
   403→   403→    if (this.detailed && this.bundle.paint) {
   404→   404→      const art = svgNode("g", { "data-layer": "vector-paint", "pointer-events": "none" });
   405→   405→      for (const p of this.bundle.paint.paths) {
   406→   406→        const gradientFill = p.fill.startsWith("url(#");
   407→   407→        art.append(
   408→   408→          svgNode("path", {
   409→   409→            d: p.d,
   410→   410→            fill: p.fill,
   411→   411→            ...(gradientFill ? {} : { stroke: p.fill, "stroke-width": 0.55, "stroke-linejoin": "round" }),
   412→   412→            "fill-rule": "evenodd",
   413→   413→          })
   414→   414→        );
   415→   415→      }
   416→   416→      this.svg.append(art);
   417→   417→    }
   418→   418→    const regions = svgNode("g", {
   419→   419→      stroke: g.stroke,
   420→   420→      "stroke-width": g.strokeWidth,
   421→   421→      "stroke-linejoin": "round",
   422→   422→      "fill-rule": "evenodd",
   423→   423→    });
   424→   424→    this.elements.clear();
   425→   425→    this.labels.clear();
   426→   426→    for (const r of g.regions) {
   427→   427→      const node = svgNode("path", {
   428→   428→        id: this.prefix + r.id,
   429→   429→        "data-region-id": r.id,
   430→   430→        "data-palette-id": r.paletteId,
   431→   431→        d: r.d,
   432→   432→        tabindex: 0,
   433→   433→        role: "button",
   434→   434→        "aria-label": `Region ${r.id}, palette ${this.mode === "memory" ? "hidden" : r.paletteId}`,
   435→   435→      }) as SVGPathElement;
   436→   436→      this.elements.set(r.id, node);
   437→   437→      regions.append(node);
   438→   438→    }
   439→   439→    this.svg.append(regions);
   440→   440→    const fixed = svgNode("g", { "pointer-events": "none", stroke: g.stroke, "stroke-width": 1.3, "stroke-linejoin": "round", "fill-rule": "evenodd" });
   441→   441→    for (const r of this.detailed ? [] : g.decorations)
   442→   442→      fixed.append(svgNode("path", { d: r.d, fill: `url(#${this.prefix}paint-${r.paletteId})` }));
   443→   443→    this.svg.append(fixed);
   444→   444→    const details = svgNode("g", { "pointer-events": "none", fill: "none", "stroke-linejoin": "round", "stroke-linecap": "round" });
   445→   445→    for (const d of g.detailPaths)
   446→   446→      details.append(svgNode("path", { d: d.d, stroke: d.stroke, "stroke-width": d.strokeWidth, opacity: d.opacity }));
   447→   447→    this.svg.append(details);
   448→   448→    if (this.detailed && this.bundle.paint) {
   449→   449→      const ink = svgNode("g", { "data-layer": "ink", "pointer-events": "none" });
   450→   450→      for (const p of this.bundle.paint.inkPaths) {
   451→   451→        if (p.strokeWidth != null || p.filled === false) {
   452→   452→          ink.append(svgNode("path", {
   453→   453→            d: p.d, fill: "none", stroke: p.fill,
   454→   454→            "stroke-width": p.strokeWidth ?? 1.5,
   455→   455→            "stroke-linecap": "round", "stroke-linejoin": "round",
   456→   456→          }));
   457→   457→        } else {
   458→   458→          ink.append(svgNode("path", { d: p.d, fill: p.fill, "fill-rule": "evenodd" }));
   459→   459→        }
   460→   460→      }
   461→   461→      this.svg.append(ink);
   462→   462→    }
   463→   463→    const labels = svgNode("g", {
   464→   464→      "pointer-events": "none",
   465→   465→      "font-family": "Arial,sans-serif",
   466→   466→      fill: "#33444C",
   467→   467→      "text-anchor": "middle",
   468→   468→      "dominant-baseline": "central",
   469→   469→    });
   470→   470→    for (const r of g.regions) {
   471→   471→      const label = svgNode("text", {
   472→   472→        "data-label-for": r.id,
   473→   473→        x: r.label.x,
   474→   474→        y: r.label.y,
   475→   475→        "font-size": r.label.fontSize,
   476→   476→      }) as SVGTextElement;
   477→   477→      label.textContent = String(r.paletteId);
   478→   478→      labels.append(label);
   479→   479→      this.labels.set(r.id, label);
   480→   480→    }
   481→   481→    this.svg.append(labels);
   482→   482→    this.listen(this.svg, "keydown", (e: KeyboardEvent) => {
   483→   483→      const id = (e.target as Element | null)?.getAttribute?.("data-region-id");
   484→   484→      if (id && (e.key === "Enter" || e.key === " ")) {
   485→   485→        e.preventDefault();
   486→   486→        this.paint(id);
   487→   487→      }
   488→   488→    });
   489→   489→  }

   491→   491→  private listen(el: EventTarget, type: string, fn: EventListener, options?: AddEventListenerOptions) {
   492→   492→    el.addEventListener(type, fn, options);
   493→   493→    this.handlers.push(() => el.removeEventListener(type, fn, options));
   494→   494→  }

   496→   496→  state(): BoardState {
   497→   497→    const completedRegionIds = [...this.completed];
   498→   498→    const groups = this.bundle.manifest.objectGroups ?? [];
   499→   499→    const objects: Record<string, { completed: number; total: number }> = {};
   500→   500→    for (const group of groups)
   501→   501→      objects[group.id] = {
   502→   502→        completed: group.regionIds.filter((id) => this.completed.has(id)).length,
   503→   503→        total: group.regionIds.length,
   504→   504→      };
   505→   505→    return {
   506→   506→      artworkId: this.session.artworkId,
   507→   507→      artworkVersion: this.session.artworkVersion,
   508→   508→      mode: this.session.mode,
   509→   509→      completedRegionIds,
   510→   510→      selectedPaletteId: this.session.selectedPaletteId,
   511→   511→      mistakes: this.session.mistakes,
   512→   512→      total: this.regions.size,
   513→   513→      completed: completedRegionIds.length,
   514→   514→      progress: this.regions.size ? completedRegionIds.length / this.regions.size : 0,
   515→   515→      objects,
   516→   516→      preview: this.preview,
   517→   517→      zoom: this.base[2] / this.view[2],
   518→   518→    };
   519→   519→  }

   521→   521→  /** In-memory only — authoring play-tests never write player progress. */
   522→   522→  private save() {
   523→   523→    this.session.completedRegionIds = [...this.completed];
   524→   524→    this.session.updatedAt = new Date().toISOString();
   525→   525→  }
   526→   526→
   527→   527→  private notify(reason: ChangeReason) {
   528→   528→    this.options.onChange?.(this.state(), reason);
   529→   529→  }
   530→   530→
   531→   531→  refresh() {
   532→   532→    for (const [id, r] of this.regions) {
   533→   533→      const done = this.completed.has(id);
   534→   534→      const node = this.elements.get(id);
   535→   535→      const label = this.labels.get(id);
   536→   536→      if (!node) continue;
   537→   537→      let fill = "#FFFFFF";
   538→   538→      if (this.preview || done) {
   539→   539→        const palette = !this.preview && this.mode === "free" ? this.session.freeColors[id] : r.paletteId;
   540→   540→        fill = palette == null ? "#FFFFFF" : `url(#${this.prefix}paint-${palette})`;
   541→   541→      } else if (this.mode === "number" && r.paletteId === this.session.selectedPaletteId) {
   542→   542→        fill = `url(#${this.prefix}selected)`;
   543→   543→      }
   544→   544→      if (this.detailed && (this.preview || (done && this.mode !== "free"))) fill = "none";
   545→   545→      node.setAttribute("fill", fill);
   546→   546→      node.setAttribute("stroke", this.detailed && (done || this.preview) ? "none" : this.bundle.geometry.stroke);
   547→   547→      node.setAttribute("data-completed", String(done));
   548→   548→      node.setAttribute("tabindex", done && this.mode !== "free" ? "-1" : "0");
   549→   549→      node.setAttribute("aria-pressed", String(done));
   550→   550→      if (label) label.style.display = done || this.preview || this.mode !== "number" ? "none" : "";
   551→   551→    }
   552→   552→    this.updateLabelVisibility();
   553→   553→    this.notify("render");
   554→   554→  }
   555→   555→
   556→   556→  updateLabelVisibility() {
   557→   557→    if (!this.detailed || this.labels.size === 0) return;
   558→   558→    const screen = this.svg.getBoundingClientRect();
   559→   559→    if (!screen.width || !screen.height) return;
   560→   560→    const scale = Math.min(screen.width / this.view[2], screen.height / this.view[3]);
   561→   561→    for (const [id, r] of this.regions) {
   562→   562→      const label = this.labels.get(id);
   563→   563→      if (!label) continue;
   564→   564→      const visible = !this.completed.has(id) && !this.preview && this.mode === "number" && r.label.fontSize * scale >= 9;
   565→   565→      label.style.display = visible ? "" : "none";
   566→   566→    }
   567→   567→  }
   568→   568→
   569→   569→  setPalette(id: number) {
   570→   570→    if (!this.bundle.palette.some((p) => p.id === id)) throw new Error("Unknown palette ID");
   571→   571→    this.session.selectedPaletteId = id;
   572→   572→    this.save();
   573→   573→    this.refresh();
   574→   574→  }
   575→   575→
   576→   576→  setPreview(value: boolean) {
   577→   577→    this.preview = !!value;
   578→   578→    this.refresh();
   579→   579→  }
   580→   580→
   581→   581→  undo(): boolean {
   582→   582→    if (this.preview) return false;
   583→   583→    const previous = this.history.pop();
   584→   584→    if (!previous) return false;
   585→   585→    if (!previous.wasCompleted) this.completed.delete(previous.id);
   586→   586→    if (previous.oldColor === undefined) delete this.session.freeColors[previous.id];
   587→   587→    else this.session.freeColors[previous.id] = previous.oldColor;
   588→   588→    this.save();
   589→   589→    this.refresh();
   590→   590→    return true;
   591→   591→  }
   592→   592→
   593→   593→  reset() {
   594→   594→    this.completed.clear();
   595→   595→    this.session.freeColors = {};
   596→   596→    this.session.mistakes = 0;
   597→   597→    this.history = [];
   598→   598→    this.preview = false;
   599→   599→    this.save();
   600→   600→    this.refresh();
   601→   601→    this.fit();
   602→   602→  }

   604→   604→  private hitTest(x: number, y: number): string | null {
   605→   605→    // Iterate in reverse document order: the topmost region wins, which
   606→   606→    // matches how stacked SVG-master masks resolve overlaps (z-order).
   607→   607→    for (const [id, r] of [...this.regions].reverse()) {
   608→   608→      const b = r.bbox;
   609→   609→      if (x >= b[0] && x <= b[2] && y >= b[1] && y <= b[3] && this.ctx!.isPointInPath(this.paths.get(id)!, x, y, "evenodd"))
   610→   610→        return id;
   611→   611→    }
   612→   612→    return null;
   613→   613→  }
   614→   614→
   615→   615→  private applyView(view: number[]) {
   616→   616→    const [bx, by, bw, bh] = this.base;
   617→   617→    const w = clamp(view[2], bw / 10, bw);
   618→   618→    const h = (w * bh) / bw;
   619→   619→    this.view = [clamp(view[0], bx, bx + bw - w), clamp(view[1], by, by + bh - h), w, h];
   620→   620→    this.svg.setAttribute("viewBox", this.view.join(" "));
   621→   621→    this.updateLabelVisibility();
   622→   622→    this.notify("viewport");
   623→   623→  }
   624→   624→
   625→   625→  fit() {
   626→   626→    this.applyView([...this.base]);
   627→   627→  }
   628→   628→
   629→   629→  zoom(factor: number, anchor?: { x: number; y: number } | null) {
   630→   630→    const v = this.view;
   631→   631→    const a = anchor ?? { x: v[0] + v[2] / 2, y: v[1] + v[3] / 2 };
   632→   632→    const w = clamp(v[2] / factor, this.base[2] / 10, this.base[2]);
   633→   633→    const h = (w * this.base[3]) / this.base[2];
   634→   634→    this.applyView([a.x - ((a.x - v[0]) * w) / v[2], a.y - ((a.y - v[1]) * h) / v[3], w, h]);
   635→   635→  }
   636→   636→
   637→   637→  nextRegion(): string | null {
   638→   638→    if (this.mode !== "number" || this.preview) return null;
   639→   639→    const r = [...this.regions.values()].find(
   640→   640→      (region) => !this.completed.has(region.id) && region.paletteId === this.session.selectedPaletteId
   641→   641→    );
   642→   642→    if (!r) return null;
   643→   643→    const w = Math.max(
   644→   644→      this.base[2] / 10,
   645→   645→      Math.min(this.base[2], Math.max(r.bbox[2] - r.bbox[0], (r.bbox[3] - r.bbox[1]) * (this.base[2] / this.base[3])) * 1.6)
   646→   646→    );
   647→   647→    this.applyView([r.label.x - w / 2, r.label.y - (w * this.base[3]) / this.base[2] / 2, w, (w * this.base[3]) / this.base[2]]);
   648→   648→    this.elements.get(r.id)?.focus({ preventScroll: true });
   649→   649→    return r.id;
   650→   650→  }

   652→   652→  private bindGestures() {
   653→   653→    const startPinch = () => {
   654→   654→      const pts = [...this.pointers.values()];
   655→   655→      if (pts.length < 2) return;
   656→   656→      const mid = { x: (pts[0].x + pts[1].x) / 2, y: (pts[0].y + pts[1].y) / 2 };
   657→   657→      const inverse = this.svg.getScreenCTM()!.inverse();
   658→   658→      this.pinch = {
   659→   659→        distance: Math.max(1, Math.hypot(pts[0].x - pts[1].x, pts[0].y - pts[1].y)),
   660→   660→        view: [...this.view],
   661→   661→        inverse,
   662→   662→        anchor: this.clientToArt(mid.x, mid.y, inverse),
   663→   663→      };
   664→   664→      this.suppressTap = true;
   665→   665→    };
   666→   666→    this.listen(this.svg, "pointerdown", (event) => {
   667→   667→      const e = event as PointerEvent;
   668→   668→      if (e.pointerType === "mouse" && e.button !== 0) return;
   669→   669→      this.pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
   670→   670→      // Synthetic/test events and stale pointers can make capture fail; a throw
   671→   671→      // here would skip drag setup and break the whole tap flow, so guard it.
   672→   672→      try {
   673→   673→        this.svg.setPointerCapture(e.pointerId);
   674→   674→      } catch {
   675→   675→        /* no active pointer with this id — safe to continue without capture */
   676→   676→      }
   677→   677→      if (this.pointers.size === 1) {
   678→   678→        this.suppressTap = false;
   679→   679→        const inverse = this.svg.getScreenCTM()!.inverse();
   680→   680→        this.drag = {
   681→   681→          x: e.clientX,
   682→   682→          y: e.clientY,
   683→   683→          view: [...this.view],
   684→   684→          inverse,
   685→   685→          anchor: this.clientToArt(e.clientX, e.clientY, inverse),
   686→   686→          moved: false,
   687→   687→        };
   688→   688→      } else if (this.pointers.size === 2) startPinch();
   689→   689→    });
   690→   690→    this.listen(this.svg, "pointermove", (event) => {
   691→   691→      const e = event as PointerEvent;
   692→   692→      if (!this.pointers.has(e.pointerId)) return;
   693→   693→      this.pointers.set(e.pointerId, { x: e.clientX, y: e.clientY });
   694→   694→      if (this.pointers.size >= 2 && this.pinch) {
   695→   695→        const pts = [...this.pointers.values()];
   696→   696→        const mid = { x: (pts[0].x + pts[1].x) / 2, y: (pts[0].y + pts[1].y) / 2 };
   697→   697→        const distance = Math.max(1, Math.hypot(pts[0].x - pts[1].x, pts[0].y - pts[1].y));
   698→   698→        const p = this.pinch;
   699→   699→        const now = this.clientToArt(mid.x, mid.y, p.inverse);
   700→   700→        const w = clamp((p.view[2] * p.distance) / distance, this.base[2] / 10, this.base[2]);
   701→   701→        const h = (w * this.base[3]) / this.base[2];
   702→   702→        if (now && p.anchor)
   703→   703→          this.applyView([
   704→   704→            p.anchor.x - ((now.x - p.view[0]) * w) / p.view[2],
   705→   705→            p.anchor.y - ((now.y - p.view[1]) * h) / p.view[3],
   706→   706→            w,
   707→   707→            h,
   708→   708→          ]);
   709→   709→      } else if (!this.suppressTap && this.drag) {
   710→   710→        const d = this.drag;
   711→   711→        if (Math.hypot(e.clientX - d.x, e.clientY - d.y) > 6) d.moved = true;
   712→   712→        if (d.moved) {
   713→   713→          const now = this.clientToArt(e.clientX, e.clientY, d.inverse);
   714→   714→          if (now && d.anchor)
   715→   715→            this.applyView([d.view[0] - (now.x - d.anchor.x), d.view[1] - (now.y - d.anchor.y), d.view[2], d.view[3]]);
   716→   716→        }
   717→   717→      }
   718→   718→    });
   719→   719→    const finish = (event: Event, cancel = false) => {
   720→   720→      const e = event as PointerEvent;
   721→   721→      if (!this.pointers.has(e.pointerId)) return;
   722→   722→      const tap = !cancel && this.pointers.size === 1 && !this.suppressTap && !this.drag?.moved;
   723→   723→      this.pointers.delete(e.pointerId);
   724→   724→      if (this.svg.hasPointerCapture(e.pointerId)) this.svg.releasePointerCapture(e.pointerId);
   725→   725→      if (tap) {
   726→   726→        const p = this.clientToArt(e.clientX, e.clientY);
   727→   727→        if (p) this.paint(this.hitTest(p.x, p.y));
   728→   728→      }
   729→   729→      if (!this.pointers.size) {
   730→   730→        this.drag = null;
   731→   731→        this.pinch = null;
   732→   732→        this.suppressTap = false;
   733→   733→      } else this.suppressTap = true;
   734→   734→    };
   735→   735→    this.listen(this.svg, "pointerup", (e) => finish(e));
   736→   736→    this.listen(this.svg, "pointercancel", (e) => finish(e, true));
   737→   737→    this.listen(
   738→   738→      this.svg,
   739→   739→      "wheel",
   740→   740→      (event) => {
   741→   741→        const e = event as WheelEvent;
   742→   742→        e.preventDefault();
   743→   743→        this.zoom(Math.exp(-e.deltaY * 0.0015), this.clientToArt(e.clientX, e.clientY));
   744→   744→      },
   745→   745→      { passive: false }
   746→   746→    );
   747→   747→  }
   748→   748→
   749→   749→  destroy() {
   750→   750→    for (const remove of this.handlers) remove();
   751→   751→    this.handlers = [];
   752→   752→    this.svg.replaceChildren();
   753→   753→    this.pointers.clear();
   754→   754→  }
