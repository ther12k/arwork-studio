    35→function MicroCaption({ children, className = "" }: { children: React.ReactNode; className?: string }) {
    36→  return <p className={`text-[10px] leading-relaxed text-[#778481] ${className}`}>{children}</p>;
    37→}

    39→/** Pull the ±N px tolerance out of the flattened-rings derivation note. */
    40→function flattenToleranceLine(derivation: string): string | null {
    41→  const match = /±\s*([\d.]+)\s*px/.exec(derivation);
    42→  return match ? match[1] : null;
    43→}
    44→
    45→/** Region ids referenced by a QA warning/error line ("r-00001", "r-m-abc123…").
    46→ *  Empty when the line is not actionable (no region drill-down possible). */
    47→function regionIdsIn(text: string): string[] {
    48→  const ids = text.match(/\br-[A-Za-z0-9_-]{4,}\b/g);
    49→  return ids ? [...new Set(ids)] : [];
    50→}

    55→function NumberField({
    56→  id,
    57→  label,
    58→  value,
    59→  min,
    60→  max,
    61→  step = 1,
    62→  onCommit,
    63→}: {
    64→  id: string;
    65→  label: string;
    66→  value: number;
    67→  min: number;
    68→  max: number;
    69→  step?: number;
    70→  onCommit: (v: number) => void;
    71→}) {
    72→  return (
    73→    <div>
    74→      <Label htmlFor={id} className="text-[10px] leading-snug text-[#657671]">
    75→        {label}
    76→      </Label>
    77→      <Input
    78→        id={id}
    79→        type="number"
    80→        min={min}
    81→        max={max}
    82→        step={step}
    83→        value={value}
    84→        className="mt-1 h-8 rounded-md bg-white text-xs"
    85→        onChange={(e) => {
    86→          const v = Number(e.target.value);
    87→          if (e.target.value !== "" && Number.isFinite(v)) onCommit(v);
    88→        }}
    89→        onBlur={(e) => {
    90→          const v = Number(e.target.value);
    91→          if (!Number.isFinite(v) || v < min || v > max) onCommit(Math.min(max, Math.max(min, Number.isFinite(v) ? v : min)));
    92→        }}
    93→      />
    94→    </div>
    95→  );
    96→}

   528→              className="h-8 rounded-md border-[#e1e5df] bg-white text-[9px]"

   530→              onClick={() => doEdit("merge", { palette_id: Number(editPalette) || 1 })}
   531→            >
   532→              <Merge className="size-3" aria-hidden />
   533→              Merge selected
   534→            </Button>
   535→            <Button
   536→              variant="outline"
   537→              size="sm"
   538→              className="h-8 rounded-md border-[#e1e5df] bg-white text-[9px]"
   539→              disabled={busy || !selected.size}
   540→              onClick={() => doEdit("decorate")}
   541→            >
   542→              <Tag className="size-3" aria-hidden />
   543→              Make detail
   544→            </Button>
   545→            <Button
   546→              variant="outline"
   547→              size="sm"
   548→              className="h-8 rounded-md border-[#e1e5df] bg-white text-[9px]"
   549→              disabled={busy || !selected.size}
   550→              title="Split disconnected tap targets into separate regions"

   553→                  toast("Select exactly one region to split.");

   557→              }}
   558→            >
   559→              <SplitSquareHorizontal className="size-3" aria-hidden />
   560→              Split
   561→            </Button>
   562→            <Button
   563→              variant="outline"
   564→              size="sm"
   565→              className={`h-8 rounded-md text-[9px] ${
   566→                placing
   567→                  ? "border-[#087f74] bg-[#087f74] text-white"
   568→                  : "border-[#e1e5df] bg-white text-[#183837]"
   569→              }`}
   570→              disabled={selected.size !== 1 || busy}
   571→              onClick={startPlacing}
   572→            >
   573→              <Crosshair className="size-3" aria-hidden />
   574→              {placing ? "Tap new position" : "Place number"}

   577→              variant="outline"
   578→              size="sm"
   579→              className="h-8 rounded-md border-[#e1e5df] bg-white text-[9px]"
   580→              disabled={!selected.size}
   581→              onClick={clearSelection}
   582→            >
   583→              <Trash2 className="size-3" aria-hidden />
   584→              Clear selection
   585→            </Button>
   586→          </div>
   587→
   588→          {/* Recolor appearance — changes what the artwork looks like (paint layer),
   589→              distinct from the number-group association above. */}

   602→                disabled={busy}

   613→                spellCheck={false}

   615→                className="h-8 rounded-md bg-white font-mono text-xs"

   617→                aria-describedby={hexValid ? undefined : "studio-recolor-hex-hint"}

   628→                checked={preserveShading}
   629→                onCheckedChange={(v) => setPreserveShading(v === true)}
   630→                disabled={busy}
   631→                className="mt-0.5"
   632→              />
   633→              <Label
   634→                htmlFor="studio-preserve-shading"
   635→                className="text-[10px] leading-snug font-normal text-[#657671]"

   637→                Preserve shading
   638→              </Label>
   639→            </div>
   640→            <MicroCaption className="mt-1">
   641→              Tint gradients toward the target instead of replacing the fill.
   642→            </MicroCaption>
   643→            <Button
   644→              variant="outline"
   645→              size="sm"
   646→              className="mt-2 h-8 w-full rounded-md border-[#e1e5df] bg-white text-[9px]"

   648→              title={
   649→                hexValid
   650→                  ? "Recolor the selected regions' artwork (SVG-master builds only)"

   652→              }
   653→              onClick={() =>
   654→                doEdit("recolor", { color: recolorHex.toUpperCase(), preserve_shading: preserveShading })
   655→              }
   656→            >
   657→              <Paintbrush className="size-3" aria-hidden />
   658→              Recolor appearance
   659→            </Button>
   660→          </div>
   661→
   662→          <MicroCaption className="mt-2">
   663→            Number group = gameplay association (what number the region requires). Recolor = what

   665→          </MicroCaption>
   666→          <MicroCaption className="mt-1.5">
   667→            Each edit creates a new revision. Merge only neighboring regions. &ldquo;Make detail&rdquo; precolors a
   668→            region and removes it from the progress count.
   669→          </MicroCaption>
   670→        </div>
   671→      )}
   672→
   673→      {/* Revision history */}
   674→      <div className="rp-wide">
   675→        <Label htmlFor="studio-revision-list" className="studio-label mb-1.5">
   676→          Revision history
   677→        </Label>
   678→        <Select
   679→          value={revisionSelect}
   680→          onValueChange={(v) => setRevisionSelect(v)}
   681→          disabled={!project?.revisions.length}
   682→        >
   683→          <SelectTrigger id="studio-revision-list" className="h-8 w-full rounded-md bg-white text-xs">
   684→            <SelectValue placeholder={project?.revisions.length ? "Select a revision" : "No revisions"} />
   685→          </SelectTrigger>
   686→          <SelectContent>
   687→            {[...(project?.revisions ?? [])].reverse().map((rev) => (
   688→              <SelectItem key={rev.id} value={rev.id}>
   689→                v{rev.version} · {rev.kind} · {rev.regionCount} regions
   690→              </SelectItem>
   691→            ))}
   692→          </SelectContent>
   693→        </Select>
   694→        <button
   695→          type="button"
   696→          className="block py-2 text-left text-[11px] font-medium text-[#087f74] hover:underline disabled:opacity-50"
   697→          disabled={!revisionSelect || busy}
   698→          onClick={() =>
   699→            void activateSelectedRevision().catch((e: Error) => toast(e.message))
   700→          }
   701→        >
   702→          Restore selected revision
   703→        </button>
   704→      </div>
   705→
   706→      <div className="rp-wide">
   707→        <ReviewDialog>
   708→          <Button
   709→            variant="outline"
   710→            className="mt-2 w-full rounded-xl border-[#cfe6db] bg-[#e5f3ed] py-2.5 text-[11px] font-semibold text-[#126e5e] hover:bg-[#d9ede2]"
   711→            disabled={!revision || busy}
   712→          >
   713→            <CheckCircle2 className="size-4" aria-hidden />
   714→            Mark visually reviewed
   715→          </Button>
   716→        </ReviewDialog>
   717→      </div>
   718→
   719→      {/* Export links */}
   720→      <div className="rp-wide mt-3">
   721→        {project && revision ? (
   722→          <div className="flex flex-col gap-1">
   723→          <a
   724→            href={exportUrl(project.id, revision.id)}
   725→            download
   726→            className="flex w-full items-center justify-between rounded-xl bg-[#087f74] px-4 py-3 text-[11px] font-semibold text-white transition-colors hover:bg-[#056c62]"
   727→          >
   728→            <span className="flex items-center gap-2">
   729→              <FileArchive className="size-4" aria-hidden />
   730→              Export game bundle .zip
   731→            </span>
   732→            <span aria-hidden>↧</span>
   733→          </a>
   734→          <a
   735→            href={exportUrl(project.id, revision.id, true)}
   736→            download
   737→            className="block py-1.5 text-left text-[11px] font-medium text-[#087f74] hover:underline"
   738→          >
   739→            Include source &amp; authoring files
   740→          </a>
   741→          <a
   742→            href={renderUrl(project.id, revision.id, 2048)}
   743→            download
   744→            className="block py-1.5 text-left text-[11px] font-medium text-[#087f74] hover:underline"
   745→          >
   746→            <ImageIcon className="mr-1 inline size-3.5" aria-hidden />
   747→            Render 2048px PNG from vector
   748→          </a>
   749→        </div>
   750→        ) : (
   751→          <div className="flex flex-col gap-1 opacity-40" aria-disabled="true">
   752→          <span className="flex w-full cursor-not-allowed items-center justify-between rounded-xl bg-[#087f74] px-4 py-3 text-[11px] font-semibold text-white">
   753→            <span className="flex items-center gap-2">
   754→              <Download className="size-4" aria-hidden />
   755→              Export game bundle .zip
   756→            </span>
   757→            <span aria-hidden>↧</span>
   758→          </span>
   759→          <span className="block cursor-not-allowed py-1.5 text-left text-[11px] font-medium text-[#087f74]">
   760→            Include source &amp; authoring files
   761→          </span>
   762→          <span className="block cursor-not-allowed py-1.5 text-left text-[11px] font-medium text-[#087f74]">
   763→            Render 2048px PNG from vector
   764→          </span>
   765→        </div>

   767→      </div>
   768→      <MicroCaption className="rp-wide mt-1">
   769→        Export contains SVG + region, paint and palette JSON. Automatic output is a reviewable draft — not a
