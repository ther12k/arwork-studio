   387→          return "inspected";
   388→        }
   389→        const next = new Set(selectedRef.current);
   390→        if (next.has(id)) next.delete(id);
   391→        else next.add(id);

   504→    async (pid: string) => {

   509→        const changed = projectRef.current.currentRevision !== next.currentRevision;
   510→        setProjectSync(next);
   511→        if (!isBusyProject(next)) syncFields(next);
   512→        if (changed && next.currentRevision) {
   513→          viewRef.current = "colored";
   514→          setView("colored");
   515→          await mountBoard();
   516→          await refreshList();
   517→        } else if (
   518→          !isBusyProject(next) &&
   519→          next.currentRevision &&
   520→          loadedRevisionRef.current !== next.currentRevision
   521→        ) {
   522→          // job finished between the POST response and the first poll — mount the missed revision
   523→          viewRef.current = "colored";
   524→          setView("colored");
   525→          await mountBoard();
   526→          await refreshList();
   527→        }
   528→        if (isBusyProject(next)) {
   529→          pollTimerRef.current = setTimeout(() => void pollRef.current(pid), 900);
   530→        } else if (next.job?.status === "failed") {
   531→          toast(next.job.message || "Job failed");
   532→        }
   533→      } catch (e) {
   534→        toast(`Job polling stopped: ${(e as Error).message}`);
   535→      }
   536→    },
   537→    [mountBoard, refreshList, setProjectSync, syncFields]
   538→  );
   539→  useEffect(() => {
   540→    pollRef.current = poll;
   541→  }, [poll]);
   542→
   543→  const job = useCallback(
   544→    async (post: () => Promise<{ jobId: string }>) => {
   545→      const pid = projectRef.current?.id;
   546→      if (!pid) throw new Error("Create a project first.");
   547→      await post();
   548→      setProjectSync(await getProject(pid));
   549→      void poll(pid);
   550→    },
   551→    [poll, setProjectSync]
   552→  );
   553→
   554→  // ---------------------------------------------------------------- projects
   555→
   556→  const openProject = useCallback(
   557→    async (p: Project) => {
   558→      if (pollTimerRef.current) clearTimeout(pollTimerRef.current);
   559→      clearBoard();

   561→      setProjectSync(p);
   562→      syncFields(p);
   563→      setChatInput("");
   564→      setPaidConsent(false);
   565→      viewRef.current = p.currentRevision ? "colored" : "master";
   566→      setView(viewRef.current);
   567→      try {
   568→        localStorage.setItem(PROJECT_STORAGE_KEY, p.id);
   569→      } catch {
   570→        /* storage unavailable */
   571→      }
   572→      await refreshList();
   573→      if (p.currentRevision) await mountBoard();
   574→      if (isBusyProject(p)) void poll(p.id);
   575→    },
   576→    [clearBoard, mountBoard, poll, refreshList, setProjectSync, syncFields]
   577→  );
   578→

   580→    if (projectRef.current) return projectRef.current;
   581→    const created = await createProject("New illustrated world");
   582→    await openProject(created);
   583→    return created;
   584→  }, [openProject]);

   587→    async (pid: string) => {
   588→      await openProject(await getProject(pid));
   589→    },
   590→    [openProject]
   591→  );

   595→    await openProject(created);
   596→    toast("New project created.");
   597→  }, [openProject]);
   598→
   599→  // -------------------------------------------------------------------- init
   600→
   601→  useEffect(() => {
   602→    let cancelled = false;
   603→    (async () => {
   604→      try {
   605→        const cfg = await getConfig();
   606→        if (cancelled) return;
   607→        setConfig(cfg);
   608→        const items = await refreshList();
   609→        let saved: Project | undefined;
   610→        try {
   611→          const stored = localStorage.getItem(PROJECT_STORAGE_KEY);
   612→          saved = stored ? items.find((p) => p.id === stored) : undefined;
   613→        } catch {
   614→          /* storage unavailable */
   615→        }
   616→        const target = saved ?? items[0];
   617→        if (target) {
   618→          await openProject(target);
   619→        } else {
   620→          const created = await createProject("Cascade Treehouse", DEFAULT_BRIEF);
   621→          await openProject(created);
   622→        }
   623→      } catch (e) {
   624→        if (cancelled) return;
   625→        setConnectionError(true);
   626→        toast((e as Error).message);
   627→      }
   628→    })();
   629→    return () => {
   630→      cancelled = true;
   631→      if (pollTimerRef.current) clearTimeout(pollTimerRef.current);
   632→    };
   633→  }, []);
   634→
   635→  // ----------------------------------------------------------------- actions
   636→
   637→  const saveBrief = useCallback(async () => {
   638→    const p = projectRef.current;
   639→    if (!p) throw new Error("Create a project first.");

   641→      title: titleRef.current.trim() || "Untitled artwork",
   642→      brief: briefRef.current,
   643→    });
   644→    setProjectSync(next);
   645→    await refreshList();
   646→    return next;
   647→  }, [refreshList, setProjectSync]);

   652→      const next = await apiUploadImage(p.id, file, role, rightsRef.current);

   654→      syncFields(next);
   655→      if (role === "master") {
   656→        viewRef.current = "master";
   657→        setView("master");
   658→      }

   660→        role === "master" ? "Master ready. Build a vector draft." : "Reference added. Use chat to develop an original brief."
   661→      );
   662→    },
   663→    [ensureProject, setProjectSync, syncFields]
   664→  );

   666→  const promoteReference = useCallback(async () => {

   671→    viewRef.current = "master";

   768→  const runEdit = useCallback(
   769→    async (action: EditAction, extra: Partial<EditPayload> = {}, regionIds?: string[]) => {
   770→      const p = projectRef.current;
   771→      if (!p) throw new Error("Create a project first.");
   772→      if (isBusyProject(p)) throw new Error("Wait for the current job.");
   773→      if (!p.currentRevision) throw new Error("Build the vector regions first.");
   774→      // Cut/draw supply their own region ids (the target region / none);
   775→      // everything else uses the current selection.
   776→      const ids = regionIds ?? [...selectedRef.current];
   777→      if (!ids.length && action !== "draw") throw new Error("Select at least one region in Edit regions.");
   778→      const body: EditPayload = {
   779→        base_revision: p.currentRevision,
   780→        action,
   781→        region_ids: ids,
   782→        ...extra,

   830→  /** Apply a custom free-mode color (contract §4) and remember it in the
   831→   *  recent list (capped at 10, persisted in localStorage OUTSIDE the board). */
   832→  const setBoardFreeColor = useCallback((hex: string) => {
   833→    if (!FREE_HEX_RE.test(hex)) {
   834→      toast("Enter a 6-digit hex color like #66AA33.");
   835→      return;
   836→    }
   837→    const norm = hex.toUpperCase();
   838→    try {
   839→      boardRef.current?.setFreeColor(norm);
   840→    } catch (e) {
   841→      toast((e as Error).message);
   842→      return;
   843→    }
   844→    setFreeColor(norm);
   845→    setRecentColors((prev) => {
   846→      const next = [norm, ...prev.filter((c) => c !== norm)].slice(0, 10);
   847→      try {
   848→        localStorage.setItem(RECENT_COLORS_KEY, JSON.stringify(next));
   849→      } catch {
   850→        /* storage unavailable */
   851→      }
   852→      return next;
   853→    });
   854→  }, []);

   857→  const changeBoardMode = useCallback((mode: BoardMode) => {

   859→      boardRef.current?.setMode(mode);

   863→      setBoardMode(mode);

   869→  const startPlacing = useCallback(() => {

   879→  /** QA drill-down: make `id` the ONLY selected region, show the interactive

   881→   *  wrapper (selectedRef + setSelected + highlightSelection + inspector inputs). */
   882→  const inspectRegion = useCallback(
   883→    (id: string) => {
   884→      const board = boardRef.current;

   900→        setEditPalette(String(reg.paletteId));
   901→        setObjectGroup(!reg.objectId || reg.objectId === "unassigned" ? "roof" : reg.objectId);

   978→    svgRef,
   979→    canvasRef,
   980→    zoomRef,
   981→    messagesRef,
   982→    boardRef,
   983→    tool,
   984→    setTool: changeTool,
   985→    cutRegion,
   986→    drawRegion,
   987→    freeColor,
   988→    setBoardFreeColor,
   989→    recentColors,
   990→    boardMode,
   991→    setBoardMode: changeBoardMode,
   992→    titleInput,
   993→    briefInput,
   994→    chatInput,
   995→    paidConsent,
   996→    rightsConfirmed,
   997→    generationSource,
   998→    quality,
   999→    editPalette,
  1000→    objectGroup,
  1001→    revisionSelect,
  1002→    buildSettings,
  1003→    setTitleInput,
  1004→    setBriefInput,
  1005→    setChatInput,
  1006→    setPaidConsent,

  1008→    setGenerationSource,
