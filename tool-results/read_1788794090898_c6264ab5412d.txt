     3→/**
     4→ * useStudio — page-level state + data flow for the Color Duel Art Studio.
     5→ * Faithful port of the vanilla-JS `web/studio.mjs` controller onto React state.
     6→ *
     7→ * Board specifics (imperative, non-React):
     8→ *  - the <svg> element is rendered ONCE by React with no children; VectorBoard
     9→ *    owns its DOM children and is re-created only when the revision changes;
    10→ *  - `board.paint` is wrapped so "Edit regions" taps select instead of fill;
    11→ *  - `board.clientToArt` is wrapped to capture the last tap point (Place number).
    12→ */

   152→export function StudioProvider({ children }: { children: React.ReactNode }) {
   153→  const [config, setConfig] = useState<StudioConfig | null>(null);
   154→  const [projects, setProjects] = useState<Project[]>([]);
   155→  const [project, setProject] = useState<Project | null>(null);
   156→  const [view, setView] = useState<StudioView>("master");
   157→  const [bundle, setBundle] = useState<Bundle | null>(null);
   158→  const [boardState, setBoardState] = useState<BoardState | null>(null);
   159→  const [selected, setSelected] = useState<Set<string>>(new Set());
   160→  const [placing, setPlacing] = useState(false);
   161→  const [connectionError, setConnectionError] = useState(false);
   162→
   163→  // editable fields
   164→  const [titleInput, setTitleInput] = useState("New illustrated world");
   165→  const [briefInput, setBriefInput] = useState("");
   166→  const [chatInput, setChatInput] = useState("");
   167→  const [paidConsent, setPaidConsent] = useState(false);
   168→  const [rightsConfirmed, setRightsConfirmed] = useState(false);
   169→  const [generationSource, setGenerationSource] = useState<GenerateSource>("brief");
   170→  const [quality, setQuality] = useState<ImageQuality>("medium");
   171→  const [editPalette, setEditPalette] = useState("1");
   172→  const [objectGroup, setObjectGroup] = useState("roof");
   173→  const [revisionSelect, setRevisionSelect] = useState("");
   174→  const [buildSettings, setBuildSettings] = useState<BuildSettings>({
   175→    target_regions: 650,
   176→    palette_colors: 32,
   177→    paint_colors: 80,
   178→    max_edge: 1024,
   179→    compactness: 10,
   180→    min_region_pixels: 35,
   181→    min_label_radius: 3,
   182→    ink_threshold: 40,
   183→    backend: "spline-local",
   184→    curve_tolerance: 1.0,
   185→    corner_angle_deg: 60,
   186→  });
   187→  const [svgPrompt, setSvgPrompt] = useState("");
   188→  const [svgAspect, setSvgAspect] = useState<"1024x1536" | "1536x1024" | "1024x1024">("1024x1536");
   189→  const [svgPaidConsent, setSvgPaidConsent] = useState(false);
   190→
   191→  // refs mirroring state for closures created once
   192→  const projectRef = useRef<Project | null>(null);
   193→  const viewRef = useRef<StudioView>("master");
   194→  const selectedRef = useRef<Set<string>>(new Set());
   195→  const placingRef = useRef(false);
   196→  const boardRef = useRef<VectorBoard | null>(null);
   197→  const bundleRef = useRef<Bundle | null>(null);
   198→  const loadedRevisionRef = useRef<string | null>(null);
   199→  const loadingTokenRef = useRef(0);
   200→  const pollTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   201→  const titleRef = useRef(titleInput);
   202→  const briefRef = useRef(briefInput);
   203→  const chatRef = useRef(chatInput);
   204→  const paidRef = useRef(paidConsent);
   205→  const rightsRef = useRef(rightsConfirmed);
   206→  const generationSourceRef = useRef(generationSource);
   207→  const qualityRef = useRef(quality);
   208→  const buildSettingsRef = useRef(buildSettings);
   209→  const revisionSelectRef = useRef(revisionSelect);
   210→  const svgPromptRef = useRef(svgPrompt);
   211→  const svgAspectRef = useRef(svgAspect);
   212→  const svgPaidRef = useRef(svgPaidConsent);
   213→  const configRef = useRef<StudioConfig | null>(null);
   214→
   215→  // DOM refs
   216→  const svgRef = useRef<SVGSVGElement | null>(null);
   217→  const canvasRef = useRef<HTMLDivElement | null>(null);
   218→  const zoomRef = useRef<HTMLSpanElement | null>(null);
   219→  const messagesRef = useRef<HTMLDivElement | null>(null);
   220→
   221→  // keep field refs in sync
   222→  useEffect(() => void (titleRef.current = titleInput), [titleInput]);
   223→  useEffect(() => void (briefRef.current = briefInput), [briefInput]);
   224→  useEffect(() => void (chatRef.current = chatInput), [chatInput]);
   225→  useEffect(() => void (paidRef.current = paidConsent), [paidConsent]);
   226→  useEffect(() => void (rightsRef.current = rightsConfirmed), [rightsConfirmed]);
   227→  useEffect(() => void (generationSourceRef.current = generationSource), [generationSource]);
   228→  useEffect(() => void (qualityRef.current = quality), [quality]);
   229→  useEffect(() => void (svgPromptRef.current = svgPrompt), [svgPrompt]);
   230→  useEffect(() => void (svgAspectRef.current = svgAspect), [svgAspect]);
   231→  useEffect(() => void (svgPaidRef.current = svgPaidConsent), [svgPaidConsent]);
   232→  useEffect(() => void (configRef.current = config), [config]);
   233→  useEffect(() => void (buildSettingsRef.current = buildSettings), [buildSettings]);
   234→  useEffect(() => void (revisionSelectRef.current = revisionSelect), [revisionSelect]);
   235→
   236→  const busy = isBusyProject(project);
   237→  const revision = project?.revisions.find((r) => r.id === project.currentRevision) ?? null;
   238→  const selectionInfo = selected.size
   239→    ? `${selected.size} selected · ${[...selected].slice(0, 4).join(", ")}${selected.size > 4 ? "…" : ""}`
   240→    : "Tap regions to inspect or select them. Drag to pan.";
   241→
   242→  // ---------------------------------------------------------------- helpers
   243→
   244→  const setProjectSync = useCallback((next: Project | null) => {
   245→    projectRef.current = next;
   246→    setProject(next);
   247→    if (next) {
   248→      const current = next.revisions.find((r) => r.id === next.currentRevision);
   249→      const fallback = current?.id ?? next.revisions[0]?.id ?? "";
   250→      setRevisionSelect((prev) => (prev && next.revisions.some((r) => r.id === prev) ? prev : fallback));
   251→    } else {
   252→      setRevisionSelect("");
   253→    }
   254→  }, []);
   255→
   256→  /** Overwrite editable fields from the server (title / brief). */
   257→  const syncFields = useCallback((p: Project) => {
   258→    setTitleInput(p.title);
   259→    setBriefInput(p.brief);
   260→  }, []);
   261→
   262→  const refreshList = useCallback(async (): Promise<Project[]> => {
   263→    const items = await listProjects();
   264→    setProjects(items);
   265→    return items;
   266→  }, []);
   267→
   268→  /** Pure DOM update — selection text is derived during render from `selected`. */
   269→  const highlightSelection = useCallback(() => {
   270→    const board = boardRef.current;
   271→    if (!board) return;
   272→    for (const [id, el] of board.elements) el.classList.toggle("selected-region", selectedRef.current.has(id));
   273→  }, []);
   274→
   275→  const clearBoard = useCallback(() => {
   276→    loadingTokenRef.current++;
   277→    boardRef.current?.destroy();
   278→    boardRef.current = null;
   279→    bundleRef.current = null;
   280→    setBundle(null);
   281→    setBoardState(null);
   282→    loadedRevisionRef.current = null;
   283→    selectedRef.current = new Set();
   284→    setSelected(new Set());
   285→    placingRef.current = false;
   286→    setPlacing(false);
   287→  }, []);
   288→
   289→  // ------------------------------------------------------------- board view
   290→
   291→  const applyBoardView = useCallback(() => {
   292→    const board = boardRef.current;
   293→    if (!board) return;
   294→    board.setPreview(viewRef.current === "colored" || viewRef.current === "inspect");
   295→    highlightSelection();
   296→  }, [highlightSelection]);
   297→
   298→  const handleBoardChange = useCallback(
   299→    (state: BoardState, reason: string) => {
   300→      if (zoomRef.current) zoomRef.current.textContent = `${Math.round(state.zoom * 100)}%`;
   301→      if (reason !== "viewport") setBoardState(state);
   302→      if (reason === "wrong-color") toast("That region needs a different palette group.");
   303→      if (viewRef.current === "inspect") highlightSelection();
   304→    },
   305→    [highlightSelection]
   306→  );
   307→
   308→  // runEdit is called from inside the board paint wrapper; declare it via ref
   309→  // first so the wrapper (created once per board) always calls the latest one.
   310→  const runEditRef = useRef<(action: EditAction, extra?: Partial<EditPayload>) => Promise<void>>(async () => {
   311→    /* replaced below */
   312→  });
   313→
   314→  /** Wrap board interactions for authoring mode (selection instead of paint). */
   315→  const wrapBoard = useCallback(
   316→    (board: VectorBoard) => {
   317→      const normalPaint = board.paint.bind(board);
   318→      board.paint = (id: string | null) => {
   319→        if (viewRef.current !== "inspect") return normalPaint(id);
   320→        if (!id) return "ignored";
   321→        if (placingRef.current) {
   322→          const p = board.lastTapPoint;
   323→          if (!p) return "ignored";
   324→          placingRef.current = false;
   325→          setPlacing(false);
   326→          void runEditRef.current("label", { x: p.x, y: p.y }).catch((e: Error) => toast(e.message));
   327→          return "inspected";
   328→        }
   329→        const next = new Set(selectedRef.current);
   330→        if (next.has(id)) next.delete(id);
   331→        else next.add(id);
   332→        selectedRef.current = next;
   333→        setSelected(next);
   334→        if (next.size === 1) {
   335→          const reg = board.regions.get(id);
   336→          if (reg) {
   337→            setEditPalette(String(reg.paletteId));
   338→            setObjectGroup(!reg.objectId || reg.objectId === "unassigned" ? "roof" : reg.objectId);
   339→          }
   340→        }
   341→        highlightSelection();
   342→        return "inspected";
   343→      };
   344→      const convert = board.clientToArt.bind(board);
   345→      board.clientToArt = (x: number, y: number, matrix?: DOMMatrix | null) => {
   346→        const pt = convert(x, y, matrix);
   347→        board.lastTapPoint = pt;
   348→        return pt;
   349→      };
   350→    },
   351→    [highlightSelection]
   352→  );
   353→
   354→  /** Create the VectorBoard whenever a new bundle arrives (revision change). */
   355→  useEffect(() => {
   356→    if (!bundle || !svgRef.current) return;
   357→    const board = new VectorBoard(svgRef.current, bundle, {
   358→      persist: false,
   359→      mode: "number",
   360→      onChange: handleBoardChange,
   361→    });
   362→    boardRef.current = board;
   363→    wrapBoard(board);
   364→    applyBoardView();
   365→    return () => {
   366→      board.destroy();
   367→      if (boardRef.current === board) boardRef.current = null;
   368→    };
   369→  }, [bundle]);
   370→
   371→  // keep board preview/selection in sync with the active view
   372→  useEffect(() => {
   373→    applyBoardView();
   374→  }, [view, bundle, applyBoardView]);
   375→
   376→  // auto-scroll chat messages
   377→  useEffect(() => {
   378→    const el = messagesRef.current;
   379→    if (el) el.scrollTop = el.scrollHeight;
   380→  }, [project?.messages.length, project?.id]);
   381→
   382→  // re-measure labels when the canvas resizes
   383→  useEffect(() => {
   384→    const el = canvasRef.current;
   385→    if (!el || typeof ResizeObserver === "undefined") return;
   386→    const observer = new ResizeObserver(() => boardRef.current?.updateLabelVisibility());
   387→    observer.observe(el);
   388→    return () => observer.disconnect();
   389→  }, []);
   390→
   391→  // ----------------------------------------------------------------- polling
   392→
   393→  const mountBoard = useCallback(async () => {
   394→    const p = projectRef.current;
   395→    const r = p?.revisions.find((x) => x.id === p.currentRevision);
   396→    if (!p || !r || loadedRevisionRef.current === r.id) return;
   397→    clearBoard();
   398→    const token = loadingTokenRef.current;
   399→    try {
   400→      const b = await loadBundle(p.id, r.id);
   401→      if (token !== loadingTokenRef.current) return;
   402→      bundleRef.current = b;
   403→      loadedRevisionRef.current = r.id;
   404→      setBundle(b);
   405→    } catch (e) {
   406→      toast((e as Error).message);
   407→    }
   408→  }, [clearBoard]);
   409→
   410→  const pollRef = useRef<(pid: string) => Promise<void>>(async () => {
   411→    /* replaced below */
   412→  });
   413→  const poll = useCallback(
   414→    async (pid: string) => {
   415→      if (pollTimerRef.current) clearTimeout(pollTimerRef.current);
   416→      try {
   417→        const next = await getProject(pid);
   418→        if (projectRef.current?.id !== pid) return;
   419→        const changed = projectRef.current.currentRevision !== next.currentRevision;
   420→        setProjectSync(next);
   421→        if (!isBusyProject(next)) syncFields(next);
   422→        if (changed && next.currentRevision) {
   423→          viewRef.current = "colored";
   424→          setView("colored");
   425→          await mountBoard();
   426→          await refreshList();
   427→        } else if (
   428→          !isBusyProject(next) &&
   429→          next.currentRevision &&
   430→          loadedRevisionRef.current !== next.currentRevision
   431→        ) {
   432→          // job finished between the POST response and the first poll — mount the missed revision
   433→          viewRef.current = "colored";
   434→          setView("colored");
   435→          await mountBoard();
   436→          await refreshList();
   437→        }
   438→        if (isBusyProject(next)) {
   439→          pollTimerRef.current = setTimeout(() => void pollRef.current(pid), 900);
   440→        } else if (next.job?.status === "failed") {
   441→          toast(next.job.message || "Job failed");
   442→        }
   443→      } catch (e) {
   444→        toast(`Job polling stopped: ${(e as Error).message}`);
   445→      }
   446→    },
   447→    [mountBoard, refreshList, setProjectSync, syncFields]
   448→  );
   449→  useEffect(() => {
   450→    pollRef.current = poll;
   451→  }, [poll]);
   452→
   453→  const job = useCallback(
   454→    async (post: () => Promise<{ jobId: string }>) => {
   455→      const pid = projectRef.current?.id;
   456→      if (!pid) throw new Error("Create a project first.");
   457→      await post();
   458→      setProjectSync(await getProject(pid));
   459→      void poll(pid);
   460→    },
   461→    [poll, setProjectSync]
   462→  );
   463→
   464→  // ---------------------------------------------------------------- projects
   465→
   466→  const openProject = useCallback(
   467→    async (p: Project) => {
   468→      if (pollTimerRef.current) clearTimeout(pollTimerRef.current);
   469→      clearBoard();
   470→      setProjectSync(p);
   471→      syncFields(p);
   472→      setChatInput("");
   473→      setPaidConsent(false);
   474→      viewRef.current = p.currentRevision ? "colored" : "master";
   475→      setView(viewRef.current);
   476→      try {
   477→        localStorage.setItem(PROJECT_STORAGE_KEY, p.id);
   478→      } catch {
   479→        /* storage unavailable */
   480→      }
   481→      await refreshList();
   482→      if (p.currentRevision) await mountBoard();
   483→      if (isBusyProject(p)) void poll(p.id);
   484→    },
   485→    [clearBoard, mountBoard, poll, refreshList, setProjectSync, syncFields]
   486→  );
   487→
   488→  const ensureProject = useCallback(async () => {
   489→    if (projectRef.current) return projectRef.current;
   490→    const created = await createProject("New illustrated world");
   491→    await openProject(created);
   492→    return created;
   493→  }, [openProject]);
   494→
   495→  const openProjectById = useCallback(
   496→    async (pid: string) => {
   497→      await openProject(await getProject(pid));
   498→    },
   499→    [openProject]
   500→  );
   501→
   502→  const createNewProject = useCallback(async () => {
   503→    const created = await createProject("New illustrated world");
   504→    await openProject(created);
   505→    toast("New project created.");
   506→  }, [openProject]);
   507→
   508→  // -------------------------------------------------------------------- init
   509→
   510→  useEffect(() => {
   511→    let cancelled = false;
   512→    (async () => {
   513→      try {
   514→        const cfg = await getConfig();
   515→        if (cancelled) return;
   516→        setConfig(cfg);
   517→        const items = await refreshList();
   518→        let saved: Project | undefined;
   519→        try {
   520→          const stored = localStorage.getItem(PROJECT_STORAGE_KEY);
   521→          saved = stored ? items.find((p) => p.id === stored) : undefined;
   522→        } catch {
   523→          /* storage unavailable */
   524→        }
   525→        const target = saved ?? items[0];
   526→        if (target) {
   527→          await openProject(target);
   528→        } else {
   529→          const created = await createProject("Cascade Treehouse", DEFAULT_BRIEF);
   530→          await openProject(created);
   531→        }
   532→      } catch (e) {
   533→        if (cancelled) return;
   534→        setConnectionError(true);
   535→        toast((e as Error).message);
   536→      }
   537→    })();
   538→    return () => {
   539→      cancelled = true;
   540→      if (pollTimerRef.current) clearTimeout(pollTimerRef.current);
   541→    };
   542→  }, []);
   543→
   544→  // ----------------------------------------------------------------- actions
   545→
   546→  const saveBrief = useCallback(async () => {
   547→    const p = projectRef.current;
   548→    if (!p) throw new Error("Create a project first.");
   549→    const next = await patchProject(p.id, {
   550→      title: titleRef.current.trim() || "Untitled artwork",
   551→      brief: briefRef.current,
   552→    });
   553→    setProjectSync(next);
   554→    await refreshList();
   555→    return next;
   556→  }, [refreshList, setProjectSync]);
   557→
   558→  const uploadFile = useCallback(
   559→    async (file: File, role: "reference" | "master") => {
   560→      const p = await ensureProject();
   561→      const next = await apiUploadImage(p.id, file, role, rightsRef.current);
   562→      setProjectSync(next);
   563→      syncFields(next);
   564→      if (role === "master") {
   565→        viewRef.current = "master";
   566→        setView("master");
   567→      }
   568→      toast(
   569→        role === "master" ? "Master ready. Build a vector draft." : "Reference added. Use chat to develop an original brief."
   570→      );
   571→    },
   572→    [ensureProject, setProjectSync, syncFields]
   573→  );
   574→
   575→  const promoteReference = useCallback(async () => {
   576→    const p = projectRef.current;
   577→    if (!p) throw new Error("Create a project first.");
   578→    const next = await promoteReferenceApi(p.id, rightsRef.current);
   579→    setProjectSync(next);
   580→    viewRef.current = "master";
   581→    setView("master");
   582→    toast("Reference promoted to master.");
   583→  }, [setProjectSync]);
   584→
   585→  const loadSampleProject = useCallback(async () => {
   586→    const p = await ensureProject();
   587→    const next = await loadSample(p.id);
   588→    setProjectSync(next);
   589→    viewRef.current = "master";
   590→    setView("master");
   591→    toast("Example loaded. Build vector draft to convert it locally.");
   592→  }, [ensureProject, setProjectSync]);
   593→
   594→  const loadSvgSampleProject = useCallback(async () => {
   595→    const p = await ensureProject();
   596→    const next = await loadSvgSample(p.id);
   597→    setProjectSync(next);
   598→    syncFields(next);
   599→    viewRef.current = "master";
   600→    setView("master");
   601→    toast("Curved SVG master loaded — curves preserved, never rasterized. Build the draft.");
   602→  }, [ensureProject, setProjectSync, syncFields]);
   603→
   604→  const uploadSvgFile = useCallback(
   605→    async (file: File) => {
   606→      const p = await ensureProject();
   607→      const next = await apiUploadSvgMaster(p.id, file, rightsRef.current);
   608→      setProjectSync(next);
   609→      syncFields(next);
   610→      viewRef.current = "master";
   611→      setView("master");
   612→      toast("SVG master imported (sanitized, curves preserved). Build the vector draft.");
   613→    },
   614→    [ensureProject, setProjectSync, syncFields]
   615→  );
   616→
   617→  const sendChat = useCallback(async () => {
   618→    const p = projectRef.current;
   619→    if (!p) throw new Error("Create a project first.");
   620→    const message = chatRef.current.trim();
   621→    if (!message) throw new Error("Write a direction for the AI first.");
   622→    await saveBrief();
   623→    await job(() => apiSendChat(p.id, { message, include_reference: true, confirm_paid: paidRef.current }));
   624→    setChatInput("");
   625→    setPaidConsent(false);
   626→  }, [job, saveBrief]);
   627→
   628→  const generate = useCallback(async () => {
   629→    const p = projectRef.current;
   630→    if (!p) throw new Error("Create a project first.");
   631→    await saveBrief();
   632→    await job(() =>
   633→      generateMaster(p.id, {
   634→        source: generationSourceRef.current,
   635→        prompt: briefRef.current,
   636→        quality: qualityRef.current,
   637→        size: "1024x1536",
   638→        confirm_paid: paidRef.current,
   639→      })
   640→    );
   641→    setPaidConsent(false);
   642→    viewRef.current = "master";
   643→    setView("master");
   644→  }, [job, saveBrief]);
   645→
   646→  const generateSvg = useCallback(async () => {
   647→    const p = projectRef.current;
   648→    if (!p) throw new Error("Create a project first.");
   649→    const prompt = svgPromptRef.current.trim();
   650→    if (!prompt) throw new Error("Describe the SVG artwork first.");
   651→    if (!svgPaidRef.current) throw new Error("Tick the paid-request consent first.");
   652→    if (!configRef.current?.ai?.configured) throw new Error("AI provider is not configured on this server.");
   653→    await saveBrief();
   654→    await job(() =>
   655→      generateSvgMaster(p.id, {
   656→        prompt,
   657→        aspect: svgAspectRef.current,
   658→        include_reference: true,
   659→        confirm_paid: true,
   660→      })
   661→    );
   662→    setSvgPrompt("");
   663→    setSvgPaidConsent(false);
   664→    viewRef.current = "master";
   665→    setView("master");
   666→  }, [job, saveBrief]);
   667→
   668→  const build = useCallback(async () => {
   669→    const p = projectRef.current;
   670→    if (!p) throw new Error("Create a project first.");
   671→    await saveBrief();
   672→    await job(() => buildDraft(p.id, buildSettingsRef.current));
   673→  }, [job, saveBrief]);
   674→
   675→  const runEdit = useCallback(
   676→    async (action: EditAction, extra: Partial<EditPayload> = {}) => {
   677→      const p = projectRef.current;
   678→      if (!p) throw new Error("Create a project first.");
   679→      if (!selectedRef.current.size) throw new Error("Select at least one region in Edit regions.");
   680→      if (isBusyProject(p)) throw new Error("Wait for the current job.");
   681→      if (!p.currentRevision) throw new Error("Build the vector regions first.");
   682→      const body: EditPayload = {
   683→        base_revision: p.currentRevision,
   684→        action,
   685→        region_ids: [...selectedRef.current],
   686→        ...extra,
   687→      };
   688→      await job(() => apiRunEdit(p.id, body));
   689→      selectedRef.current = new Set();
   690→      setSelected(new Set());
   691→      highlightSelection();
   692→    },
   693→    [highlightSelection, job]
   694→  );
   695→  useEffect(() => {
   696→    runEditRef.current = runEdit;
   697→  }, [runEdit]);
   698→
   699→  const clearSelection = useCallback(() => {
   700→    selectedRef.current = new Set();
   701→    setSelected(new Set());
   702→    placingRef.current = false;
   703→    setPlacing(false);
   704→    highlightSelection();
   705→  }, [highlightSelection]);
   706→
   707→  const startPlacing = useCallback(() => {
   708→    if (selectedRef.current.size !== 1) {
   709→      toast("Select exactly one region first.");
   710→      return;
   711→    }
   712→    placingRef.current = true;
   713→    setPlacing(true);
   714→    toast("Tap a roomy point inside the selected region.");
   715→  }, []);
   716→
   717→  const activateSelectedRevision = useCallback(async () => {
   718→    const p = projectRef.current;
   719→    if (!p) throw new Error("Create a project first.");
   720→    const target = revisionSelectRef.current;
   721→    if (!target) throw new Error("No revision selected.");
   722→    const next = await activateRevision(p.id, target);
   723→    clearBoard();
   724→    viewRef.current = "colored";
   725→    setView("colored");
   726→    setProjectSync(next);
   727→    syncFields(next);
   728→    await mountBoard();
   729→    toast("Revision restored.");
   730→  }, [clearBoard, mountBoard, setProjectSync, syncFields]);
   731→
   732→  const submitReviewNote = useCallback(
   733→    async (note: string) => {
   734→      const p = projectRef.current;
   735→      if (!p) throw new Error("Create a project first.");
   736→      const r = p.revisions.find((x) => x.id === p.currentRevision);
   737→      if (!r) throw new Error("Build a draft first.");
   738→      const next = await apiSubmitReview(p.id, r.id, note);
   739→      setProjectSync(next);
   740→      toast("Visual review recorded.");
   741→    },
   742→    [setProjectSync]
   743→  );
   744→
   745→  const switchView = useCallback(
   746→    (next: StudioView) => {
   747→      if (next !== "master" && !projectRef.current?.currentRevision) {
   748→        toast("Build the vector regions first.");
   749→        return;
   750→      }
   751→      viewRef.current = next;
   752→      setView(next);
   753→      void mountBoard();
   754→    },
   755→    [mountBoard]
   756→  );
   757→
   758→  const setBuildSetting = useCallback((key: keyof BuildSettings, value: number | string) => {
   759→    setBuildSettings((prev) => ({ ...prev, [key]: value }));
   760→  }, []);
   761→
   762→  const api: StudioApi = {
   763→    config,
   764→    projects,
   765→    project,
   766→    busy,
   767→    revision,
   768→    connectionError,
   769→    view,
   770→    bundle,
   771→    boardState,
   772→    selected,
   773→    placing,
   774→    selectionInfo,
   775→    setBoardPalette: (id: number) => {
   776→      boardRef.current?.setPalette(id);
   777→    },
   778→    findRegion: () => boardRef.current?.nextRegion(),
   779→    undoFill: () => boardRef.current?.undo(),
   780→    resetTest: () => boardRef.current?.reset(),
   781→    svgRef,
   782→    canvasRef,
   783→    zoomRef,
   784→    messagesRef,
   785→    titleInput,
   786→    briefInput,
   787→    chatInput,
   788→    paidConsent,
   789→    rightsConfirmed,
   790→    generationSource,
   791→    quality,
   792→    editPalette,
   793→    objectGroup,
   794→    revisionSelect,
   795→    buildSettings,
   796→    setTitleInput,
   797→    setBriefInput,
   798→    setChatInput,
   799→    setPaidConsent,
   800→    setRightsConfirmed,
   801→    setGenerationSource,
   802→    setQuality,
   803→    setEditPalette,
   804→    setObjectGroup,
   805→    setRevisionSelect,
   806→    setBuildSetting,
   807→    svgPrompt,
   808→    setSvgPrompt,
   809→    svgAspect,
   810→    setSvgAspect,
   811→    svgPaidConsent,
   812→    setSvgPaidConsent,
   813→    openProjectById,
   814→    createNewProject,
   815→    saveBrief,
   816→    uploadFile,
   817→    uploadSvgFile,
   818→    promoteReference,
   819→    loadSampleProject,
   820→    loadSvgSampleProject,
   821→    sendChat,
   822→    generate,
   823→    generateSvg,
   824→    build,
   825→    runEdit,
   826→    clearSelection,
   827→    startPlacing,
   828→    activateSelectedRevision,
   829→    submitReviewNote,
