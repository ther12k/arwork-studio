    66→type ArtPt = { x: number; y: number };
    67→
    68→/** Drop points closer than `minDist` art units to the previous kept point. */
    69→function dropDensePoints(pts: ArtPt[], minDist: number): ArtPt[] {
    70→  const out: ArtPt[] = [];
    71→  for (const p of pts) {
    72→    const last = out[out.length - 1];
    73→    if (!last || Math.hypot(p.x - last.x, p.y - last.y) >= minDist) out.push(p);
    74→  }
    75→  return out;
    76→}
    77→
    78→function perpendicularDistance(p: ArtPt, a: ArtPt, b: ArtPt): number {
    79→  const dx = b.x - a.x;
    80→  const dy = b.y - a.y;
    81→  const len = Math.hypot(dx, dy);
    82→  if (len < 1e-9) return Math.hypot(p.x - a.x, p.y - a.y);
    83→  return Math.abs(dy * p.x - dx * p.y + b.x * a.y - b.y * a.x) / len;
    84→}
    85→
    86→/** Ramer–Douglas–Peucker simplification (art units). */
    87→function rdp(pts: ArtPt[], epsilon: number): ArtPt[] {
    88→  if (pts.length < 3) return pts;
    89→  const a = pts[0];
    90→  const b = pts[pts.length - 1];
    91→  let maxDist = 0;
    92→  let index = 0;
    93→  for (let i = 1; i < pts.length - 1; i++) {
    94→    const d = perpendicularDistance(pts[i], a, b);
    95→    if (d > maxDist) {
    96→      maxDist = d;
    97→      index = i;
    98→    }
    99→  }
   100→  if (maxDist > epsilon) {
   101→    const left = rdp(pts.slice(0, index + 1), epsilon);
   102→    const right = rdp(pts.slice(index), epsilon);
   103→    return [...left.slice(0, -1), ...right];
   104→  }
   105→  return [a, b];
   106→}
   107→
   108→/** Shoelace area of a closed polygon (art units²). */
   109→function polygonArea(pts: ArtPt[]): number {
   110→  let sum = 0;
   111→  for (let i = 0; i < pts.length; i++) {
   112→    const a = pts[i];
   113→    const b = pts[(i + 1) % pts.length];
   114→    sum += a.x * b.y - b.x * a.y;
   115→  }
   116→  return Math.abs(sum) / 2;
   117→}
   118→
   119→/** Point at half the total arc length of the polyline (stroke midpoint). */
   120→function polylineMidpoint(pts: ArtPt[]): ArtPt | null {
   121→  if (!pts.length) return null;
   122→  const segs: number[] = [];
   123→  let total = 0;
   124→  for (let i = 1; i < pts.length; i++) {
   125→    const d = Math.hypot(pts[i].x - pts[i - 1].x, pts[i].y - pts[i - 1].y);
   126→    segs.push(d);
   127→    total += d;
   128→  }
   129→  if (!segs.length) return pts[0];
   130→  let half = total / 2;
   131→  for (let i = 0; i < segs.length; i++) {
   132→    if (half <= segs[i]) {
   133→      const t = segs[i] > 0 ? half / segs[i] : 0;
   134→      return { x: pts[i].x + (pts[i + 1].x - pts[i].x) * t, y: pts[i].y + (pts[i + 1].y - pts[i].y) * t };
   135→    }
   136→    half -= segs[i];
   137→  }
   138→  return pts[pts.length - 1];
   139→}

   141→const fmt = (n: number) => (Math.round(n * 100) / 100).toString();
   142→
   143→/** Build the edit-action path string: `M x,y L …` (+ ` Z` when closed). */
   144→function pathFromPoints(pts: ArtPt[], close: boolean): string {
   145→  return `M ${pts.map((p) => `${fmt(p.x)},${fmt(p.y)}`).join(" L ")}${close ? " Z" : ""}`;
   146→}

   148→/** Art-space point → canvas-container-relative CSS position (px, clamped
   149→ *  inside the canvas; "50%" center fallback). Anchors the Pen confirm
   150→ *  popover at the drawn shape's centroid. */
   151→function artPointToCanvas(board: VectorBoard, x: number, y: number): { left: number | string; top: number | string } {
   152→  try {
   153→    const ctm = board.svg.getScreenCTM();
   154→    const host = board.svg.parentElement;
   155→    if (!ctm || !host) return { left: "50%", top: "50%" };
   156→    const screen = new DOMPoint(x, y).matrixTransform(ctm);
   157→    const rect = host.getBoundingClientRect();
   158→    if (!rect.width || !rect.height) return { left: "50%", top: "50%" };
   159→    const clamp = (v: number, lo: number, hi: number) => Math.max(lo, Math.min(hi, v));
   160→    return { left: clamp(screen.x - rect.left, 48, rect.width - 48), top: clamp(screen.y - rect.top, 48, rect.height - 48) };
   161→  } catch {
   162→    return { left: "50%", top: "50%" };
   163→  }
   164→}
