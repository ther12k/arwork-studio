     1→"use client";

     3→/**
     4→ * useStudio — page-level state + data flow for the Color Duel Art Studio.
     5→ * Faithful port of the vanilla-JS `web/studio.mjs` controller onto React state.
     6→ *
     7→ * Board specifics (imperative, non-React):
     8→ *  - the <svg> element is rendered ONCE by React with no children; VectorBoard
     9→ *    owns its DOM children and is re-created only when the revision changes;
    10→ *  - `board.paint` is wrapped so "Edit regions" taps select instead of fill;
    11→ *  - `board.clientToArt` is wrapped to capture the last tap point (Place number).
    12→ */

   130→const StudioContext = createContext<StudioApi | null>(null);

   132→export function useStudioContext(): StudioApi {
   133→  const ctx = useContext(StudioContext);
   134→  if (!ctx) throw new Error("useStudioContext must be used inside <StudioProvider>");

   137→
   138→export function StudioProvider({ children }: { children: React.ReactNode }) {
   139→  const [config, setConfig] = useState<StudioConfig | null>(null);
   140→  const [projects, setProjects] = useState<Project[]>([]);
   141→  const [project, setProject] = useState<Project | null>(null);
   142→  const [view, setView] = useState<StudioView>("master");
   143→  const [bundle, setBundle] = useState<Bundle | null>(null);
   144→  const [boardState, setBoardState] = useState<BoardState | null>(null);
   145→  const [selected, setSelected] = useState<Set<string>>(new Set());
   146→  const [placing, setPlacing] = useState(false);
   147→  const [connectionError, setConnectionError] = useState(false);
   148→
   149→  // editable fields
   150→  const [titleInput, setTitleInput] = useState("New illustrated world");
   151→  const [briefInput, setBriefInput] = useState("");
   152→  const [chatInput, setChatInput] = useState("");
   153→  const [paidConsent, setPaidConsent] = useState(false);
   154→  const [rightsConfirmed, setRightsConfirmed] = useState(false);
   155→  const [generationSource, setGenerationSource] = useState<GenerateSource>("brief");
   156→  const [quality, setQuality] = useState<ImageQuality>("medium");
   157→  const [editPalette, setEditPalette] = useState("1");
   158→  const [objectGroup, setObjectGroup] = useState("roof");
   159→  const [revisionSelect, setRevisionSelect] = useState("");
   160→  const [buildSettings, setBuildSettings] = useState<BuildSettings>({
   161→    target_regions: 650,
   162→    palette_colors: 32,
   163→    paint_colors: 80,
   164→    max_edge: 1024,
   165→    compactness: 10,
   166→    min_region_pixels: 35,
   167→    min_label_radius: 3,
   168→    ink_threshold: 40,
   169→  });
   170→
   171→  // refs mirroring state for closures created once
   172→  const projectRef = useRef<Project | null>(null);
   173→  const viewRef = useRef<StudioView>("master");
   174→  const selectedRef = useRef<Set<string>>(new Set());
   175→  const placingRef = useRef(false);
   176→  const boardRef = useRef<VectorBoard | null>(null);
   177→  const bundleRef = useRef<Bundle | null>(null);
   178→  const loadedRevisionRef = useRef<string | null>(null);
   179→  const loadingTokenRef = useRef(0);
   180→  const pollTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   181→  const titleRef = useRef(titleInput);
   182→  const briefRef = useRef(briefInput);
   183→  const chatRef = useRef(chatInput);
   184→  const paidRef = useRef(paidConsent);
   185→  const rightsRef = useRef(rightsConfirmed);
   186→  const generationSourceRef = useRef(generationSource);
   187→  const qualityRef = useRef(quality);
   188→  const buildSettingsRef = useRef(buildSettings);
   189→  const revisionSelectRef = useRef(revisionSelect);
   190→
   191→  // DOM refs
   192→  const svgRef = useRef<SVGSVGElement | null>(null);
   193→  const canvasRef = useRef<HTMLDivElement | null>(null);
   194→  const zoomRef = useRef<HTMLSpanElement | null>(null);
   195→  const messagesRef = useRef<HTMLDivElement | null>(null);
   196→
   197→  // keep field refs in sync
   198→  useEffect(() => void (titleRef.current = titleInput), [titleInput]);
   199→  useEffect(() => void (briefRef.current = briefInput), [briefInput]);
   200→  useEffect(() => void (chatRef.current = chatInput), [chatInput]);
   201→  useEffect(() => void (paidRef.current = paidConsent), [paidConsent]);
   202→  useEffect(() => void (rightsRef.current = rightsConfirmed), [rightsConfirmed]);
   203→  useEffect(() => void (generationSourceRef.current = generationSource), [generationSource]);
   204→  useEffect(() => void (qualityRef.current = quality), [quality]);
   205→  useEffect(() => void (buildSettingsRef.current = buildSettings), [buildSettings]);
   206→  useEffect(() => void (revisionSelectRef.current = revisionSelect), [revisionSelect]);
   207→
   208→  const busy = isBusyProject(project);
   209→  const revision = project?.revisions.find((r) => r.id === project.currentRevision) ?? null;
   210→  const selectionInfo = selected.size
   211→    ? `${selected.size} selected · ${[...selected].slice(0, 4).join(", ")}${selected.size > 4 ? "…" : ""}`
   212→    : "Tap regions to inspect or select them. Drag to pan.";
   213→
   214→  // ---------------------------------------------------------------- helpers
   215→
   216→  const setProjectSync = useCallback((next: Project | null) => {
   217→    projectRef.current = next;
   218→    setProject(next);
   219→    if (next) {
   220→      const current = next.revisions.find((r) => r.id === next.currentRevision);
   221→      const fallback = current?.id ?? next.revisions[0]?.id ?? "";
   222→      setRevisionSelect((prev) => (prev && next.revisions.some((r) => r.id === prev) ? prev : fallback));
   223→    } else {
   224→      setRevisionSelect("");
   225→    }
   226→  }, []);
   227→
   228→  /** Overwrite editable fields from the server (title / brief). */
   229→  const syncFields = useCallback((p: Project) => {
   230→    setTitleInput(p.title);
   231→    setBriefInput(p.brief);
   232→  }, []);
   233→
   234→  const refreshList = useCallback(async (): Promise<Project[]> => {
   235→    const items = await listProjects();
   236→    setProjects(items);
   237→    return items;
   238→  }, []);
   239→
   240→  /** Pure DOM update — selection text is derived during render from `selected`. */
   241→  const highlightSelection = useCallback(() => {
   242→    const board = boardRef.current;
   243→    if (!board) return;
   244→    for (const [id, el] of board.elements) el.classList.toggle("selected-region", selectedRef.current.has(id));
   245→  }, []);
   246→
   247→  const clearBoard = useCallback(() => {
   248→    loadingTokenRef.current++;
   249→    boardRef.current?.destroy();
   250→    boardRef.current = null;
   251→    bundleRef.current = null;
   252→    setBundle(null);
   253→    setBoardState(null);
   254→    loadedRevisionRef.current = null;
   255→    selectedRef.current = new Set();
   256→    setSelected(new Set());
   257→    placingRef.current = false;
   258→    setPlacing(false);
   259→  }, []);
   260→
   261→  // ------------------------------------------------------------- board view
   262→
   263→  const applyBoardView = useCallback(() => {
   264→    const board = boardRef.current;
   265→    if (!board) return;
   266→    board.setPreview(viewRef.current === "colored" || viewRef.current === "inspect");
   267→    highlightSelection();
   268→  }, [highlightSelection]);
   269→
   270→  const handleBoardChange = useCallback(
   271→    (state: BoardState, reason: string) => {
   272→      if (zoomRef.current) zoomRef.current.textContent = `${Math.round(state.zoom * 100)}%`;
   273→      if (reason !== "viewport") setBoardState(state);
   274→      if (reason === "wrong-color") toast("That region needs a different palette group.");
   275→      if (viewRef.current === "inspect") highlightSelection();
   276→    },
   277→    [highlightSelection]
   278→  );
   279→
   280→  // runEdit is called from inside the board paint wrapper; declare it via ref
   281→  // first so the wrapper (created once per board) always calls the latest one.
   282→  const runEditRef = useRef<(action: EditAction, extra?: Partial<EditPayload>) => Promise<void>>(async () => {
   283→    /* replaced below */
   284→  });
   285→
   286→  /** Wrap board interactions for authoring mode (selection instead of paint). */
   287→  const wrapBoard = useCallback(
   288→    (board: VectorBoard) => {
   289→      const normalPaint = board.paint.bind(board);
   290→      board.paint = (id: string | null) => {
   291→        if (viewRef.current !== "inspect") return normalPaint(id);
   292→        if (!id) return "ignored";
   293→        if (placingRef.current) {
   294→          const p = board.lastTapPoint;
   295→          if (!p) return "ignored";
   296→          placingRef.current = false;
   297→          setPlacing(false);
   298→          void runEditRef.current("label", { x: p.x, y: p.y }).catch((e: Error) => toast(e.message));
   299→          return "inspected";
   300→        }
   301→        const next = new Set(selectedRef.current);
   302→        if (next.has(id)) next.delete(id);
   303→        else next.add(id);
   304→        selectedRef.current = next;
   305→        setSelected(next);
   306→        if (next.size === 1) {
   307→          const reg = board.regions.get(id);
   308→          if (reg) {
   309→            setEditPalette(String(reg.paletteId));
   310→            setObjectGroup(!reg.objectId || reg.objectId === "unassigned" ? "roof" : reg.objectId);
   311→          }
   312→        }
   313→        highlightSelection();
   314→        return "inspected";
   315→      };
   316→      const convert = board.clientToArt.bind(board);
   317→      board.clientToArt = (x: number, y: number, matrix?: DOMMatrix | null) => {
   318→        const pt = convert(x, y, matrix);
   319→        board.lastTapPoint = pt;
   320→        return pt;
   321→      };
   322→    },
   323→    [highlightSelection]
   324→  );
   325→
   326→  /** Create the VectorBoard whenever a new bundle arrives (revision change). */
   327→  useEffect(() => {
   328→    if (!bundle || !svgRef.current) return;
   329→    const board = new VectorBoard(svgRef.current, bundle, {
   330→      persist: false,
   331→      mode: "number",
   332→      onChange: handleBoardChange,
   333→    });
   334→    boardRef.current = board;
   335→    wrapBoard(board);
   336→    applyBoardView();
   337→    return () => {
   338→      board.destroy();
   339→      if (boardRef.current === board) boardRef.current = null;
   340→    };
   341→  }, [bundle]);
   342→
   343→  // keep board preview/selection in sync with the active view
   344→  useEffect(() => {
   345→    applyBoardView();
   346→  }, [view, bundle, applyBoardView]);
   347→
   348→  // auto-scroll chat messages
   349→  useEffect(() => {
   350→    const el = messagesRef.current;
   351→    if (el) el.scrollTop = el.scrollHeight;
   352→  }, [project?.messages.length, project?.id]);
   353→
   354→  // re-measure labels when the canvas resizes
   355→  useEffect(() => {
   356→    const el = canvasRef.current;
   357→    if (!el || typeof ResizeObserver === "undefined") return;
   358→    const observer = new ResizeObserver(() => boardRef.current?.updateLabelVisibility());
   359→    observer.observe(el);
   360→    return () => observer.disconnect();
   361→  }, []);
   362→
   363→  // ----------------------------------------------------------------- polling
   364→
   365→  const mountBoard = useCallback(async () => {
   366→    const p = projectRef.current;
   367→    const r = p?.revisions.find((x) => x.id === p.currentRevision);
   368→    if (!p || !r || loadedRevisionRef.current === r.id) return;
   369→    clearBoard();
   370→    const token = loadingTokenRef.current;
   371→    try {
   372→      const b = await loadBundle(p.id, r.id);
   373→      if (token !== loadingTokenRef.current) return;
   374→      bundleRef.current = b;
   375→      loadedRevisionRef.current = r.id;
   376→      setBundle(b);
   377→    } catch (e) {
   378→      toast((e as Error).message);
   379→    }
   380→  }, [clearBoard]);
   381→
   382→  const pollRef = useRef<(pid: string) => Promise<void>>(async () => {
   383→    /* replaced below */
   384→  });
   385→  const poll = useCallback(
   386→    async (pid: string) => {
   387→      if (pollTimerRef.current) clearTimeout(pollTimerRef.current);
   388→      try {
   389→        const next = await getProject(pid);
   390→        if (projectRef.current?.id !== pid) return;
   391→        const changed = projectRef.current.currentRevision !== next.currentRevision;
   392→        setProjectSync(next);
   393→        if (!isBusyProject(next)) syncFields(next);
   394→        if (changed && next.currentRevision) {
   395→          viewRef.current = "colored";
   396→          setView("colored");
   397→          await mountBoard();
   398→          await refreshList();
   399→        } else if (
   400→          !isBusyProject(next) &&
   401→          next.currentRevision &&
   402→          loadedRevisionRef.current !== next.currentRevision
   403→        ) {
   404→          // job finished between the POST response and the first poll — mount the missed revision
   405→          viewRef.current = "colored";
   406→          setView("colored");
   407→          await mountBoard();
   408→          await refreshList();
   409→        }
   410→        if (isBusyProject(next)) {
   411→          pollTimerRef.current = setTimeout(() => void pollRef.current(pid), 900);
   412→        } else if (next.job?.status === "failed") {
   413→          toast(next.job.message || "Job failed");
   414→        }
   415→      } catch (e) {
   416→        toast(`Job polling stopped: ${(e as Error).message}`);
   417→      }
   418→    },
   419→    [mountBoard, refreshList, setProjectSync, syncFields]
   420→  );
   421→  useEffect(() => {
   422→    pollRef.current = poll;
   423→  }, [poll]);
   424→
   425→  const job = useCallback(
   426→    async (post: () => Promise<{ jobId: string }>) => {
   427→      const pid = projectRef.current?.id;
   428→      if (!pid) throw new Error("Create a project first.");
   429→      await post();
   430→      setProjectSync(await getProject(pid));
   431→      void poll(pid);
   432→    },
   433→    [poll, setProjectSync]
   434→  );
   435→
   436→  // ---------------------------------------------------------------- projects
   437→
   438→  const openProject = useCallback(
   439→    async (p: Project) => {
   440→      if (pollTimerRef.current) clearTimeout(pollTimerRef.current);
   441→      clearBoard();
   442→      setProjectSync(p);
   443→      syncFields(p);
   444→      setChatInput("");
   445→      setPaidConsent(false);
   446→      viewRef.current = p.currentRevision ? "colored" : "master";
   447→      setView(viewRef.current);
   448→      try {
   449→        localStorage.setItem(PROJECT_STORAGE_KEY, p.id);
   450→      } catch {
   451→        /* storage unavailable */
   452→      }
   453→      await refreshList();
   454→      if (p.currentRevision) await mountBoard();
   455→      if (isBusyProject(p)) void poll(p.id);
   456→    },
   457→    [clearBoard, mountBoard, poll, refreshList, setProjectSync, syncFields]
   458→  );
   459→
   460→  const ensureProject = useCallback(async () => {
   461→    if (projectRef.current) return projectRef.current;
   462→    const created = await createProject("New illustrated world");
   463→    await openProject(created);
   464→    return created;
   465→  }, [openProject]);
   466→
   467→  const openProjectById = useCallback(
   468→    async (pid: string) => {
   469→      await openProject(await getProject(pid));
   470→    },
   471→    [openProject]
   472→  );
   473→
   474→  const createNewProject = useCallback(async () => {
   475→    const created = await createProject("New illustrated world");
   476→    await openProject(created);
   477→    toast("New project created.");
   478→  }, [openProject]);
   479→
   480→  // -------------------------------------------------------------------- init
   481→
   482→  useEffect(() => {
   483→    let cancelled = false;
   484→    (async () => {
   485→      try {
   486→        const cfg = await getConfig();
   487→        if (cancelled) return;
   488→        setConfig(cfg);
   489→        const items = await refreshList();
   490→        let saved: Project | undefined;
   491→        try {
   492→          const stored = localStorage.getItem(PROJECT_STORAGE_KEY);
   493→          saved = stored ? items.find((p) => p.id === stored) : undefined;
   494→        } catch {
   495→          /* storage unavailable */
   496→        }
   497→        const target = saved ?? items[0];
   498→        if (target) {
   499→          await openProject(target);
   500→        } else {
   501→          const created = await createProject("Cascade Treehouse", DEFAULT_BRIEF);
   502→          await openProject(created);
   503→        }
   504→      } catch (e) {
   505→        if (cancelled) return;
   506→        setConnectionError(true);
   507→        toast((e as Error).message);
   508→      }
   509→    })();
   510→    return () => {
   511→      cancelled = true;
   512→      if (pollTimerRef.current) clearTimeout(pollTimerRef.current);
   513→    };
   514→  }, []);
   515→
   516→  // ----------------------------------------------------------------- actions
   517→
   518→  const saveBrief = useCallback(async () => {
   519→    const p = projectRef.current;
   520→    if (!p) throw new Error("Create a project first.");
   521→    const next = await patchProject(p.id, {
   522→      title: titleRef.current.trim() || "Untitled artwork",
   523→      brief: briefRef.current,
   524→    });
   525→    setProjectSync(next);
   526→    await refreshList();
   527→    return next;
   528→  }, [refreshList, setProjectSync]);
   529→
   530→  const uploadFile = useCallback(
   531→    async (file: File, role: "reference" | "master") => {
   532→      const p = await ensureProject();
   533→      const next = await apiUploadImage(p.id, file, role, rightsRef.current);
   534→      setProjectSync(next);
   535→      syncFields(next);
   536→      if (role === "master") {
   537→        viewRef.current = "master";
   538→        setView("master");
   539→      }
   540→      toast(
   541→        role === "master" ? "Master ready. Build a vector draft." : "Reference added. Use chat to develop an original brief."
   542→      );
   543→    },
   544→    [ensureProject, setProjectSync, syncFields]
   545→  );
   546→
   547→  const promoteReference = useCallback(async () => {
   548→    const p = projectRef.current;
   549→    if (!p) throw new Error("Create a project first.");
   550→    const next = await promoteReferenceApi(p.id, rightsRef.current);
   551→    setProjectSync(next);
   552→    viewRef.current = "master";
   553→    setView("master");
   554→    toast("Reference promoted to master.");
   555→  }, [setProjectSync]);
   556→
   557→  const loadSampleProject = useCallback(async () => {
   558→    const p = await ensureProject();
   559→    const next = await loadSample(p.id);
   560→    setProjectSync(next);
   561→    viewRef.current = "master";
   562→    setView("master");
   563→    toast("Example loaded. Build vector draft to convert it locally.");
   564→  }, [ensureProject, setProjectSync]);
   565→
   566→  const sendChat = useCallback(async () => {
   567→    const p = projectRef.current;
   568→    if (!p) throw new Error("Create a project first.");
   569→    const message = chatRef.current.trim();
   570→    if (!message) throw new Error("Write a direction for the AI first.");
   571→    await saveBrief();
   572→    await job(() => apiSendChat(p.id, { message, include_reference: true, confirm_paid: paidRef.current }));
   573→    setChatInput("");
   574→    setPaidConsent(false);
   575→  }, [job, saveBrief]);
   576→
   577→  const generate = useCallback(async () => {
   578→    const p = projectRef.current;
   579→    if (!p) throw new Error("Create a project first.");
   580→    await saveBrief();
   581→    await job(() =>
   582→      generateMaster(p.id, {
   583→        source: generationSourceRef.current,
   584→        prompt: briefRef.current,
   585→        quality: qualityRef.current,
   586→        size: "1024x1536",
   587→        confirm_paid: paidRef.current,
   588→      })
   589→    );
   590→    setPaidConsent(false);
   591→    viewRef.current = "master";
   592→    setView("master");
   593→  }, [job, saveBrief]);
   594→
   595→  const build = useCallback(async () => {
   596→    const p = projectRef.current;
   597→    if (!p) throw new Error("Create a project first.");
   598→    await saveBrief();
   599→    await job(() => buildDraft(p.id, buildSettingsRef.current));
   600→  }, [job, saveBrief]);
   601→
   602→  const runEdit = useCallback(
   603→    async (action: EditAction, extra: Partial<EditPayload> = {}) => {
   604→      const p = projectRef.current;
   605→      if (!p) throw new Error("Create a project first.");
   606→      if (!selectedRef.current.size) throw new Error("Select at least one region in Edit regions.");
   607→      if (isBusyProject(p)) throw new Error("Wait for the current job.");
   608→      if (!p.currentRevision) throw new Error("Build the vector regions first.");
   609→      const body: EditPayload = {
   610→        base_revision: p.currentRevision,
   611→        action,
   612→        region_ids: [...selectedRef.current],
   613→        ...extra,
   614→      };
   615→      await job(() => apiRunEdit(p.id, body));
   616→      selectedRef.current = new Set();
   617→      setSelected(new Set());
   618→      highlightSelection();
   619→    },
   620→    [highlightSelection, job]
   621→  );
   622→  useEffect(() => {
   623→    runEditRef.current = runEdit;
   624→  }, [runEdit]);
   625→
   626→  const clearSelection = useCallback(() => {
   627→    selectedRef.current = new Set();
   628→    setSelected(new Set());
   629→    placingRef.current = false;
   630→    setPlacing(false);
   631→    highlightSelection();
   632→  }, [highlightSelection]);
   633→
   634→  const startPlacing = useCallback(() => {
   635→    if (selectedRef.current.size !== 1) {
   636→      toast("Select exactly one region first.");
   637→      return;
   638→    }
   639→    placingRef.current = true;
   640→    setPlacing(true);
   641→    toast("Tap a roomy point inside the selected region.");
   642→  }, []);
   643→
   644→  const activateSelectedRevision = useCallback(async () => {
   645→    const p = projectRef.current;
   646→    if (!p) throw new Error("Create a project first.");
   647→    const target = revisionSelectRef.current;
   648→    if (!target) throw new Error("No revision selected.");
   649→    const next = await activateRevision(p.id, target);
   650→    clearBoard();
   651→    viewRef.current = "colored";
   652→    setView("colored");
   653→    setProjectSync(next);
   654→    syncFields(next);
   655→    await mountBoard();
   656→    toast("Revision restored.");
   657→  }, [clearBoard, mountBoard, setProjectSync, syncFields]);
   658→
   659→  const submitReviewNote = useCallback(
   660→    async (note: string) => {
   661→      const p = projectRef.current;
   662→      if (!p) throw new Error("Create a project first.");
   663→      const r = p.revisions.find((x) => x.id === p.currentRevision);
   664→      if (!r) throw new Error("Build a draft first.");
   665→      const next = await apiSubmitReview(p.id, r.id, note);
   666→      setProjectSync(next);
   667→      toast("Visual review recorded.");
   668→    },
   669→    [setProjectSync]
   670→  );
   671→
   672→  const switchView = useCallback(
   673→    (next: StudioView) => {
   674→      if (next !== "master" && !projectRef.current?.currentRevision) {
   675→        toast("Build the vector regions first.");
   676→        return;
   677→      }
   678→      viewRef.current = next;
   679→      setView(next);
   680→      void mountBoard();
   681→    },
   682→    [mountBoard]
   683→  );
   684→
   685→  const setBuildSetting = useCallback((key: keyof BuildSettings, value: number) => {
   686→    setBuildSettings((prev) => ({ ...prev, [key]: value }));
   687→  }, []);
   688→
   689→  const api: StudioApi = {
   690→    config,
   691→    projects,
   692→    project,
   693→    busy,
   694→    revision,
   695→    connectionError,
   696→    view,
   697→    bundle,
   698→    boardState,
   699→    selected,
   700→    placing,
   701→    selectionInfo,
   702→    setBoardPalette: (id: number) => {
   703→      boardRef.current?.setPalette(id);
   704→    },
   705→    findRegion: () => boardRef.current?.nextRegion(),
   706→    undoFill: () => boardRef.current?.undo(),
   707→    resetTest: () => boardRef.current?.reset(),
   708→    svgRef,
   709→    canvasRef,
   710→    zoomRef,
   711→    messagesRef,
   712→    titleInput,
   713→    briefInput,
   714→    chatInput,
   715→    paidConsent,
   716→    rightsConfirmed,
   717→    generationSource,
   718→    quality,
   719→    editPalette,
   720→    objectGroup,
   721→    revisionSelect,
   722→    buildSettings,
   723→    setTitleInput,
   724→    setBriefInput,
   725→    setChatInput,
   726→    setPaidConsent,
   727→    setRightsConfirmed,
   728→    setGenerationSource,
   729→    setQuality,
   730→    setEditPalette,
   731→    setObjectGroup,
   732→    setRevisionSelect,
   733→    setBuildSetting,
   734→    openProjectById,
   735→    createNewProject,
   736→    saveBrief,
   737→    uploadFile,
   738→    promoteReference,
   739→    loadSampleProject,
   740→    sendChat,
   741→    generate,
   742→    build,
   743→    runEdit,
   744→    clearSelection,
   745→    startPlacing,
   746→    activateSelectedRevision,
   747→    submitReviewNote,
   748→    switchView,
   749→    zoomIn: () => boardRef.current?.zoom(1.4),
   750→    zoomOut: () => boardRef.current?.zoom(1 / 1.4),
   751→    fit: () => boardRef.current?.fit(),
   752→  };
   753→
   754→  return <StudioContext.Provider value={api}>{children}</StudioContext.Provider>;
   755→}
